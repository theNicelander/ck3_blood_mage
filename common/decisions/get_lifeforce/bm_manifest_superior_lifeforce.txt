					remove_character_modifier = lifeforce_modifier_minor

				}

			# Failure - Minor lost, Major retained
			25 = {
				desc = manifest_superior_lifeforce_minor_failure_tt
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5

					title = manifest_superior_lifeforce_minor_failure_tt
