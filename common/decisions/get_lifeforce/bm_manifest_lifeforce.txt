bm_manifest_lifeforce_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
	decision_group_type = bm_decision_group

	cost = { piety = 100 }
	cooldown = { years = 1 }

	is_shown = {
		OR = {
			has_trait = lifestyle_blood_mage
			has_trait = lifestyle_blood_knight
		}
	}
	is_valid = {
		piety_level >= 2
	}

	effect = {
		debug_log = "BloodMageDecision: manifest_lifeforce"
		debug_log_scopes = yes
		if = {
			limit = { has_trait = lifestyle_blood_mage }
			add_xp_bm_dynamic = {
				TRACK = ancient
				XP = 3
			}
			duel = {
				skill = learning
				value = 0

				# Success - major lifeforce for mage
				15 = {
					desc = manifest_lifeforce_major_success_tt
					compare_modifier = {
						value = scope:duel_value
						multiplier = 1
					}
					send_interface_toast = {
						title = manifest_lifeforce_major_success_tt
						left_icon = root
						add_character_modifier = { modifier = lifeforce_modifier_major }
					}
				}

				# Partial success - minor lifeforce
				40 = {
					desc = manifest_lifeforce_minor_success_tt
					compare_modifier = {
						value = scope:duel_value
						multiplier = 2
					}
					send_interface_toast = {
						title = manifest_lifeforce_minor_success_tt
						left_icon = root
						add_character_modifier = { modifier = lifeforce_modifier_minor }
					}
				}

				# Failure - nothing happens
				40 = {
					desc = manifest_lifeforce_minor_failure_tt
					send_interface_toast = {
						title = manifest_lifeforce_minor_failure_tt
						left_icon = root
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}
				}

				# Critical failure
				15 = {
					desc = manifest_lifeforce_major_failure_tt
					send_interface_toast = {
						title = manifest_lifeforce_major_failure_tt
						left_icon = root
						temporary_negative_lifeforce_effect_minor = yes
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.25
					}
				}
			}
		}
		else = {
			add_blood_knight_xp = {
				TRACK = resilience
				XP = 3
			}
			duel = {
				skill = prowess
				value = 0

				# Success - major lifeforce for knight
				15 = {
					desc = manifest_lifeforce_major_success_tt
					compare_modifier = {
						value = scope:duel_value
						multiplier = 1
					}
					send_interface_toast = {
						title = manifest_lifeforce_major_success_tt
						left_icon = root
						add_character_modifier = { modifier = lifeforce_modifier_major }
					}
				}

				# Partial success - minor lifeforce
				40 = {
					desc = manifest_lifeforce_minor_success_tt
					compare_modifier = {
						value = scope:duel_value
						multiplier = 2
					}
					send_interface_toast = {
						title = manifest_lifeforce_minor_success_tt
						left_icon = root
						add_character_modifier = { modifier = lifeforce_modifier_minor }
					}
				}

				# Failure - nothing happens
				40 = {
					desc = manifest_lifeforce_minor_failure_tt
					send_interface_toast = {
						title = manifest_lifeforce_minor_failure_tt
						left_icon = root
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}
				}

				# Critical failure
				15 = {
					desc = manifest_lifeforce_major_failure_tt
					send_interface_toast = {
						title = manifest_lifeforce_major_failure_tt
						left_icon = root
						temporary_negative_lifeforce_effect_minor = yes
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.25
					}
				}
			}
		}
	}

	ai_check_interval = 12
	ai_will_do = {
		base = 10
		modifier = {
			piety > 500
			add = 50
		}
		modifier = {
			piety > 1000
			add = 100
		}
		modifier = {
			NOT = { has_character_modifier = lifeforce_modifier_major }
			add = 50
		}
	}
}
