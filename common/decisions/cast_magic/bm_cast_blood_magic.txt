		}

		close_view = {
			view = decisions
			player = root
		}
	}

	ai_check_interval = 6
	ai_will_do = {
		base = 100
		modifier = {
			health <= fine_health
			add = -100
		}
		modifier = {
			add = -80
			OR = {
				any_spouse = {
					is_alive = yes
					NOT = { is_blood_mage_any_tier_trigger = yes }
				}
				primary_heir ?= {
					is_alive = yes
					NOT = { is_blood_mage_any_tier_trigger = yes }
				}
			}
		}
	}
