bm_elevate_blood_mage_tier_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/bm_decision_personal_religious.dds" }
	decision_group_type = bm_decision_group

	desc = bm_elevate_blood_mage_tier_decision_desc
	selection_tooltip = bm_elevate_blood_mage_tier_decision_tooltip
	confirm_text = bm_elevate_blood_mage_tier_decision_confirm

	is_shown = {
		OR = {
			has_trait = lifestyle_bm_acolyte
			has_trait = lifestyle_blood_mage
		}
	}

	is_valid = {
		custom_tooltip = {
			text = bm_elevate_tier_ancient_req
			bm_has_track_xp_any_tier_trigger = { TRACK = ancient VALUE >= 100 }
		}
		trigger_if = {
			limit = { has_trait = lifestyle_bm_acolyte }
			custom_tooltip = {
				text = bm_elevate_tier_devotion_acolyte_req
				piety_level >= 2
			}
			custom_tooltip = {
				text = bm_elevate_tier_rune_acolyte_req
				bm_has_blood_rune_trigger = yes
			}
			custom_tooltip = {
				text = bm_elevate_tier_lifeforce_acolyte_req
				has_character_modifier = lifeforce_modifier_major
			}
			piety >= 250
		}
		trigger_else = {
			custom_tooltip = {
				text = bm_elevate_tier_devotion_mage_req
				piety_level >= 4
			}
			custom_tooltip = {
				text = bm_elevate_tier_rune_mage_req
				OR = {
					has_character_modifier = major_blood_rune_modifier
					has_character_modifier = superior_blood_rune_modifier
				}
			}
			custom_tooltip = {
				text = bm_elevate_tier_lifeforce_mage_req
				has_character_modifier = lifeforce_modifier_superior
			}
			piety >= 500
		}
	}

	cost = {
		piety = {
			value = 0
			if = {
				limit = { has_trait = lifestyle_bm_acolyte }
				add = 250
			}
			else = {
				add = 500
			}
		}
	}

	effect = {
		debug_log = "BloodMageDecision: bm_elevate_blood_mage_tier_decision | Actor: [root.GetShortUINameNoTooltip] (ID:[root.GetID], [Select_CString(root.IsPlayer, 'Player', 'AI')])"
		if = {
			limit = { has_trait = lifestyle_bm_acolyte }
			remove_character_modifier = lifeforce_modifier_major
		}
		else = {
			remove_character_modifier = lifeforce_modifier_superior
		}
		hidden_effect = {
			bm_recount_lifeforce_stacks_effect = yes
		}
		trigger_event = { id = bm_ascension_duel.0001 }
		close_view = {
			view = decisions
			player = root
		}
	}

	ai_potential = {
		OR = {
			has_trait = lifestyle_bm_acolyte
			has_trait = lifestyle_blood_mage
		}
	}

	ai_check_interval = 6
	ai_will_do = {
		base = 100
	}
}
