bm_crimson_empowerment_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
	decision_group_type = bm_decision_group

	cost = { piety = 150 }

	is_shown = {
		has_trait = lifestyle_blood_mage
	}
	is_valid = {
		has_character_modifier = lifeforce_modifier_major
		piety_level >= 1
	}

	effect = {
		debug_log = "BloodMageDecision: crimson_empowerment | Actor: [root.GetShortUINameNoTooltip] (ID:[root.GetID], [Select_CString(root.IsPlayer, 'Player', 'AI')])"
		if = {
			limit = { NOT = { has_trait = lifestyle_crimson_empowerment } }
			add_trait = lifestyle_crimson_empowerment
		}
		trigger_event = { id = bm_crimson_empowerment_event.001 }
		blood_magic_used_major = yes
		add_xp_bm_dynamic = {
			TRACK = enlightenment
			XP = 3
		}
	}

	ai_check_interval = 6
	ai_will_do = {
		base = 100
		modifier = {
			health <= fine_health
			add = -100
		}
	}
}
