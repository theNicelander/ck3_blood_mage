bm_blood_conclave_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/bm_decision_personal_religious.dds" }
	decision_group_type = bm_decision_group

	desc = bm_blood_conclave_decision_desc
	selection_tooltip = bm_blood_conclave_decision_tooltip
	confirm_text = bm_blood_conclave_decision_confirm

	cooldown = { years = 3 }

	cost = {
		piety = 100
	}

	is_shown = {
		is_blood_mage_any_tier_trigger = yes
	}

	is_valid = {
		is_imprisoned = no
		is_incapable = no
		is_in_activity = no
		is_travelling = no
		custom_description = {
			text = bm_blood_conclave_attendees_req
			any_courtier = {
				count >= 2
				is_alive = yes
				is_adult = yes
				is_imprisoned = no
				NOT = { has_trait = incapable }
				is_blood_mage_any_tier_trigger = yes
				NOT = { house ?= house:bm_house_golem }
				NOT = { this = root }
			}
		}
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "BM_BLOOD_CONCLAVE_NEXT_STEP_BUTTON"
		show_from_start = yes

		# Shared Discipline Study
		item = {
			value = bm_conclave_study
			current_description = bm_conclave_study_desc
			localization = bm_conclave_study
			icon = "gfx/interface/icons/trait_level_tracks/wisdom.dds"
			is_shown = {
				is_blood_mage_any_tier_trigger = yes
			}
			is_valid = {
				always = yes
			}
			ai_chance = {
				value = 40
			}
		}

		# Pool Blood Resonance
		item = {
			value = bm_conclave_resonance
			current_description = bm_conclave_resonance_desc
			localization = bm_conclave_resonance
			icon = "gfx/interface/icons/trait_level_tracks/ancient.dds"
			is_shown = {
				is_blood_mage_any_tier_trigger = yes
			}
			is_valid = {
				always = yes
			}
			ai_chance = {
				value = 30
				if = {
					limit = { NOT = { has_character_modifier = lifeforce_modifier_major } }
					add = 40
				}
			}
		}

		# Coven Communion
		item = {
			value = bm_conclave_communion
			current_description = bm_conclave_communion_desc
			localization = bm_conclave_communion
			icon = "gfx/interface/icons/trait_level_tracks/radiance.dds"
			is_shown = {
				is_blood_mage_any_tier_trigger = yes
			}
			is_valid = {
				always = yes
			}
			ai_chance = {
				value = 25
				if = {
					limit = { stress >= 50 }
					add = 40
				}
			}
		}
	}

	effect = {
		if = {
			limit = { scope:bm_conclave_study = yes }
			debug_log = "BloodMageDecision: bm_blood_conclave_decision (study)"
			hidden_effect = { add_character_flag = bm_conclave_focus_study }
			trigger_event = { id = bm_blood_conclave.0001 }
		}
		else_if = {
			limit = { scope:bm_conclave_resonance = yes }
			debug_log = "BloodMageDecision: bm_blood_conclave_decision (resonance)"
			hidden_effect = { add_character_flag = bm_conclave_focus_resonance }
			trigger_event = { id = bm_blood_conclave.0001 }
		}
		else_if = {
			limit = { scope:bm_conclave_communion = yes }
			debug_log = "BloodMageDecision: bm_blood_conclave_decision (communion)"
			hidden_effect = { add_character_flag = bm_conclave_focus_communion }
			trigger_event = { id = bm_blood_conclave.0001 }
		}
		else = {
			custom_tooltip = bm_blood_conclave_decision_tooltip
		}

		close_view = {
			view = decisions
			player = root
		}
	}

	ai_check_interval = 12
	ai_potential = {
		is_blood_mage_any_tier_trigger = yes
	}
	ai_will_do = {
		base = 30
		modifier = {
			health <= fine_health
			factor = 0
		}
	}
}
