channel_lifeforce_bloodline = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }

    decision_group_type = bm_decision_group

    cooldown = { years = 3 }
    cost = { piety = 350 }

    is_shown = { has_trait = lifestyle_blood_mage }
    is_valid = {
        has_character_modifier = lifeforce_modifier_major
        piety_level >= 1
    }
    effect = {
        debug_log = "BloodMageDecision: channel_lifeforce_bloodline | Actor: [root.GetShortUINameNoTooltip] (ID:[root.GetID], [Select_CString(root.IsPlayer, 'Player', 'AI')])"
        trigger_event = { id = bm_channel_lifeforce_bloodline.001 }
        add_xp_bm_dynamic = {
            TRACK = bloodline
            XP = 5
        }
        blood_magic_used_major = yes
        custom_tooltip = channel_lifeforce_bloodline.effects_list
    }
    ai_check_interval = 6
    ai_will_do = {
        base = 50
        ai_modifier_based_on_health = yes
    }
}

bm_manifest_lifeforce_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
	decision_group_type = bm_decision_group

	cost = { piety = 100 }
	cooldown = { years = 1 }

	is_shown = {
		has_trait = lifestyle_blood_mage
	}
	is_valid = {
		piety_level >= 2
	}

	effect = {
		debug_log = "BloodMageDecision: manifest_lifeforce | Actor: [root.GetShortUINameNoTooltip] (ID:[root.GetID], [Select_CString(root.IsPlayer, 'Player', 'AI')])"
		add_xp_bm_dynamic = {
			TRACK = enlightenment
			XP = 5
		}
		duel = {
			skill = learning
			value = 0

			# Success - major lifeforce
			15 = {
				desc = manifest_lifeforce_major_success_tt
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1
				}
				send_interface_toast = {
					title = manifest_lifeforce_major_success_tt
					left_icon = root
					add_character_modifier = { modifier = lifeforce_modifier_major }
				}
			}

			# Partial success - minor lifeforce
			40 = {
				desc = manifest_lifeforce_minor_success_tt
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2
				}
				send_interface_toast = {
					title = manifest_lifeforce_minor_success_tt
					left_icon = root
					add_character_modifier = { modifier = lifeforce_modifier_minor }
				}
			}

			# Failure - nothing happens
			40 = {
				desc = manifest_lifeforce_minor_failure_tt
				send_interface_toast = {
					title = manifest_lifeforce_minor_failure_tt
					left_icon = root
				}
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
			}

			# Critical failure
			15 = {
				desc = manifest_lifeforce_major_failure_tt
				send_interface_toast = {
					title = manifest_lifeforce_major_failure_tt
					left_icon = root
					temporary_negative_lifeforce_effect_minor = yes
				}
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.25
				}
			}
		}
	}

	ai_check_interval = 12
	ai_will_do = {
		base = 10
		modifier = {
			piety > 2000
			add = 1000
		}
		modifier = {
			NOT = { has_character_modifier = lifeforce_modifier_major }
			add = 50
		}
	}
}
