bm_enhance_education_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
    desc = bm_enhance_education_decision_desc
    selection_tooltip = bm_enhance_education_decision_tooltip
    decision_group_type = bm_decision_group

    cooldown = { years = 2 }
    cost = { piety = scaled_education_enhancement_cost }

    is_shown = {
        has_trait = lifestyle_blood_mage
        OR = { # Has a non level 5 education trait
            AND = {
                has_trait = education_diplomacy
                NOT = { has_trait = education_diplomacy_5 }
            }
            AND = {
                has_trait = education_martial
                NOT = { has_trait = education_martial_5 }
            }
            AND = {
                has_trait = education_learning
                NOT = { has_trait = education_learning_5 }
            }
            AND = {
                has_trait = education_stewardship
                NOT = { has_trait = education_stewardship_5 }
            }
            AND = {
                has_trait = education_intrigue
                NOT = { has_trait = education_intrigue_5 }
            }
        }
    }

    is_valid = {
        custom_description = {
            text = bm_cooldown_active_tt
            NOT = { has_character_flag = bm_enhance_education_cooldown }
        }
        has_character_modifier = lifeforce_modifier_major
        piety_level >= 3
        custom_description = {
            text = "bm_cast_blood_magic_self_major_improve_education_xp_requirement"
            OR = {
                has_trait_xp = {
                    trait = lifestyle_crimson_empowerment
                    track = charisma
                    value >= required_xp_improve_education
                }
                has_trait_xp = {
                    trait = lifestyle_crimson_empowerment
                    track = fury
                    value >= required_xp_improve_education
                }
                has_trait_xp = {
                    trait = lifestyle_crimson_empowerment
                    track = prosperity
                    value >= required_xp_improve_education
                }
                has_trait_xp = {
                    trait = lifestyle_crimson_empowerment
                    track = shadows
                    value >= required_xp_improve_education
                }
                has_trait_xp = {
                    trait = lifestyle_crimson_empowerment
                    track = insight
                    value >= required_xp_improve_education
                }
                has_trait_xp = {
                    trait = lifestyle_crimson_empowerment
                    track = legacy
                    value >= required_xp_improve_education
                }
                has_trait_xp = {
                    trait = lifestyle_crimson_empowerment
                    track = expertise
                    value >= required_xp_improve_education
                }
            }
        }
    }

    effect = {
        debug_log = debug_log.blood_mage_log
        add_character_flag = { flag = bm_enhance_education_cooldown years = 2 }
        blood_magic_used_major = yes
        add_xp_bm_dynamic = {
            TRACK = enlightenment
            XP = 5
        }
        trigger_event = { id = bm_education_enhancement.001 }
    }

    ai_check_interval = 12
    ai_will_do = {
        base = 100
        modifier = {
            add = -100
            health <= fine_health
        }
    }
}
