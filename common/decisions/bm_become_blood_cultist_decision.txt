become_blood_cultist_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }
    desc = become_blood_cultist_decision_desc
    selection_tooltip = become_blood_cultist_decision_tooltip
    confirm_text = become_blood_cultist_decision_confirm
    decision_group_type = major

    cooldown = { years = 10 }

    is_shown = {
        has_trait = lifestyle_blood_mage
        NOR = {
            bm_is_blood_cult_faith_trigger = yes
        }
        # Disable for overhaul mods
        NOT = { is_playing_overhaul_mod = yes }

        OR = {
            has_game_rule = blodtru_religion_enabled
            AND = {
                has_game_rule = blodtru_religion_enabled_player_only
                is_ai = no
            }
        }
    }
    is_valid = {
        has_trait = lifestyle_blood_mage
        has_character_modifier = lifeforce_modifier_major
        piety_level >= 3
    }

    effect = {
        debug_log = debug_log.blood_mage_log

        custom_tooltip = become_blood_cultist_decision_effect_tooltip
        trigger_event = bm_faith_conversion.0001
    }

    # Yearly chance AI take up the religion if they are a blood mage
    ai_check_interval = 12
    ai_will_do = {
        base = 10
        modifier = {
            add = -100
            OR = {
                has_trait = zealous
                has_trait = content
            }
        }
        modifier = {
            add = 10
            OR = {
                has_trait = ambitious
                has_trait = lifestyle_mystic
            }
        }
    }
}
