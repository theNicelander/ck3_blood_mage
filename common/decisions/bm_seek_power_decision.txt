seek_power_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }

    decision_group_type = bm_decision_group

    is_shown = {
        has_trait = lifestyle_blood_mage
        is_landless_adventurer = no
    }
    is_valid = {
        custom_description = {
            text = bm_cooldown_active_tt
            NOT = { has_character_flag = bm_seek_power_cooldown }
        }
    }
    cooldown = { years = 2 }
    effect = {
        debug_log = debug_log.blood_mage_log
        add_character_flag = { flag = bm_seek_power_cooldown years = 2 }
        trigger_event = { id = seek_power.001 }
    }

    ai_check_interval = 3
    ai_will_do = { base = 100 }
}

    picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }

    decision_group_type = bm_decision_group

    is_shown = {
        has_trait = lifestyle_blood_mage
        is_landless_adventurer = yes
    }
    is_valid = {
        custom_description = {
            text = bm_cooldown_active_tt

        debug_log = debug_log.blood_mage_log

    }

    ai_check_interval = 3
    ai_potential = { always = yes }
}
