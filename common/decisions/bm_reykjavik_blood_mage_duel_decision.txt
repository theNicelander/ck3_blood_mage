# Challenge the Blood Mage of Reykjavik
bm_reykjavik_blood_mage_duel_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
	desc = bm_reykjavik_blood_mage_duel_decision_desc
	selection_tooltip = bm_reykjavik_blood_mage_duel_decision_tooltip
	confirm_text = bm_reykjavik_blood_mage_duel_decision_confirm
	decision_group_type = major
	ai_check_interval = 60

	is_shown = {
		is_adult = yes
		NOT = { has_trait = lifestyle_blood_mage }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no
		custom_tooltip = {
			text = bm_reykjavik_blood_mage_duel_location_requirement_tt
			location.county ?= title:c_vestisland
		}
	}

	cooldown = { years = 5 }
	cost = {
		piety = bm_reykjavik_duel_piety_cost
	}

	effect = {
		hidden_effect = {
			create_character = {
				age = { 35 45 }
				gender_female_chance = 50
				location = root.location
				culture = root.culture
				faith = faith:quintessence_faith
				trait = lifestyle_blood_mage
				trait = education_learning_4
				learning = 22
				prowess = 18
				save_scope_as = bm_reykjavik_rival
			}
		}

		duel = {
			skill = learning
			target = scope:bm_reykjavik_rival

			# SUCCESS
			0 = {
				desc = bm_reykjavik_blood_mage_duel_decision.success_tt

				compare_modifier = {
					value = scope:duel_value
					multiplier = 2
				}
				modifier = {
					add = bm_reykjavik_duel_prowess_bonus_value
				}

				send_interface_toast = {
					title = bm_reykjavik_blood_mage_duel_decision.success_tt
					left_icon = root
					bm_become_blood_mage_effect = yes
				}
			}

			# FAILURE - MENTAL COLLAPSE
			75 = {
				desc = bm_reykjavik_blood_mage_duel_decision.mental_failure_tt
				send_interface_toast = {
					title = bm_reykjavik_blood_mage_duel_decision.mental_failure_tt
					left_icon = root
					add_stress = 30

					if = {
						limit = { has_trait = lunatic_1 }
						add_trait = dull
					}
					if = {
						limit = { has_trait = depressed_1 }
						add_trait = lunatic_1
					}
					if = {
						limit = { has_trait = possessed_1 }
						add_trait = depressed_1
					}
					add_trait = possessed_1
				}
			}

			# FAILURE - PHYSICAL TRAUMA
			75 = {
				desc = bm_reykjavik_blood_mage_duel_decision.physical_failure_tt
				send_interface_toast = {
					title = bm_reykjavik_blood_mage_duel_decision.physical_failure_tt
					left_icon = root
					add_stress = 20
					if = {
						limit = { has_trait = blind }
						add_piety_level = -1
					}
					if = {
						limit = { has_trait = one_eyed }
						add_trait = blind
					}
					if = {
						limit = { has_trait = weak }
						add_trait = one_eyed
					}
					add_trait = weak
				}
			}

			# FAILURE - INJURY
			150 = {
				desc = bm_reykjavik_blood_mage_duel_decision.injury_failure_tt
				send_interface_toast = {
					title = bm_reykjavik_blood_mage_duel_decision.injury_failure_tt
					left_icon = root
					add_stress = 10

					if = {
						limit = { has_trait = wounded_2 }
						add_trait = maimed
					}

					if = {
						limit = { NOT = { has_trait = wounded_1 } }
						add_trait = wounded_1
					}
					else = {
						change_trait_rank = {
							trait = wounded
							rank = 1
							max = 2
						}
					}
				}
			}
		}

		hidden_effect = {
			if = {
				limit = { exists = scope:bm_reykjavik_rival }
				scope:bm_reykjavik_rival = {
					death = { death_reason = death_vanished }
				}
			}
		}
	}

	ai_will_do = {
		base = 2
		bm_blood_mage_prevalence_ai_acquisition_modifier = yes
	}
}
