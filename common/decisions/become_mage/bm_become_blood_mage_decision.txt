	decision_group_type = major
	ai_check_interval = 12
	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }

	desc = bm_enhance_blood_ritual_decision_desc
	selection_tooltip = bm_enhance_blood_ritual_decision_tooltip
	confirm_text = bm_enhance_blood_ritual_decision_confirm

	is_shown = {
		is_adult = yes
		NOT = { has_trait = lifestyle_blood_mage }
		NOT = { has_trait = lifestyle_blood_knight }
	}

	is_valid = {
		piety_level >= 2
		OR = {
			learning >= 20
			prowess >= 20
			martial >= 20

			has_trait = education_martial_4
			has_trait = education_martial_5
			has_trait = intellect_good_3
			has_trait = physique_good_3
			has_trait = lifestyle_mystic
			has_trait = lifestyle_scholar
			has_trait = lifestyle_blademaster
			has_trait = berserker
			has_trait = theologian
			stress_level >= 2
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable = no
		custom_tooltip = {
			text = bm_enhance_blood_ritual_location_requirement_tt
			bm_is_at_blood_magic_initiation_site_trigger = yes
		}
	}

	effect = {
		debug_log = "BloodMageDecision: enhance_blood_ritual | Actor: [root.GetShortUINameNoTooltip] (ID:[root.GetID], [Select_CString(root.IsPlayer, 'Player', 'AI')])"
		debug_log_scopes = yes
		trigger_event = { id = bm_geyser_duel.0001 }
	}

	ai_will_do = {
		base = 10
		bm_blood_mage_prevalence_ai_acquisition_modifier = yes
	}
}
