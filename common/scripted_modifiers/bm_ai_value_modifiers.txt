# Lifedraining
ai_willingness_to_lifedrain = {
    # 1. Additive base modifiers & opinion weighting (evaluated FIRST)
    # Opinion counts (more willing to drain those you dislike)
    opinion_modifier = {
        who = scope:actor
        opinion_target = scope:recipient
        multiplier = -0.5
    }

    # Motivated to drain rivals and nemeses
    modifier = {
        OR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
        add = 100
    }

    # Disinclined to drain characters of equal or higher title tier than actor
    modifier = {
        scope:recipient.highest_held_title_tier >= scope:actor.highest_held_title_tier
        add = -150
    }

    # Disinclined to drain house members
    modifier = {
        scope:actor = {
            exists = house
            exists = scope:recipient.house
            house = scope:recipient.house
        }
        add = -100
    }

    # 2. Absolute vetoes (factor = 0 evaluated LAST so nothing can overwrite zero)
    # Absolute veto: Never drain spouse or consort under any circumstances
    modifier = {
        factor = 0
        scope:actor = {
            OR = {
                is_spouse_of = scope:recipient
                is_consort_of = scope:recipient
            }
        }
    }

    # Absolute veto: Never drain your children under any circumstances
    modifier = {
        factor = 0
        scope:recipient = { is_child_of = scope:actor }
    }

    # Absolute veto: Never drain close family under any circumstances
    modifier = {
        factor = 0
        scope:actor = { is_close_family_of = scope:recipient }
    }

    # Absolute veto: Never drain councillors, knights/champions, or powerful vassals
    modifier = {
        factor = 0
        OR = {
            scope:recipient = { is_knight_of = scope:actor }
            scope:recipient = { is_councillor_of = scope:actor }
            scope:recipient = { is_powerful_vassal_of = scope:actor }
        }
    }

    # Absolute veto: Never drain court position holders (physicians, bodyguards, seneschal, caravan master, etc.)
    modifier = {
        factor = 0
        scope:recipient = {
            any_court_position_employer = { this = scope:actor }
        }
    }

    # Absolute veto: Never drain blood golems
    modifier = {
        factor = 0
        scope:recipient = { house ?= house:bm_house_golem }
    }

    # Absolute veto: Never drain friends, best friends, soulmates, lovers, or wards
    modifier = {
        factor = 0
        scope:actor = {
            OR = {
                has_relation_friend = scope:recipient
                has_relation_best_friend = scope:recipient
                has_relation_soulmate = scope:recipient
                has_relation_lover = scope:recipient
                has_relation_ward = scope:recipient
                has_secret_relation_lover = scope:recipient
            }
        }
    }

    # Absolute veto: Never drain your liege (unless rival/nemesis)
    modifier = {
        factor = 0
        scope:actor = {
            target_is_liege_or_above = scope:recipient
        }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }

    # Absolute veto: Never drain your dynasty members (unless rival/nemesis)
    modifier = {
        factor = 0
        scope:actor = {
            exists = dynasty
            exists = scope:recipient.dynasty
            dynasty = scope:recipient.dynasty
        }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }

    # Absolute veto: Never drain people you actually like (positive opinion), unless rival/nemesis
    modifier = {
        factor = 0
        scope:actor = {
            opinion = {
                target = scope:recipient
                value > 0
            }
        }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }
}

ai_willingness_to_do_positive_magic = {
    modifier = { # Prioritise myself
        scope:actor = scope:recipient
        add = 100
    }
    modifier = { # don't do magic, if dying
        add = -100
        scope:actor = { health < good_health }
    }

    # Opinion counts
    opinion_modifier = {
        trigger = {
            NOT = { scope:actor = scope:recipient }
        }
        who = scope:actor
        opinion_target = scope:recipient
        multiplier = 0.2
    }

    # Disliked targets penalty
    modifier = {
        NOT = { scope:actor = scope:recipient }
        scope:actor = {
            opinion = {
                target = scope:recipient
                value < 0
            }
        }
        add = -50
    }

    # Liege priority
    modifier = {
        scope:actor = { target_is_liege_or_above = scope:recipient }
        add = 50
    }

    # Councillor priority
    modifier = {
        scope:recipient = { is_councillor_of = scope:actor }
        add = 50
    }

    # Spouse / Consort priority
    modifier = {
        scope:actor = {
            OR = {
                is_spouse_of = scope:recipient
                is_consort_of = scope:recipient
            }
        }
        add = 200
    }

    # Lover / Soulmate priority
    modifier = {
        scope:actor = {
            OR = {
                has_relation_soulmate = scope:recipient
                has_relation_lover = scope:recipient
                has_secret_relation_lover = scope:recipient
            }
        }
        add = 60
    }

    # Dynasty priority
    modifier = {
        scope:actor = {
            exists = dynasty
            exists = scope:recipient.dynasty
            dynasty = scope:recipient.dynasty
        }
        add = 40
    }

    # House priority
    modifier = {
        scope:actor = {
            exists = house
            exists = scope:recipient.house
            house = scope:recipient.house
        }
        add = 30
    }

    # Children priority
    modifier = {
        scope:recipient = { is_child_of = scope:actor }
        add = 50
    }

    # Primary heir priority
    modifier = {
        scope:recipient = { is_primary_heir_of = scope:actor }
        add = 150
    }

    # Close family priority
    modifier = {
        scope:actor = { is_close_family_of = scope:recipient }
        add = 25
    }

    # Friends priority
    modifier = {
        scope:actor = {
            OR = {
                has_relation_friend = scope:recipient
                has_relation_best_friend = scope:recipient
            }
        }
        add = 35
    }

    # Knights priority
    modifier = {
        scope:recipient = { is_knight_of = scope:actor }
        add = 40
    }

    # Powerful vassals and courtiers
    modifier = {
        scope:recipient = { is_powerful_vassal_of = scope:actor }
        add = 20
    }
    modifier = {
        scope:recipient = { is_courtier_of = scope:actor }
        add = 20
    }

    # Higher title tier
    modifier = {
        add = {
            value = scope:recipient.highest_held_title_tier
            subtract = scope:actor.highest_held_title_tier
            multiply = 20
        }
    }

    # Absolute veto: Never help rivals or nemeses (placed at bottom to prevent resurrection by additive modifiers)
    modifier = {
        factor = 0
        NOT = { scope:actor = scope:recipient }
        OR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }
}
