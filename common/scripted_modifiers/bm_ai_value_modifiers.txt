ai_modifier_based_on_health  = {
    modifier = {
        add = 100
        root = { health >= excellent_health }
    }
    modifier = {
        add = 40
        root = { health >= good_health }
    }
    modifier = {
        add = 20
        root = { health >= fine_health }
    }
    modifier = { # don't do magic, if dying
        add = -250
        root = { health <= poor_health }
    }
}

# Lifedraining
ai_willingness_to_lifedrain = {
    # Absolute veto: Never drain your close family (unless rival/nemesis)
    modifier = {
        factor = 0
        scope:actor = { is_close_family_of = scope:recipient }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }

    # Absolute veto: Never drain your children (unless rival/nemesis)
    modifier = {
        factor = 0
        scope:recipient = { is_child_of = scope:actor }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }

    # Absolute veto: Never drain spouse/consort (unless rival/nemesis or exposed cheat)
    modifier = {
        factor = 0
        scope:actor = {
            OR = {
                is_spouse_of = scope:recipient
                is_consort_of = scope:recipient
            }
        }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
            scope:recipient = { exposed_cheating_on_spouse_trigger = { SPOUSE = scope:actor } }
        }
    }

    # Absolute veto: Never drain friends, best friends, soulmates, lovers, or wards
    modifier = {
        factor = 0
        scope:actor = {
            OR = {
                has_relation_friend = scope:recipient
                has_relation_best_friend = scope:recipient
                has_relation_soulmate = scope:recipient
                has_relation_lover = scope:recipient
                has_relation_ward = scope:recipient
                has_secret_relation_lover = scope:recipient
            }
        }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }

    # Absolute veto: Never drain councillors, knights/champions, or powerful vassals
    modifier = {
        factor = 0
        OR = {
            scope:recipient = { is_knight_of = scope:actor }
            scope:recipient = { is_councillor_of = scope:actor }
            scope:recipient = { is_powerful_vassal_of = scope:actor }
        }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }

    # Absolute veto: Never drain people you actually like (opinion > 20)
    modifier = {
        factor = 0
        scope:actor = {
            opinion = {
                target = scope:recipient
                value > low_positive_opinion
            }
        }
        NOR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
    }

    # Opinion counts (more willing to drain those you dislike)
    opinion_modifier = {
        who = scope:actor
        opinion_target = scope:recipient
        multiplier = -0.2
    }

    # Disinclined to drain characters of higher title tier than actor
    modifier = {
        add = {
            value = scope:recipient.highest_held_title_tier
            subtract = scope:actor.highest_held_title_tier
            multiply = -100
        }
    }

    # Disinclined to drain house members
    modifier = {
        scope:actor = {
            exists = house
            exists = scope:recipient.house
            house = scope:recipient.house
        }
        add = -100
    }

    # Motivated to drain rivals and nemeses
    modifier = {
        OR = {
            scope:recipient = { has_relation_rival = scope:actor }
            scope:recipient = { has_relation_nemesis = scope:actor }
        }
        add = 100
    }
}

ai_willingness_to_do_postitive_magic = {
    modifier = { # Prioritise myself
        scope:actor = scope:recipient
        add = 100
    }
    modifier = { # don't do magic, if dying
        add = -100
        scope:actor = { health < good_health }
    }

    # Opinion counts
    opinion_modifier = {
        who = scope:actor
        opinion_target = scope:recipient
        multiplier = 0.2
    }

    # Are they important (to me)
    modifier = {
        add = {
            value = scope:recipient.highest_held_title_tier
            subtract = scope:actor.highest_held_title_tier
            multiply = 20
        }
    }
    modifier = {
        OR = {
            scope:actor = { is_close_family_of = scope:recipient }
            scope:actor = { has_relation_friend = scope:recipient }
        }
        add = 20
    }
    modifier = {
        OR = {

            scope:actor = { has_relation_soulmate = scope:recipient }
            scope:actor = { is_consort_of = scope:recipient }
            scope:actor = { has_relation_lover = scope:recipient }
        }
        add = 50
    }
    modifier = {
        scope:actor = {
            exists = house
            exists = scope:recipient.house
            house = scope:recipient.house
        }
        add = 30
    }
    modifier = {
        scope:actor = { has_relation_best_friend = scope:recipient }
        scope:recipient = { is_knight_of = scope:actor }
        add = 40
    }
    modifier = {
        scope:recipient = { is_powerful_vassal_of = scope:actor }
        scope:recipient = { is_courtier_of = scope:actor }
        add = 20
    }
}
