bm_captive_price_value = {
	value = 50
	add = {
		value = prowess
		multiply = 5
	}
	if = {
		limit = { prowess >= 20 }
		add = 50
	}
	if = {
		limit = {
			OR = {
				has_trait = physique_good_1
				has_trait = beauty_good_1
				has_trait = intellect_good_1
				has_trait = strong
			}
		}
		add = 75
	}
	if = {
		limit = {
			OR = {
				has_trait = physique_good_2
				has_trait = beauty_good_2
				has_trait = intellect_good_2
				has_trait = giant
				has_trait = lifestyle_blademaster
			}
		}
		add = 150
	}
	if = {
		limit = {
			OR = {
				has_trait = physique_good_3
				has_trait = beauty_good_3
				has_trait = intellect_good_3
				has_trait = lifestyle_blood_mage
			}
		}
		add = 300
	}
}

bm_captive_all_remaining_price_value = {
	value = 0
	if = {
		limit = {
			exists = scope:captive_1
			scope:captive_1 = {
				is_alive = yes
				is_imprisoned = no
			}
		}
		add = scope:captive_1.bm_captive_price_value
	}
	if = {
		limit = {
			exists = scope:captive_2
			scope:captive_2 = {
				is_alive = yes
				is_imprisoned = no
			}
		}
		add = scope:captive_2.bm_captive_price_value
	}
	if = {
		limit = {
			exists = scope:captive_3
			scope:captive_3 = {
				is_alive = yes
				is_imprisoned = no
			}
		}
		add = scope:captive_3.bm_captive_price_value
	}
}
