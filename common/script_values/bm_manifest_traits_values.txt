# Manifest Traits Costs and Duel Modifiers

# Scaled Piety Costs
bm_manifest_traits_intellect_piety_cost = {
	if = {
		limit = { has_trait = intellect_bad_3 }
		value = 200
	}
	else_if = {
		limit = { has_trait = intellect_bad_2 }
		value = 250
	}
	else_if = {
		limit = { has_trait = intellect_bad_1 }
		value = 300
	}
	else_if = {
		limit = { NOT = { has_trait = intellect_good } }
		value = 450
	}
	else_if = {
		limit = { has_trait = intellect_good_1 }
		value = 600
	}
	else_if = {
		limit = { has_trait = intellect_good_2 }
		value = 750
	}
	else_if = {
		limit = { has_trait_rank = { trait = intellect_good rank = 3 } }
		value = 900
	}
	else_if = {
		limit = { has_trait_rank = { trait = intellect_good rank >= 4 } }
		value = 1050
	}
	else = {
		value = 750
	}
}

# Per-Branch Target Tier Failure Additions (higher target tier = harder)
bm_manifest_traits_intellect_tier_failure_addition = {
	value = 0
	if = {
		limit = { has_trait_rank = { trait = intellect_good rank >= 4 } }
		add = 200
	}
	else_if = {
		limit = { has_trait_rank = { trait = intellect_good rank = 3 } }
		add = 150
	}
	else_if = {
		limit = { has_trait = intellect_good_2 }
		add = 110
	}
	else_if = {
		limit = { has_trait = intellect_good_1 }
		add = 60
	}
	else_if = {
		limit = {
			NOT = { has_trait = intellect_good }
			NOT = { has_trait = intellect_bad }
		}
		add = 30
	}
}
