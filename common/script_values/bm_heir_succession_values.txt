bm_anointed_transfer_ancient_xp_value = {
	value = 0
	scope:dying_ruler = {
		if = {
			limit = { has_trait = lifestyle_blood_mage }
			add = "has_trait_xp(lifestyle_blood_mage|ancient)"
		}
		else_if = {
			limit = { has_trait = lifestyle_blood_knight }
			add = "has_trait_xp(lifestyle_blood_knight|ancient)"
		}
	}
	multiply = 0.25
	round = yes
	min = 0
}
