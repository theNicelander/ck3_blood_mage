bm_anointed_transfer_ancient_xp_value = {
	value = 0
	scope:dying_ruler = {
		if = {
			limit = { has_trait = lifestyle_bm_acolyte }
			add = "has_trait_xp(lifestyle_bm_acolyte|ancient)"
		}
		else_if = {
			limit = { has_trait = lifestyle_blood_mage }
			add = "has_trait_xp(lifestyle_blood_mage|ancient)"
		}
		else_if = {
			limit = { has_trait = lifestyle_bm_archmage }
			add = "has_trait_xp(lifestyle_bm_archmage|ancient)"
		}
	}
	multiply = 0.25
	round = yes
	min = 0
}
