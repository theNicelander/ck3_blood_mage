on_piety_level_gain = {
	effect = {
		if = {
			limit = {
				has_trait = lifestyle_blood_knight
				has_trait_xp = {
					trait = lifestyle_blood_knight
					track = slaughter
					value >= 50
				}
			}
			bm_update_slaughter_devotion_knights_effect = yes
		}
	}
}

on_piety_level_loss = {
	effect = {
		if = {
			limit = {
				has_trait = lifestyle_blood_knight
				has_trait_xp = {
					trait = lifestyle_blood_knight
					track = slaughter
					value >= 50
				}
			}
			bm_update_slaughter_devotion_knights_effect = yes
		}
	}
}
