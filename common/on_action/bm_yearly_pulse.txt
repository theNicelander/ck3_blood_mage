yearly_playable_pulse = {
    on_actions = {
        bm_yearly_pulse_playable
    }
}

random_yearly_everyone_pulse = {
    on_actions = {
        bm_yearly_pulse_everyone
        bm_blood_golem_yearly_decay
        bm_review_unmarked_blood_mage_prevalence
        bm_yearly_interpersonal_pulse
    }
}

bm_yearly_pulse_playable = {
    trigger = {
        OR = {
            is_blood_mage_any_tier_trigger = yes
            bm_has_blood_rune_trigger = yes
        }
    }
    effect = {
        trigger_event = blood_mage_yearly_events.001
        trigger_event = blood_mage_yearly_events.002
        trigger_event = blood_mage_yearly_events.003
        if = {
            limit = {
                is_ai = no
                is_blood_mage_any_tier_trigger = yes
                age >= 85
                NOT = { has_character_flag = bm_age_85_milestone_done }
            }
            trigger_event = bm_interpersonal_events.0007
        }
        if = {
            limit = {
                is_ai = no
                is_blood_mage_any_tier_trigger = yes
                age >= 100
                NOT = { has_character_flag = bm_age_100_milestone_done }
            }
            trigger_event = bm_interpersonal_events.0011
        }
    }
}

bm_yearly_pulse_everyone = {
    trigger = {
        is_playable_character = no
        OR = {
            is_blood_mage_any_tier_trigger = yes
            bm_has_blood_rune_trigger = yes
        }
    }
    effect = {
        trigger_event = blood_mage_yearly_events.001
        trigger_event = blood_mage_yearly_events.002
        trigger_event = blood_mage_yearly_events.003
    }
}

bm_blood_golem_yearly_decay = {
    trigger = {
        has_trait = bm_blood_golem
    }
    effect = {
        add_trait_xp = {
            trait = bm_blood_golem
            value = -2
        }
    }
}


bm_review_unmarked_blood_mage_prevalence = {
    trigger = {
        is_ai = yes
        is_blood_mage_any_tier_trigger = yes
        NOT = { has_character_flag = bm_blood_mage_prevalence_reviewed }
        OR = {
            has_game_rule = bm_prevalence_player_only
            has_game_rule = bm_prevalence_one_in_10000
            has_game_rule = bm_prevalence_one_in_1000
        }
    }
    effect = {
        bm_apply_blood_mage_prevalence_retention_effect = yes
    }
}

bm_yearly_interpersonal_pulse = {
    trigger = {
        is_ai = no
        is_blood_mage_any_tier_trigger = yes
        NOT = { has_character_flag = bm_interpersonal_event_cooldown }
    }
    effect = {
        random_list = {
            100 = {
                trigger = {
                    OR = {
                        any_relation = { type = soulmate is_alive = yes }
                        any_relation = { type = lover is_alive = yes }
                        any_spouse = { is_alive = yes }
                        any_consort = { is_alive = yes }
                    }
                }
                trigger_event = bm_interpersonal_events.0001
            }
            100 = {
                trigger = {
                    OR = {
                        any_child = {
                            is_alive = yes
                            age >= 6
                            age <= 15
                        }
                        any_relation = {
                            type = ward
                            is_alive = yes
                            age >= 6
                            age <= 15
                        }
                    }
                }
                trigger_event = bm_interpersonal_events.0002
            }
            100 = {
                trigger = {
                    any_courtier = {
                        is_alive = yes
                        bm_is_blood_golem_trigger = yes
                    }
                }
                trigger_event = bm_interpersonal_events.0004
            }
            100 = {
                trigger = {
                    any_courtier = {
                        is_alive = yes
                        is_adult = yes
                        NOT = { is_blood_mage_any_tier_trigger = yes }
                        NOT = { bm_is_blood_golem_trigger = yes }
                    }
                }
                trigger_event = bm_interpersonal_events.0005
            }
            100 = {
                trigger = {
                    OR = {
                        has_character_modifier = bm_overdrained_modifier
                        has_character_modifier = lifeforce_modifier_negative_minor
                        has_character_modifier = lifeforce_modifier_negative_major
                        has_character_modifier = lifeforce_modifier_negative_superior
                    }
                    any_courtier = {
                        is_alive = yes
                        is_adult = yes
                        NOT = { is_close_family_of = root }
                        NOT = { bm_is_blood_golem_trigger = yes }
                    }
                }
                modifier = {
                    add = 50
                    has_character_modifier = bm_overdrained_modifier
                }
                trigger_event = bm_interpersonal_events.0006
            }
            100 = {
                trigger = {
                    is_blood_mage_tier_2_or_higher_trigger = yes
                    is_imprisoned = no
                    is_incapable = no
                }
                trigger_event = bm_interpersonal_events.0008
            }
            100 = {
                trigger = {
                    any_courtier = {
                        is_alive = yes
                        bm_is_blood_golem_trigger = yes
                    }
                }
                trigger_event = bm_interpersonal_events.0009
            }
            100 = {
                trigger = {
                    is_blood_mage_tier_2_or_higher_trigger = yes
                    any_courtier = {
                        is_alive = yes
                        age >= 12
                        has_trait = lifestyle_bm_acolyte
                        NOT = { this = root }
                    }
                }
                trigger_event = bm_interpersonal_events.0010
            }
        }
    }
}
