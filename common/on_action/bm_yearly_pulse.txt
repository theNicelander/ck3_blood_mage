yearly_playable_pulse = {
    on_actions = {
        bm_yearly_pulse_playable
    }
}

quarterly_playable_pulse = {
    on_actions = {
        bm_quarterly_blood_power_xp_pulse
        bm_quarterly_channel_skill_xp_pulse
    }
}

bm_quarterly_blood_power_xp_pulse = {
    trigger = {
        has_trait = lifestyle_blood_power
        bm_has_blood_power_xp_trigger = { VALUE = 10 }
    }
    effect = {
        trigger_event = blood_mage_yearly_events.005
    }
}

bm_quarterly_channel_skill_xp_pulse = {
    trigger = {
        OR = {
            has_character_modifier = diplomacy_lifeforce_modifier
            has_character_modifier = martial_lifeforce_modifier
            has_character_modifier = stewardship_lifeforce_modifier
            has_character_modifier = intrigue_lifeforce_modifier
            has_character_modifier = learning_lifeforce_modifier
            has_character_modifier = prowess_lifeforce_modifier
            has_character_modifier = exp_lifeforce_modifier
        }
    }
    effect = {
        trigger_event = blood_mage_yearly_events.006
    }
}

random_yearly_everyone_pulse = {
    on_actions = {
        bm_yearly_pulse_everyone
        bm_review_unmarked_blood_mage_prevalence
        bm_review_unmarked_blood_knight_prevalence
        bm_yearly_interpersonal_pulse
    }
}

bm_yearly_pulse_playable = {
    trigger = {
        OR = {
            has_trait = lifestyle_blood_mage
            has_trait = lifestyle_blood_knight
            bm_has_blood_rune_trigger = yes
        }
    }
    effect = {
        trigger_event = blood_mage_yearly_events.001
        trigger_event = blood_mage_yearly_events.002
        trigger_event = blood_mage_yearly_events.003
        trigger_event = blood_mage_yearly_events.004
    }
}

bm_yearly_pulse_everyone = {
    trigger = {
        is_playable_character = no
        OR = {
            has_trait = lifestyle_blood_mage
            has_trait = lifestyle_blood_knight
            bm_has_blood_rune_trigger = yes
        }
    }
    effect = {
        trigger_event = blood_mage_yearly_events.001
        trigger_event = blood_mage_yearly_events.002
        trigger_event = blood_mage_yearly_events.003
        trigger_event = blood_mage_yearly_events.004
    }
}


bm_review_unmarked_blood_mage_prevalence = {
    trigger = {
        is_ai = yes
        has_trait = lifestyle_blood_mage
        NOT = { has_character_flag = bm_blood_mage_prevalence_reviewed }
        OR = {
            has_game_rule = bm_prevalence_player_only
            has_game_rule = bm_prevalence_one_in_10000
            has_game_rule = bm_prevalence_one_in_1000
        }
    }
    effect = {
        bm_apply_blood_mage_prevalence_retention_effect = yes
    }
}

bm_review_unmarked_blood_knight_prevalence = {
    trigger = {
        is_ai = yes
        has_trait = lifestyle_blood_knight
        NOT = { has_character_flag = bm_blood_knight_prevalence_reviewed }
        OR = {
            has_game_rule = bm_blood_knight_prevalence_player_only
            has_game_rule = bm_blood_knight_prevalence_one_in_10000
            has_game_rule = bm_blood_knight_prevalence_one_in_1000
        }
    }
    effect = {
        bm_apply_blood_knight_prevalence_retention_effect = yes
    }
}

bm_yearly_interpersonal_pulse = {
    trigger = {
        is_ai = no
        OR = {
            has_trait = lifestyle_blood_mage
            has_trait = lifestyle_blood_knight
        }
        NOT = { has_character_flag = bm_interpersonal_event_cooldown }
    }
    effect = {
        random_list = {
            100 = {
                trigger = {
                    OR = {
                        any_relation = { type = soulmate is_alive = yes }
                        any_relation = { type = lover is_alive = yes }
                        any_spouse = { is_alive = yes }
                        any_consort = { is_alive = yes }
                    }
                }
                trigger_event = bm_interpersonal_events.0001
            }
            100 = {
                trigger = {
                    OR = {
                        any_child = {
                            is_alive = yes
                            age >= 6
                            age <= 15
                        }
                        any_relation = {
                            type = ward
                            is_alive = yes
                            age >= 6
                            age <= 15
                        }
                    }
                }
                trigger_event = bm_interpersonal_events.0002
            }
            100 = {
                trigger = {
                    any_courtier = {
                        is_alive = yes
                        house ?= house:bm_house_golem
                    }
                }
                trigger_event = bm_interpersonal_events.0004
            }
            100 = {
                trigger = {
                    any_courtier = {
                        is_alive = yes
                        is_adult = yes
                        NOT = { has_trait = lifestyle_blood_mage }
                        NOT = { has_trait = lifestyle_blood_knight }
                        NOT = { house ?= house:bm_house_golem }
                    }
                }
                trigger_event = bm_interpersonal_events.0005
            }
        }
    }
}
