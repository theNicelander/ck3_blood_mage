on_combat_end_winner = {
	on_actions = {
		bm_on_combat_end_winner
	}
}

bm_on_combat_end_winner = {
	effect = {
		every_side_knight = {
			limit = {
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = vanguard XP = 5 }
		}
		every_side_commander = {
			limit = {
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = vanguard XP = 6 }
		}
	}
}

on_combat_end_loser = {
	on_actions = {
		bm_on_combat_end_loser
	}
}

bm_on_combat_end_loser = {
	effect = {
		every_side_knight = {
			limit = {
				is_alive = yes
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = vanguard XP = 1 }
			add_crimson_warrior_xp = { TRACK = resilience XP = 1 }
		}
		every_side_commander = {
			limit = {
				is_alive = yes
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = vanguard XP = 1 }
			add_crimson_warrior_xp = { TRACK = resilience XP = 1 }
		}
	}
}

on_travel_activity_complete = {
	on_actions = {
		bm_on_travel_activity_complete_tournament
	}
}

bm_on_travel_activity_complete_tournament = {
	effect = {
		if = {
			limit = {
				has_trait = lifestyle_crimson_warrior
				involved_activity ?= { has_activity_type = activity_tournament }
			}
			add_crimson_warrior_xp = { TRACK = slaughter XP = 3 }
		}
	}
}
