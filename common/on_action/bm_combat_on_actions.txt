on_combat_end_winner = {
	on_actions = {
		bm_on_combat_end_winner
	}
}

bm_on_combat_end_winner = {
	effect = {
		every_side_knight = {
			limit = {
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = slaughter XP = 5 }
		}
		every_side_commander = {
			limit = {
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = vanguard XP = 6 }
		}
	}
}

on_combat_end_loser = {
	on_actions = {
		bm_on_combat_end_loser
	}
}

bm_on_combat_end_loser = {
	effect = {
		every_side_knight = {
			limit = {
				is_alive = yes
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = resilience XP = 2 }
		}
		every_side_commander = {
			limit = {
				is_alive = yes
				has_trait = lifestyle_crimson_warrior
			}
			add_crimson_warrior_xp = { TRACK = resilience XP = 2 }
		}
	}
}
