on_combat_end_winner = {
	on_actions = {
		bm_on_combat_end_winner
	}
}

bm_on_combat_end_winner = {
	effect = {
		save_scope_as = winning_side
		every_side_commander = {
			limit = {
				is_alive = yes
				OR = {
					has_trait = lifestyle_blood_knight
					bm_has_superior_stave_trigger = { STAVE = nabrokarstafur }
				}
			}
			if = {
				limit = { has_trait = lifestyle_blood_knight }
				bm_combat_winner_commander_reward_effect = yes
				bm_battle_victory_lifeforce_harvest_effect = yes
			}
			if = {
				limit = { bm_has_superior_stave_trigger = { STAVE = nabrokarstafur } }
				add_gold = bm_nabrokarstafur_battle_gold_value
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_nabrokarstafur_battle_gold_toast
					left_icon = this
					custom_tooltip = bm_nabrokarstafur_battle_gold_tt
				}
			}
		}
		every_side_knight = {
			limit = {
				is_alive = yes
				OR = {
					has_trait = lifestyle_blood_knight
					bm_has_superior_stave_trigger = { STAVE = nabrokarstafur }
				}
				NOT = { this = scope:winning_side.side_commander }
			}
			if = {
				limit = { has_trait = lifestyle_blood_knight }
				bm_combat_winner_knight_reward_effect = yes
				bm_battle_victory_lifeforce_harvest_effect = yes
			}
			if = {
				limit = { bm_has_superior_stave_trigger = { STAVE = nabrokarstafur } }
				add_gold = bm_nabrokarstafur_battle_gold_value
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_nabrokarstafur_battle_gold_toast
					left_icon = this
					custom_tooltip = bm_nabrokarstafur_battle_gold_tt
				}
			}
		}
		every_side_commander = {
			limit = {
				is_ai = no
				is_alive = yes
				OR = {
					has_trait = lifestyle_blood_mage
					has_trait = lifestyle_blood_knight
				}
				NOT = { has_character_flag = bm_interpersonal_event_cooldown }
				any_knight = {
					is_alive = yes
					NOT = { this = prev }
				}
			}
			random = {
				chance = 15
				trigger_event = bm_interpersonal_events.0003
			}
		}
		every_side_participant = {
			limit = {
				is_ai = no
				is_alive = yes
				NOT = { this = scope:winning_side.side_commander }
				OR = {
					has_trait = lifestyle_blood_mage
					has_trait = lifestyle_blood_knight
				}
				NOT = { has_character_flag = bm_interpersonal_event_cooldown }
				any_knight = {
					is_alive = yes
					NOT = { this = prev }
				}
			}
			random = {
				chance = 15
				trigger_event = bm_interpersonal_events.0003
			}
		}
	}
}

on_combat_end_loser = {
	on_actions = {
		bm_on_combat_end_loser
	}
}

bm_on_combat_end_loser = {
	effect = {
		save_scope_as = losing_side
		every_side_commander = {
			limit = {
				is_alive = yes
				has_trait = lifestyle_blood_knight
			}
			bm_combat_loser_commander_reward_effect = yes
		}
		every_side_knight = {
			limit = {
				is_alive = yes
				has_trait = lifestyle_blood_knight
				NOT = { this = scope:losing_side.side_commander }
			}
			bm_combat_loser_knight_reward_effect = yes
		}
	}
}

on_travel_activity_complete = {
	on_actions = {
		bm_on_travel_activity_complete_tournament
	}
}

bm_on_travel_activity_complete_tournament = {
	effect = {
		if = {
			limit = {
				has_trait = lifestyle_blood_knight
				involved_activity ?= { has_activity_type = activity_tournament }
			}
			add_blood_knight_xp = { TRACK = slaughter XP = 3 }
		}
	}
}
