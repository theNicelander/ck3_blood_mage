on_death = {
	on_actions = {
		bm_on_character_death_duel
	}
}

bm_on_character_death_duel = {
	effect = {
		if = {
			limit = {
				exists = scope:killer
				scope:killer = {
					OR = {
						has_trait = lifestyle_blood_knight
						has_character_modifier = bm_rune_aegishjalmur_superior
					}
					has_variable = engaged_in_single_combat
				}
				has_variable = engaged_in_single_combat
			}
			scope:killer = {
				if = {
					limit = { has_trait = lifestyle_blood_knight }
					add_blood_knight_xp = { TRACK = slaughter XP = 5 }
				}
				add_character_modifier = { modifier = lifeforce_modifier_major }
				if = {
					limit = {
						root = {
							OR = {
								prowess >= 35
								has_trait = lifestyle_blood_mage
								has_trait = lifestyle_blood_knight
							}
						}
					}
					random = {
						chance = 25
						add_character_modifier = { modifier = lifeforce_modifier_superior }
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_duel_kill_blood_knight_toast
					left_icon = scope:killer
					right_icon = root
					custom_tooltip = bm_duel_kill_blood_knight_tt
				}
			}
		}
	}
}
