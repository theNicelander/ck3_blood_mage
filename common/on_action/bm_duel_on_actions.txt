# Duel & Combat Clash On-Actions for Blood Mages

on_combat_start = {
	on_actions = {
		bm_on_combat_start_aegishjalmur_terror
	}
}

bm_on_combat_start_aegishjalmur_terror = {
	effect = {
		# Attacker has Aegishjalmur -> Defender terrified
		if = {
			limit = {
				combat_attacker = {
					OR = {
						side_commander ?= { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
						any_side_knight = { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
					}
				}
			}
			combat_defender = {
				side_commander ?= {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
				every_side_knight = {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
			}
		}
		# Defender has Aegishjalmur -> Attacker terrified
		if = {
			limit = {
				combat_defender = {
					OR = {
						side_commander ?= { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
						any_side_knight = { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
					}
				}
			}
			combat_attacker = {
				side_commander ?= {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
				every_side_knight = {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
			}
		}
	}
}

on_death = {
	on_actions = {
		bm_on_character_death_duel
	}
}

bm_on_character_death_duel = {
	effect = {
		if = {
			limit = {
				exists = scope:killer
				scope:killer = {
					is_blood_mage_any_tier_trigger = yes
					has_variable = engaged_in_single_combat
				}
				has_variable = engaged_in_single_combat
			}
			scope:killer = {
				bm_add_blood_mage_xp_effect = { TRACK = bloodlust XP = 5 }
				add_character_modifier = { modifier = lifeforce_modifier_major }
				if = {
					limit = {
						root = {
							OR = {
								prowess >= 20
								has_variable = bm_is_seek_power_duel_target
								is_blood_mage_any_tier_trigger = yes
							}
						}
					}
					random = {
						chance = 25
						add_character_modifier = { modifier = lifeforce_modifier_superior }
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_duel_kill_bloodlust_toast
					left_icon = scope:killer
					right_icon = root
					custom_tooltip = bm_duel_kill_bloodlust_tt
				}
			}
		}
	}
}
