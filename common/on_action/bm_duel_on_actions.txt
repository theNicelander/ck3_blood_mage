# Duel & Combat Clash On-Actions for Blood Mages

on_combat_start = {
	on_actions = {
		bm_on_combat_start_aegishjalmur_terror
	}
}

bm_on_combat_start_aegishjalmur_terror = {
	effect = {
		# Attacker has Aegishjalmur -> Defender terrified
		if = {
			limit = {
				combat_attacker = {
					OR = {
						side_commander ?= { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
						any_side_knight = { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
					}
				}
			}
			combat_defender = {
				side_commander ?= {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
				every_side_knight = {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
			}
		}
		# Defender has Aegishjalmur -> Attacker terrified
		if = {
			limit = {
				combat_defender = {
					OR = {
						side_commander ?= { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
						any_side_knight = { bm_has_superior_stave_trigger = { STAVE = aegishjalmur } }
					}
				}
			}
			combat_attacker = {
				side_commander ?= {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
				every_side_knight = {
					add_character_modifier = {
						modifier = bm_aegishjalmur_terror_modifier
						years = 1
					}
				}
			}
		}
	}
}

on_death = {
	on_actions = {
		bm_on_character_death_duel
	}
}

bm_on_character_death_duel = {
	effect = {
		if = {
			limit = {
				exists = scope:killer
				scope:killer = {
					is_blood_mage_any_tier_trigger = yes
					has_variable = engaged_in_single_combat
				}
				has_variable = engaged_in_single_combat
			}
			scope:killer = {
				send_interface_toast = {
					type = bm_event_toast_effect_good
					title = bm_duel_kill_fury_toast
					left_icon = scope:killer
					right_icon = root
					bm_duel_victory_lifeforce_effect = {
						OPPONENT = root
					}
					bm_add_blood_mage_xp_effect = { TRACK = fury XP = 5 }
				}
			}
		}
	}
}
