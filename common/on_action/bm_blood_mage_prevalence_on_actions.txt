on_birth_child = {
	on_actions = {
		bm_apply_blood_mage_prevalence_at_birth
		bm_on_birth_bloodline_xp
	}
}

bm_apply_blood_mage_prevalence_at_birth = {
	effect = {
		if = {
			limit = {
				is_ai = yes
				has_trait = lifestyle_blood_mage
			}
			bm_apply_blood_mage_prevalence_retention_effect = yes
		}
		else_if = {
			limit = {
				is_ai = yes
				has_game_rule = bm_prevalence_everyone
			}
			bm_become_blood_mage_effect = yes
		}
		else_if = {
			limit = {
				is_ai = yes
				has_game_rule = bm_prevalence_one_in_10
			}
			if = {
				limit = {
					OR = {
						scope:mother = { has_trait = lifestyle_blood_mage }
						scope:real_father ?= { has_trait = lifestyle_blood_mage }
					}
				}
				random = {
					chance = 60
					bm_become_blood_mage_effect = yes
				}
			}
			else = {
				random = {
					chance = 10
					bm_become_blood_mage_effect = yes
				}
			}
		}
		else_if = {
			limit = {
				is_ai = yes
				has_game_rule = bm_prevalence_one_in_100
			}
			if = {
				limit = {
					OR = {
						scope:mother = { has_trait = lifestyle_blood_mage }
						scope:real_father ?= { has_trait = lifestyle_blood_mage }
					}
				}
				random = {
					chance = 35
					bm_become_blood_mage_effect = yes
				}
			}
			else = {
				random = {
					chance = 1
					bm_become_blood_mage_effect = yes
				}
			}
		}
	}
}

bm_on_birth_bloodline_xp = {
	effect = {
		if = {
			limit = {
				scope:mother ?= { has_trait = lifestyle_blood_mage }
			}
			scope:mother = {
				add_xp_bm_dynamic = {
					TRACK = bloodline
					XP = 2
				}
				if = {
					limit = { root = { has_trait = lifestyle_blood_mage } }
					add_xp_bm_dynamic = {
						TRACK = bloodline
						XP = 2
					}
					send_interface_toast = {
						type = event_toast_effect_good
						title = bm_birth_blood_mage_child_toast
						left_icon = root
						custom_tooltip = bm_birth_blood_mage_child_tt
					}
				}
				else = {
					send_interface_toast = {
						type = event_toast_effect_good
						title = bm_birth_child_bloodline_toast
						left_icon = root
						custom_tooltip = bm_birth_child_bloodline_tt
					}
				}
			}
		}
		if = {
			limit = {
				scope:real_father ?= { has_trait = lifestyle_blood_mage }
			}
			scope:real_father = {
				add_xp_bm_dynamic = {
					TRACK = bloodline
					XP = 2
				}
				if = {
					limit = { root = { has_trait = lifestyle_blood_mage } }
					add_xp_bm_dynamic = {
						TRACK = bloodline
						XP = 2
					}
					send_interface_toast = {
						type = event_toast_effect_good
						title = bm_birth_blood_mage_child_toast
						left_icon = root
						custom_tooltip = bm_birth_blood_mage_child_tt
					}
				}
				else = {
					send_interface_toast = {
						type = event_toast_effect_good
						title = bm_birth_child_bloodline_toast
						left_icon = root
						custom_tooltip = bm_birth_child_bloodline_tt
					}
				}
			}
		}
	}
}
