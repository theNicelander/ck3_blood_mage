on_birth_child = {
	on_actions = {
		bm_apply_blood_mage_prevalence_at_birth
		bm_on_birth_bloodline_xp
	}
}

bm_apply_blood_mage_prevalence_at_birth = {
	effect = {
		if = {
			limit = {
				is_ai = yes
				is_blood_mage_any_tier_trigger = yes
			}
			bm_apply_blood_mage_prevalence_retention_effect = yes
		}
		else_if = {
			limit = {
				is_ai = no
				is_blood_mage_any_tier_trigger = yes
			}
			bm_ensure_blood_mage_story_effect = yes
		}
		else_if = {
			limit = {
				is_ai = yes
				has_game_rule = bm_prevalence_everyone
			}
			bm_become_blood_acolyte_effect = yes
		}
		else_if = {
			limit = {
				is_ai = yes
				has_game_rule = bm_prevalence_one_in_10
			}
			if = {
				limit = {
					OR = {
						scope:mother = { is_blood_mage_any_tier_trigger = yes }
						scope:real_father ?= { is_blood_mage_any_tier_trigger = yes }
					}
				}
				random = {
					chance = 60
					if = {
						limit = {
							OR = {
								scope:mother = { has_trait = lifestyle_bm_archmage }
								scope:real_father ?= { has_trait = lifestyle_bm_archmage }
							}
						}
						random = {
							chance = 15
							bm_become_blood_mage_tier_2_effect = yes
						}
						else = {
							bm_become_blood_acolyte_effect = yes
						}
					}
					else = {
						bm_become_blood_acolyte_effect = yes
					}
				}
			}
			else = {
				random = {
					chance = 10
					bm_become_blood_acolyte_effect = yes
				}
			}
		}
		else_if = {
			limit = {
				is_ai = yes
				has_game_rule = bm_prevalence_one_in_100
			}
			if = {
				limit = {
					OR = {
						scope:mother = { is_blood_mage_any_tier_trigger = yes }
						scope:real_father ?= { is_blood_mage_any_tier_trigger = yes }
					}
				}
				random = {
					chance = 35
					if = {
						limit = {
							OR = {
								scope:mother = { has_trait = lifestyle_bm_archmage }
								scope:real_father ?= { has_trait = lifestyle_bm_archmage }
							}
						}
						random = {
							chance = 15
							bm_become_blood_mage_tier_2_effect = yes
						}
						else = {
							bm_become_blood_acolyte_effect = yes
						}
					}
					else = {
						bm_become_blood_acolyte_effect = yes
					}
				}
			}
			else = {
				random = {
					chance = 1
					bm_become_blood_acolyte_effect = yes
				}
			}
		}
	}
}

bm_on_birth_bloodline_xp = {
	effect = {
		# Feingur Superior: Awakened Seed single-parent inheritance boost (25% -> 50%)
		if = {
			limit = {
				NOT = { is_blood_mage_any_tier_trigger = yes }
				OR = {
					AND = {
						scope:mother ?= {
							is_blood_mage_any_tier_trigger = yes
						}
						NOT = { scope:real_father ?= { is_blood_mage_any_tier_trigger = yes } }
					}
					AND = {
						scope:real_father ?= {
							is_blood_mage_any_tier_trigger = yes
						}
						NOT = { scope:mother ?= { is_blood_mage_any_tier_trigger = yes } }
					}
				}
				OR = {
					scope:mother ?= { bm_has_superior_stave_trigger = { STAVE = feingur } }
					scope:real_father ?= { bm_has_superior_stave_trigger = { STAVE = feingur } }
				}
			}
			random = {
				chance = 33
				bm_become_blood_acolyte_effect = yes
			}
		}

		# Mother XP award
		if = {
			limit = {
				scope:mother ?= {
					is_blood_mage_any_tier_trigger = yes
				}
			}
			scope:mother = {
				add_character_modifier = { modifier = lifeforce_modifier_major }
				if = {
					limit = {
						NOR = {
							any_child = {
								even_if_dead = yes
								this != root
							}
						}
					}
					add_shared_bm_bk_xp = { TRACK = ancient XP = 5 }
				}
				else = {
					add_shared_bm_bk_xp = { TRACK = ancient XP = 2 }
				}
				if = {
					limit = { bm_has_superior_stave_trigger = { STAVE = feingur } }
					add_xp_bm_dynamic = { TRACK = bloodline XP = 4 }
					if = {
						limit = { root = { is_blood_mage_any_tier_trigger = yes } }
						add_xp_bm_dynamic = { TRACK = bloodline XP = 4 }
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_blood_mage_child_toast
							left_icon = root
							custom_tooltip = bm_birth_blood_mage_child_tt
						}
					}
					else = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_child_bloodline_toast
							left_icon = root
							custom_tooltip = bm_birth_child_bloodline_tt
						}
					}
				}
				else = {
					add_xp_bm_dynamic = { TRACK = bloodline XP = 2 }
					if = {
						limit = { root = { is_blood_mage_any_tier_trigger = yes } }
						add_xp_bm_dynamic = { TRACK = bloodline XP = 2 }
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_blood_mage_child_toast
							left_icon = root
							custom_tooltip = bm_birth_blood_mage_child_tt
						}
					}
					else = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_child_bloodline_toast
							left_icon = root
							custom_tooltip = bm_birth_child_bloodline_tt
						}
					}
				}
			}
		}

		# Father XP award
		if = {
			limit = {
				scope:real_father ?= {
					is_blood_mage_any_tier_trigger = yes
				}
			}
			scope:real_father = {
				add_character_modifier = { modifier = lifeforce_modifier_major }
				if = {
					limit = {
						NOR = {
							any_child = {
								even_if_dead = yes
								this != root
							}
						}
					}
					add_shared_bm_bk_xp = { TRACK = ancient XP = 5 }
				}
				else = {
					add_shared_bm_bk_xp = { TRACK = ancient XP = 2 }
				}
				if = {
					limit = { bm_has_superior_stave_trigger = { STAVE = feingur } }
					add_xp_bm_dynamic = { TRACK = bloodline XP = 4 }
					if = {
						limit = { root = { is_blood_mage_any_tier_trigger = yes } }
						add_xp_bm_dynamic = { TRACK = bloodline XP = 4 }
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_blood_mage_child_toast
							left_icon = root
							custom_tooltip = bm_birth_blood_mage_child_tt
						}
					}
					else = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_child_bloodline_toast
							left_icon = root
							custom_tooltip = bm_birth_child_bloodline_tt
						}
					}
				}
				else = {
					add_xp_bm_dynamic = { TRACK = bloodline XP = 2 }
					if = {
						limit = { root = { is_blood_mage_any_tier_trigger = yes } }
						add_xp_bm_dynamic = { TRACK = bloodline XP = 2 }
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_blood_mage_child_toast
							left_icon = root
							custom_tooltip = bm_birth_blood_mage_child_tt
						}
					}
					else = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = bm_birth_child_bloodline_toast
							left_icon = root
							custom_tooltip = bm_birth_child_bloodline_tt
						}
					}
				}
			}
		}
	}
}
