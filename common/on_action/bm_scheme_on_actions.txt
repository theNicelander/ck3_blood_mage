# Scheme On-Actions for Blood Mages & Blood Knights

murder_success_pre_filter_on_action = {
	on_actions = {
		bm_on_murder_success_hulinhjalmur
	}
}

bm_on_murder_success_hulinhjalmur = {
	effect = {
		if = {
			limit = {
				scope:owner ?= {
					bm_has_superior_stave_trigger = { STAVE = hulinhjalmur }
				}
			}
			scope:owner = {
				add_character_modifier = { modifier = lifeforce_modifier_major }
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_hulinhjalmur_murder_lifeforce_toast
					left_icon = scope:owner
					right_icon = scope:target
					custom_tooltip = bm_hulinhjalmur_murder_lifeforce_tt
				}
			}
		}
	}
}
