level_up_trait_effect = {
    if = {
        limit = { NOT = { has_trait = $trait$ } }
        add_trait = $trait$_1
    }
    else = {
        change_trait_rank = {
            trait = $trait$
            rank = 1
            max = 5
        }
    }
}

bm_drain_leveled_trait_effect = {
    add_xp_bm_dynamic = {
        TRACK = hematurgy
        XP = 2
    }
    if = { limit = { has_trait = $TRAIT$_good_2 } add_piety_level = -1 }
    if = {
        limit = { has_trait = $TRAIT$_bad_1 }
        remove_trait = $TRAIT$_bad_1
    }
    else_if = {
        limit = { has_trait = $TRAIT$_bad }
        change_trait_rank = {
            trait = $TRAIT$_bad
            rank = -1
            max = 5
        }
    }
    else_if = {
        limit = { NOT = { has_trait = $TRAIT$_good } }
        add_trait = $TRAIT$_good_1
    }
    else = {
        change_trait_rank = {
            trait = $TRAIT$_good
            rank = 1
            max = 5
        }
    }
}

intellect_drain_trait_effects = {
    bm_drain_leveled_trait_effect = { TRAIT = intellect }
}

physique_drain_trait_effects = {
    bm_drain_leveled_trait_effect = { TRAIT = physique }
}

beauty_drain_trait_effects = {
    bm_drain_leveled_trait_effect = { TRAIT = beauty }
}

drain_trait_failed_effects = {
    temporary_negative_lifeforce_effect_major = yes
}
