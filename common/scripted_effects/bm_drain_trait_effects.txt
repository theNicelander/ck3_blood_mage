level_up_trait_effect = {
    if = {
        limit = { NOT = { has_trait = $trait$ } }
        add_trait = $trait$_1
    }
    else = {
        change_trait_rank = {
            trait = $trait$
            rank = 1
            max = 5
        }
    }
}

bm_drain_leveled_trait_effect = {
    add_xp_bm_dynamic = {
        TRACK = cunning
        XP = 3
    }
    if = { limit = { has_trait = $TRAIT$_good_2 } add_piety_level = -1 }
    if = {
        limit = { has_trait = $TRAIT$_bad_1 }
        remove_trait = $TRAIT$_bad_1
    }
    else_if = {
        limit = { has_trait = $TRAIT$_bad }
        change_trait_rank = {
            trait = $TRAIT$_bad
            rank = -1
            max = 5
        }
    }
    else_if = {
        limit = { NOT = { has_trait = $TRAIT$_good } }
        add_trait = $TRAIT$_good_1
    }
    else = {
        change_trait_rank = {
            trait = $TRAIT$_good
            rank = 1
            max = 5
        }
    }
}

bm_trait_drain_commit_cost_and_cooldown_effect = {
    remove_character_modifier = lifeforce_modifier_major
    add_character_flag = {
        flag = bm_trait_drain_cooldown
        days = 60
    }
    scope:recipient = {
        add_character_flag = {
            flag = bm_trait_drain_victim_cooldown
            days = 180
        }
    }
}

        drain_trait_failed_effects = yes
        left_icon = root
        right_icon = scope:recipient
    }
}

    add_piety = $PIETY_COST$
    bm_trait_drain_commit_cost_and_cooldown_effect = yes
    temporary_trait_drain_attempted = yes
    duel = {
        skill = learning
        target = scope:recipient
        50 = {
            desc = bm_trait_drain.001.success
            compare_modifier = {
                value = scope:duel_value
                multiplier = 5
                min = -40
            }
            modifier = { add = trait_drain_modifier }
            send_interface_toast = {
                type = bm_event_toast_effect_good
                title = bm_trait_drain.001.success
                left_icon = root
                right_icon = scope:recipient
                bm_drain_leveled_trait_effect = { TRAIT = $TRAIT$ }
            }
            sacrifice_prisoner_effect = {
                VICTIM = scope:recipient
                EXECUTIONER = root
            }
        }
        50 = {
            modifier = { add = recipient_trait_drain_modifier_per_trait }
            bm_trait_drain_duel_fail_effects = yes
        }
    }
}

                add = -70
                has_trait_rank = { trait = $TRAIT$ rank >= 3 }

            send_interface_toast = {
                type = bm_event_toast_effect_good

                if = {

                }

                    change_trait_rank = {
                        trait = $TRAIT$
                        rank = 1
                        max = 4
                    }

                add_xp_bm_dynamic = {
                    TRACK = cunning

                }
