bm_transfer_anointed_heir_legacy_effect = {
	if = {
		limit = {
			is_blood_mage_any_tier_trigger = yes
		}
		save_scope_as = dying_ruler
		if = {
			limit = {
				exists = var:bm_anointed_heir
				var:bm_anointed_heir = {
					is_alive = yes
					has_character_modifier = bm_anointed_heir_modifier
				}
			}
			var:bm_anointed_heir = {
				save_scope_as = anointed_heir
			}
		}
		else_if = {
			limit = {
				exists = primary_heir
				primary_heir = {
					is_alive = yes
					has_character_modifier = bm_anointed_heir_modifier
				}
			}
			primary_heir = {
				save_scope_as = anointed_heir
			}
		}
		else = {
			random_child = {
				limit = {
					is_alive = yes
					has_character_modifier = bm_anointed_heir_modifier
				}
				save_scope_as = anointed_heir
			}
		}
		if = {
			limit = { exists = scope:anointed_heir }
			scope:anointed_heir = {
				remove_character_modifier = bm_anointed_heir_modifier
			}
			# 1. Awaken class trait if heir lacks it
			if = {
				limit = {
					scope:anointed_heir = {
						NOT = { is_blood_mage_any_tier_trigger = yes }
					}
				}
				scope:anointed_heir = {
					if = {
						limit = {
							scope:dying_ruler = { has_trait = lifestyle_bm_archmage }
						}
						bm_become_blood_mage_tier_2_effect = yes
					}
					else = {
						bm_become_blood_acolyte_effect = yes
					}
				}
			}
			# 2. Transfer 1 stack of each Lifeforce tier held by dying ruler
			if = {
				limit = { has_character_modifier = lifeforce_modifier_minor }
				scope:anointed_heir = {
					add_character_modifier = { modifier = lifeforce_modifier_minor }
				}
			}
			if = {
				limit = { has_character_modifier = lifeforce_modifier_major }
				scope:anointed_heir = {
					add_character_modifier = { modifier = lifeforce_modifier_major }
				}
			}
			if = {
				limit = { has_character_modifier = lifeforce_modifier_superior }
				scope:anointed_heir = {
					add_character_modifier = { modifier = lifeforce_modifier_superior }
				}
			}
			# 3. Transfer 25% of trait track XP across active tracks
			scope:anointed_heir = {
				if = {
					limit = { is_blood_mage_any_tier_trigger = yes }
					if = {
						limit = { bm_anointed_transfer_ancient_xp_value > 0 }
						bm_add_blood_mage_xp_effect = {
							TRACK = ancient
							XP = bm_anointed_transfer_ancient_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_enlightenment_xp_value > 0 }
						bm_add_blood_mage_xp_effect = {
							TRACK = enlightenment
							XP = bm_anointed_transfer_enlightenment_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_benediction_xp_value > 0 }
						bm_add_blood_mage_xp_effect = {
							TRACK = benediction
							XP = bm_anointed_transfer_benediction_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_bloodline_xp_value > 0 }
						bm_add_blood_mage_xp_effect = {
							TRACK = bloodline
							XP = bm_anointed_transfer_bloodline_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_hematurgy_xp_value > 0 }
						bm_add_blood_mage_xp_effect = {
							TRACK = hematurgy
							XP = bm_anointed_transfer_hematurgy_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_bloodlust_xp_value > 0 }
						bm_add_blood_mage_xp_effect = {
							TRACK = bloodlust
							XP = bm_anointed_transfer_bloodlust_xp_value
						}
					}
				}
				if = {
					limit = { has_trait = lifestyle_blood_power }
					if = {
						limit = { bm_anointed_transfer_dynasty_xp_value > 0 }
						add_trait_xp = {
							trait = lifestyle_blood_power
							track = dynasty
							value = bm_anointed_transfer_dynasty_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_warband_xp_value > 0 }
						add_trait_xp = {
							trait = lifestyle_blood_power
							track = warband
							value = bm_anointed_transfer_warband_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_mastery_xp_value > 0 }
						add_trait_xp = {
							trait = lifestyle_blood_power
							track = mastery
							value = bm_anointed_transfer_mastery_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_presence_xp_value > 0 }
						add_trait_xp = {
							trait = lifestyle_blood_power
							track = presence
							value = bm_anointed_transfer_presence_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_prosperity_xp_value > 0 }
						add_trait_xp = {
							trait = lifestyle_blood_power
							track = prosperity
							value = bm_anointed_transfer_prosperity_xp_value
						}
					}
					if = {
						limit = { bm_anointed_transfer_shadows_xp_value > 0 }
						add_trait_xp = {
							trait = lifestyle_blood_power
							track = shadows
							value = bm_anointed_transfer_shadows_xp_value
						}
					}
				}
			}
		}
	}
}
