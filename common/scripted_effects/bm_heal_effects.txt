# Healing Scripted Effects

bm_heal_minor_cost_and_xp_effect = {
	add_piety = cure_illness_cost_minor_neg
	if = {
		limit = { has_trait = lifestyle_blood_mage }
		benediction_magic_cast_effect_minor = yes
	}
	else = {
		remove_character_modifier = lifeforce_modifier_minor
		if = {
			limit = { root = scope:recipient }
			add_blood_knight_xp = { TRACK = enlightenment XP = 2 }
		}
		else = {
			add_blood_knight_xp = { TRACK = vanguard XP = 1 }
			add_blood_knight_xp = { TRACK = benediction XP = 1 }
		}
	}
}

bm_heal_minor_opinion_effect = {
	if = {
		limit = { NOT = { root = scope:recipient } }
		scope:recipient = {
			add_opinion = {
				target = root
				modifier = magic_opinion_positive
				opinion = 20
			}
		}
	}
}

bm_heal_major_cost_and_xp_effect = {

}

bm_heal_major_opinion_effect = {

			}
		}
	}
}
