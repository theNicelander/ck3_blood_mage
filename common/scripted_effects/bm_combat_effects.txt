bm_combat_winner_commander_reward_effect = {
	if = {
		limit = {
			scope:winning_side.enemy_side.side_max_army_size >= 7500
		}
		if = {
			limit = { bm_has_track_xp_any_tier_trigger = { TRACK = fury VALUE = 50 } }
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_commander_toast
				left_icon = this
				bm_add_blood_mage_xp_effect = { TRACK = fury XP = 2 }
			}
		}
		else = {
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_commander_toast
				left_icon = this
				bm_add_blood_mage_xp_effect = { TRACK = fury XP = 3 }
			}
		}
	}
	else_if = {
		limit = {
			scope:winning_side.enemy_side.side_max_army_size >= 2500
		}
		if = {
			limit = { bm_has_track_xp_any_tier_trigger = { TRACK = fury VALUE = 50 } }
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_commander_toast
				left_icon = this
				bm_add_blood_mage_xp_effect = { TRACK = fury XP = 1 }
			}
		}
		else = {
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_commander_toast
				left_icon = this
				bm_add_blood_mage_xp_effect = { TRACK = fury XP = 2 }
			}
		}
	}
	else_if = {
		limit = {
			scope:winning_side.enemy_side.side_max_army_size >= 500
			NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = fury VALUE = 50 } }
		}
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_battle_victory_commander_toast
			left_icon = this
			bm_add_blood_mage_xp_effect = { TRACK = fury XP = 1 }
		}
	}
}

bm_battle_victory_lifeforce_harvest_effect = {
	# 8,000+ enemy troops (or 4,000+ killed) -> 100% Minor, 100% Major
	if = {
		limit = {
			OR = {
				scope:winning_side.enemy_side.side_max_army_size >= 8000
				scope:winning_side = { num_enemies_killed >= 4000 }
			}
		}
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_battle_victory_both_lifeforce_toast
			left_icon = this
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_major }
		}
	}
	# 5,000 - 7,999 enemy troops (or 2,500+ killed) -> 100% Minor, 75% Major
	else_if = {
		limit = {
			OR = {
				scope:winning_side.enemy_side.side_max_army_size >= 5000
				scope:winning_side = { num_enemies_killed >= 2500 }
			}
		}
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_battle_victory_lifeforce_toast
			left_icon = this
			add_character_modifier = { modifier = lifeforce_modifier_minor }
		}
		random = {
			chance = 75
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_major_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_major }
			}
		}
	}
	# 2,500 - 4,999 enemy troops (or 1,250+ killed) -> 100% Minor, 50% Major
	else_if = {
		limit = {
			OR = {
				scope:winning_side.enemy_side.side_max_army_size >= 2500
				scope:winning_side = { num_enemies_killed >= 1250 }
			}
		}
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_battle_victory_lifeforce_toast
			left_icon = this
			add_character_modifier = { modifier = lifeforce_modifier_minor }
		}
		random = {
			chance = 50
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_major_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_major }
			}
		}
	}
	# 1,000 - 2,499 enemy troops (or 500+ killed) -> 75% Minor, 20% Major
	else_if = {
		limit = {
			OR = {
				scope:winning_side.enemy_side.side_max_army_size >= 1000
				scope:winning_side = { num_enemies_killed >= 500 }
			}
		}
		random = {
			chance = 75
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_minor }
			}
		}
		random = {
			chance = 20
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_major_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_major }
			}
		}
	}
	# 500 - 999 enemy troops (or 250+ killed) -> 50% Minor
	else_if = {
		limit = {
			OR = {
				scope:winning_side.enemy_side.side_max_army_size >= 500
				scope:winning_side = { num_enemies_killed >= 250 }
			}
		}
		random = {
			chance = 50
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_minor }
			}
		}
	}
}
