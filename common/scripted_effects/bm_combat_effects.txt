bm_combat_winner_commander_reward_effect = {
	if = {
		limit = {
			scope:winning_side = {
				OR = {
					combat = { num_total_troops >= 8000 }
					num_enemies_killed >= 4000
				}
			}
		}
		add_blood_knight_xp = { TRACK = vanguard XP = 5 }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_battle_victory_commander_toast
			left_icon = this
			custom_tooltip = bm_battle_victory_commander_t5_tt
		}
	}
	else_if = {
		limit = {
			scope:winning_side = {
				OR = {
					combat = { num_total_troops >= 4000 }
					num_enemies_killed >= 2000
				}
			}
		}
		add_blood_knight_xp = { TRACK = vanguard XP = 5 }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_battle_victory_commander_toast
			left_icon = this
			custom_tooltip = bm_battle_victory_commander_t4_tt
		}
	}
	else_if = {
		limit = {
			scope:winning_side = {
				OR = {
					combat = { num_total_troops >= 1000 }
					num_enemies_killed >= 500
				}
			}
		}
		add_blood_knight_xp = { TRACK = vanguard XP = 4 }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_battle_victory_commander_toast
			left_icon = this
			custom_tooltip = bm_battle_victory_commander_t3_tt
		}
	}
	else_if = {
		limit = {
			scope:winning_side = {
				OR = {
					combat = { num_total_troops >= 500 }
					num_enemies_killed >= 200
				}
			}
		}
		add_blood_knight_xp = { TRACK = vanguard XP = 3 }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_battle_victory_commander_toast
			left_icon = this
			custom_tooltip = bm_battle_victory_commander_t2_tt
		}
	}
	else = {
		add_blood_knight_xp = { TRACK = vanguard XP = 2 }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_battle_victory_commander_toast
			left_icon = this
			custom_tooltip = bm_battle_victory_commander_t1_tt
		}
	}
}

bm_battle_victory_lifeforce_harvest_effect = {
	# 10,000+ troops (or 5,000+ killed) -> 100% Minor, 100% Major
	if = {
		limit = {
			scope:winning_side = {
				OR = {
					combat = { num_total_troops >= 10000 }
					num_enemies_killed >= 5000
				}
			}
		}
		add_character_modifier = { modifier = lifeforce_modifier_minor }
		add_character_modifier = { modifier = lifeforce_modifier_major }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_battle_victory_both_lifeforce_toast
			left_icon = this
			custom_tooltip = bm_battle_victory_both_lifeforce_tt
		}
	}
	# 8,000 – 9,999 troops (or 4,000+ killed) -> 100% Minor, 90% Major
	else_if = {
		limit = {
			scope:winning_side = {
				OR = {

					num_enemies_killed >= 4000
				}
			}
		}
		add_character_modifier = { modifier = lifeforce_modifier_minor }

			type = event_toast_effect_good

			left_icon = this
			custom_tooltip = bm_battle_victory_lifeforce_tt
		}
		random = {
			chance = 90
			add_character_modifier = { modifier = lifeforce_modifier_major }

				type = event_toast_effect_good
