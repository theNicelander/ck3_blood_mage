bm_combat_winner_commander_reward_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = bm_battle_fury_xp_cooldown }
			bm_battle_total_combatants_value >= 500
		}
		add_character_flag = { flag = bm_battle_fury_xp_cooldown days = 60 }
		if = {
			limit = { bm_battle_total_combatants_value > 10000 }
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_commander_toast
				left_icon = this
				bm_add_blood_mage_xp_effect = { TRACK = fury XP = 3 }
			}
		}
		else = {
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_commander_toast
				left_icon = this
				bm_add_blood_mage_xp_effect = { TRACK = fury XP = 2 }
			}
		}
	}
}

bm_battle_victory_lifeforce_harvest_effect = {
	# Minor Lifeforce: 10% per 1k total combatants (500+ floor), 90d cooldown on harvest
	if = {
		limit = {
			NOT = { has_character_flag = bm_battle_lifeforce_minor_cooldown }
			bm_battle_total_combatants_value >= 500
		}
		random = {
			chance = bm_battle_minor_lifeforce_chance_value
			add_character_flag = { flag = bm_battle_lifeforce_minor_cooldown days = 90 }
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_minor }
			}
		}
	}
	# Major Lifeforce: 1% per 1k total combatants (5,000+ total), 90d cooldown on harvest
	if = {
		limit = {
			NOT = { has_character_flag = bm_battle_lifeforce_major_cooldown }
			bm_battle_total_combatants_value >= 5000
		}
		random = {
			chance = bm_battle_major_lifeforce_chance_value
			add_character_flag = { flag = bm_battle_lifeforce_major_cooldown days = 90 }
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_major_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_major }
			}
		}
	}
	# Superior Lifeforce: 1% at 100k scaling to 100% at 1M total combatants, 90d cooldown on harvest
	if = {
		limit = {
			NOT = { has_character_flag = bm_battle_lifeforce_superior_cooldown }
			bm_battle_total_combatants_value >= 100000
		}
		random = {
			chance = bm_battle_superior_lifeforce_chance_value
			add_character_flag = { flag = bm_battle_lifeforce_superior_cooldown days = 90 }
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_battle_victory_superior_lifeforce_toast
				left_icon = this
				add_character_modifier = { modifier = lifeforce_modifier_superior }
			}
		}
	}
}
