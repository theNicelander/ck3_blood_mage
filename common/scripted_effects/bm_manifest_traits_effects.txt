# Scripted effects for Manifesting Congenital Traits

bm_manifest_grant_scaled_xp_effect = {
	if = {
		limit = {
			has_trait_rank = { trait = $BRANCH$_good rank >= 5 }
		}
		if = {
			limit = { has_trait = lifestyle_blood_knight }
			add_blood_knight_xp = {
				TRACK = $KNIGHT_TRACK$
				XP = 8
			}
		}
		else = {
			add_xp_bm_dynamic = {
				TRACK = bloodline
				XP = 8
			}
		}
	}
	else_if = {
		limit = {
			has_trait_rank = { trait = $BRANCH$_good rank = 4 }
		}
		if = {
			limit = { has_trait = lifestyle_blood_knight }
			add_blood_knight_xp = {
				TRACK = $KNIGHT_TRACK$
				XP = 7
			}
		}
		else = {
			add_xp_bm_dynamic = {
				TRACK = bloodline
				XP = 7
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = $BRANCH$_good_3
				has_trait_rank = { trait = $BRANCH$_good rank = 3 }
			}
		}
		if = {
			limit = { has_trait = lifestyle_blood_knight }
			add_blood_knight_xp = {
				TRACK = $KNIGHT_TRACK$
				XP = 6
			}
		}
		else = {
			add_xp_bm_dynamic = {
				TRACK = bloodline
				XP = 6
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = $BRANCH$_good_2
				has_trait_rank = { trait = $BRANCH$_good rank = 2 }
			}
		}
		if = {
			limit = { has_trait = lifestyle_blood_knight }
			add_blood_knight_xp = {
				TRACK = $KNIGHT_TRACK$
				XP = 5
			}
		}
		else = {
			add_xp_bm_dynamic = {
				TRACK = bloodline
				XP = 5
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = $BRANCH$_good_1
				has_trait_rank = { trait = $BRANCH$_good rank = 1 }
			}
		}
		if = {
			limit = { has_trait = lifestyle_blood_knight }
			add_blood_knight_xp = {
				TRACK = $KNIGHT_TRACK$
				XP = 4
			}
		}
		else = {
			add_xp_bm_dynamic = {
				TRACK = bloodline
				XP = 4
			}
		}
	}
	else = {
		if = {
			limit = { has_trait = lifestyle_blood_knight }
			add_blood_knight_xp = {
				TRACK = $KNIGHT_TRACK$
				XP = 3
			}
		}
		else = {
			add_xp_bm_dynamic = {
				TRACK = bloodline
				XP = 3
			}
		}
	}
}

bm_manifest_upgrade_trait_step_effect = {
	if = {
		limit = {
			OR = {
				has_trait = $BRANCH$_good_2
				has_trait_rank = { trait = $BRANCH$_good rank = 3 }
				has_trait_rank = { trait = $BRANCH$_good rank = 4 }
			}
		}
		add_piety_level = -1
	}
	if = {
		limit = { has_trait = $BRANCH$_bad_1 }
		remove_trait = $BRANCH$_bad_1
	}
	else_if = {
		limit = { has_trait = $BRANCH$_bad }
		change_trait_rank = {
			trait = $BRANCH$_bad
			rank = -1
			max = 5
		}
	}
	else_if = {
		limit = { NOT = { has_trait = $BRANCH$_good } }
		add_trait = $BRANCH$_good_1
	}
	else = {
		change_trait_rank = {
			trait = $BRANCH$_good
			rank = 1
			max = 5
		}
	}
}

bm_manifest_intellect_duel_effect = {
	blood_magic_used_major = yes
	duel = {
		skill = learning
		value = 15

		# Success
		10 = {
			desc = bm_manifest_traits.duel_success_tt
			compare_modifier = {
				value = scope:duel_value
				multiplier = 2
			}
			modifier = { add = bm_manifest_traits_learning_traits_bonus }
			modifier = { add = bm_manifest_traits_xp_bonus }
			modifier = { add = bm_manifest_traits_intellect_tier_success_bonus }

			send_interface_toast = {
				title = bm_manifest_traits.intellect_success_title
				left_icon = root
				bm_manifest_upgrade_trait_step_effect = { BRANCH = intellect }
				bm_manifest_grant_scaled_xp_effect = {
					BRANCH = intellect
					KNIGHT_TRACK = enlightenment
				}
			}
		}

		# Standard Failure
		50 = {
			desc = bm_manifest_traits.duel_failure_tt
			modifier = { add = bm_manifest_traits_existing_traits_penalty }
			modifier = { add = bm_manifest_traits_intellect_tier_failure_addition }

			send_interface_toast = {
				title = bm_manifest_traits.failure_title
				left_icon = root
				add_character_modifier = { modifier = lifeforce_modifier_minor }
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
			}
		}

		# Critical Failure
		10 = {
			desc = bm_manifest_traits.duel_crit_failure_tt
			modifier = { add = bm_manifest_traits_existing_traits_penalty }

			send_interface_toast = {
				title = bm_manifest_traits.crit_failure_title
				left_icon = root
				increase_wounds_effect = { REASON = magic }
				add_stress = 50
				temporary_negative_lifeforce_effect_minor = yes
			}
		}
	}
}
