# Shared Egill Skallagrimsson build (geyser duel + artifact tempering duel)
# Saves scope:bm_egill. Event scopes do not persist across decisions, so a fresh Egill is made per chain and removed by bm_vanish_egill_effect.

bm_create_egill_effect = {
	if = {
		limit = {
			OR = {
				NOT = { exists = scope:bm_egill }
				scope:bm_egill = { is_alive = no }
			}
		}
		if = {
			limit = { exists = title:b_olafsvik }
			title:b_olafsvik = { save_scope_as = bm_egill_loc }
		}
		else = {
			root.location.county = { save_scope_as = bm_egill_loc }
		}
		create_character = {
			age = 100
			name = "Egill"
			gender_female_chance = 0
			random_traits = no
			location = scope:bm_egill_loc.title_province
			faith = scope:bm_egill_loc.title_province.faith
			rite = scope:bm_egill_loc.title_province.rite
			culture = scope:bm_egill_loc.title_province.culture

			# Education & lifestyles
			trait = education_martial_4
			trait = lifestyle_bm_archmage
			trait = lifestyle_poet
			trait = lifestyle_blademaster
			trait = lifestyle_hunter

			# Personality & physical
			trait = berserker
			trait = wrathful
			trait = greedy
			trait = brave
			trait = physique_good_3
			trait = beauty_bad_2

			diplomacy = 4
			intrigue = 18
			learning = 20
			martial = 18
			prowess = 30

			save_scope_as = bm_egill
		}
		scope:bm_egill = {
			if = {
				limit = { exists = house:house_skallagrimsson }
				set_house = house:house_skallagrimsson
			}
			add_trait_xp = { trait = lifestyle_bm_archmage track = ancient value = 50 }
			add_trait_xp = { trait = lifestyle_bm_archmage track = wisdom value = 50 }
			add_trait_xp = { trait = lifestyle_bm_archmage track = radiance value = 50 }
			add_trait_xp = { trait = lifestyle_bm_archmage track = fortune value = 50 }
			add_trait_xp = { trait = lifestyle_bm_archmage track = cunning value = 50 }
			add_trait_xp = { trait = lifestyle_bm_archmage track = fury value = 100 }
			add_trait_xp = { trait = lifestyle_blademaster value = 100 }
			add_trait_xp = { trait = lifestyle_hunter track = hunter value = 100 }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
		}
	}
}

bm_vanish_egill_effect = {
	if = {
		limit = {
			exists = scope:bm_egill
			scope:bm_egill = { is_alive = yes }
		}
		scope:bm_egill = { death = { death_reason = death_vanished } }
	}
}
