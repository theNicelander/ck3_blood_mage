bm_grant_conclave_study_xp_effect = {
	if = {
		limit = { root = { has_character_modifier = ancient_attuned } }
		bm_add_blood_mage_xp_effect = { TRACK = ancient XP = 4 }
	}
	else_if = {
		limit = { root = { has_character_modifier = fortune_attuned } }
		bm_add_blood_mage_xp_effect = { TRACK = fortune XP = 4 }
	}
	else_if = {
		limit = { root = { has_character_modifier = radiance_attuned } }
		bm_add_blood_mage_xp_effect = { TRACK = radiance XP = 4 }
	}
	else_if = {
		limit = { root = { has_character_modifier = cunning_attuned } }
		bm_add_blood_mage_xp_effect = { TRACK = cunning XP = 4 }
	}
	else_if = {
		limit = { root = { has_character_modifier = fury_attuned } }
		bm_add_blood_mage_xp_effect = { TRACK = fury XP = 4 }
	}
	else = {
		bm_add_blood_mage_xp_effect = { TRACK = wisdom XP = 4 }
	}
}

bm_apply_blood_conclave_outcome_effect = {
	if = {
		limit = { has_character_flag = bm_conclave_focus_study }
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_blood_conclave_study_toast
			left_icon = root
			right_icon = scope:conclave_guest
			bm_grant_conclave_study_xp_effect = yes
			every_in_list = {
				list = conclave_attendees
				bm_grant_conclave_study_xp_effect = yes
			}
		}
	}
	else_if = {
		limit = { has_character_flag = bm_conclave_focus_resonance }
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_blood_conclave_resonance_toast
			left_icon = root
			right_icon = scope:conclave_guest
			if = {
				limit = {
					any_in_list = {
						list = conclave_attendees
						count >= 3
					}
				}
				add_character_modifier = { modifier = lifeforce_modifier_major }
			}
			else = {
				add_character_modifier = { modifier = lifeforce_modifier_minor }
				add_character_modifier = { modifier = lifeforce_modifier_minor }
			}
			every_in_list = {
				list = conclave_attendees
				temporary_negative_lifeforce_effect_minor = yes
				add_opinion = {
					target = root
					modifier = magic_opinion_positive
					opinion = 10
				}
			}
		}
	}
	else = {
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_blood_conclave_communion_toast
			left_icon = root
			right_icon = scope:conclave_guest
			add_stress = -30
			bm_add_blood_mage_xp_effect = { TRACK = radiance XP = 2 }
			every_in_list = {
				list = conclave_attendees
				add_stress = -30
				add_opinion = {
					target = root
					modifier = magic_opinion_positive
					opinion = 20
				}
			}
		}
	}
}
