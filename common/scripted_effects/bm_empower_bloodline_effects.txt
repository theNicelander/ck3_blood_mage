# Scripted effects for Empower Dynasty Bloodline covenants and surges

# Upgrade House Covenant to next tier
# Parameter: COVENANT (radiance, fury, fortune, cunning, wisdom, ancient)
bm_upgrade_house_covenant_effect = {
	# Tier I -> Tier II
	if = {
		limit = {
			house = { has_house_modifier = bm_covenant_$COVENANT$_minor_modifier }
		}
		blood_magic_used_major = yes
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_empower_bloodline_tier_2_toast
			left_icon = root
			house = {
				remove_house_modifier = bm_covenant_$COVENANT$_minor_modifier
				add_house_modifier = { modifier = bm_covenant_$COVENANT$_major_modifier }
			}
			add_xp_bm_dynamic = {
				TRACK = $COVENANT$
				XP = 5
			}
		}
	}
	# Tier II -> Tier III
	else_if = {
		limit = {
			house = { has_house_modifier = bm_covenant_$COVENANT$_major_modifier }
		}
		blood_magic_used_superior = yes
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_empower_bloodline_tier_3_toast
			left_icon = root
			house = {
				remove_house_modifier = bm_covenant_$COVENANT$_major_modifier
				add_house_modifier = { modifier = bm_covenant_$COVENANT$_superior_modifier }
			}
			add_xp_bm_dynamic = {
				TRACK = $COVENANT$
				XP = 8
			}
		}
	}
	# Base -> Tier I
	else = {
		blood_magic_used_minor = yes
		send_interface_toast = {
			type = bm_event_toast_effect_good
			title = bm_empower_bloodline_tier_1_toast
			left_icon = root
			house = {
				add_house_modifier = { modifier = bm_covenant_$COVENANT$_minor_modifier }
			}
			add_xp_bm_dynamic = {
				TRACK = $COVENANT$
				XP = 3
			}
		}
	}
}

# Grant one-off Covenant of Initiation
bm_grant_acolyte_covenant_effect = {
	blood_magic_used_major = yes
	send_interface_toast = {
		type = bm_event_toast_effect_good
		title = bm_empower_bloodline_acolyte_toast
		left_icon = root
		house = {
			add_house_modifier = { modifier = bm_covenant_acolyte_modifier }
		}
		add_xp_bm_dynamic = {
			TRACK = fortune
			XP = 5
		}
	}
}
