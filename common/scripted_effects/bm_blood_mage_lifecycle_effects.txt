bm_ensure_blood_mage_story_effect = {
	if = {
		limit = {
			is_blood_mage_any_tier_trigger = yes
			NOT = { any_owned_story = { story_type = bm_blood_mage_story } }
		}
		create_story = bm_blood_mage_story
	}
	if = {
		limit = {
			is_blood_mage_any_tier_trigger = yes
			NOT = { has_trait = lifestyle_blood_power }
		}
		add_trait = lifestyle_blood_power
	}
	if = {
		limit = {
			OR = {
				is_blood_mage_tier_2_or_higher_trigger = yes
				bm_has_track_xp_any_tier_trigger = { TRACK = bloodlust VALUE = 20 }
			}
			NOT = { has_character_modifier = ignores_gender_army_rules }
		}
		add_character_modifier = ignores_gender_army_rules
	}
}

bm_become_blood_acolyte_effect = {
	if = {
		limit = {
			NOT = { is_blood_mage_any_tier_trigger = yes }
		}
		add_trait = lifestyle_bm_acolyte
		if = {
			limit = { NOT = { has_trait = lifestyle_blood_power } }
			add_trait = lifestyle_blood_power
		}
	}
	bm_ensure_blood_mage_story_effect = yes
	if = {
		limit = { is_blood_mage_any_tier_trigger = yes }
		add_character_flag = bm_blood_mage_prevalence_reviewed
	}
}

bm_become_blood_mage_tier_2_effect = {
	if = {
		limit = { has_trait = lifestyle_bm_acolyte }
		remove_trait = lifestyle_bm_acolyte
	}
	if = {
		limit = { NOT = { has_trait = lifestyle_blood_mage } }
		add_trait = lifestyle_blood_mage
	}
	if = {
		limit = { NOT = { has_trait = lifestyle_blood_power } }
		add_trait = lifestyle_blood_power
	}
	bm_ensure_blood_mage_story_effect = yes
	add_character_flag = bm_blood_mage_prevalence_reviewed
}

bm_elevate_tier_effect = {
	if = {
		limit = { has_trait = lifestyle_bm_acolyte }
		# Calculate unspent XP stacks across 5 non-ancient tracks
		set_variable = {
			name = bm_elevation_overflow_stacks
			value = 0
		}
		set_variable = {
			name = bm_overflow_bloodlust
			value = "has_trait_xp(lifestyle_bm_acolyte|bloodlust)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks
			add = var:bm_overflow_bloodlust
		}
		set_variable = {
			name = bm_overflow_benediction
			value = "has_trait_xp(lifestyle_bm_acolyte|benediction)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks

		}
		set_variable = {
			name = bm_overflow_bloodline
			value = "has_trait_xp(lifestyle_bm_acolyte|bloodline)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks
			add = var:bm_overflow_bloodline
		}

			name = bm_overflow_hematurgy

			add_trait = lifestyle_blood_mage

					value = 0

			}

				bm_ensure_blood_mage_story_effect = yes
				bm_recount_lifeforce_stacks_effect = yes

				remove_variable = bm_overflow_bloodlust
				remove_variable = bm_overflow_benediction
				remove_variable = bm_overflow_bloodline
				remove_variable = bm_overflow_hematurgy
				remove_variable = bm_overflow_enlightenment
			}
		}
	}
	else_if = {
		limit = { has_trait = lifestyle_blood_mage }
		set_variable = {
			name = bm_elevation_overflow_stacks
			value = 0
		}
		set_variable = {
			name = bm_overflow_bloodlust
			value = "has_trait_xp(lifestyle_blood_mage|bloodlust)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks
			add = var:bm_overflow_bloodlust
		}
		set_variable = {
			name = bm_overflow_benediction
			value = "has_trait_xp(lifestyle_blood_mage|benediction)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks
			add = var:bm_overflow_benediction
		}
		set_variable = {
			name = bm_overflow_bloodline
			value = "has_trait_xp(lifestyle_blood_mage|bloodline)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks
			add = var:bm_overflow_bloodline
		}
		set_variable = {
			name = bm_overflow_hematurgy
			value = "has_trait_xp(lifestyle_blood_mage|hematurgy)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks
			add = var:bm_overflow_hematurgy
		}
		set_variable = {
			name = bm_overflow_enlightenment
			value = "has_trait_xp(lifestyle_blood_mage|enlightenment)"
			divide = 20
			floor = yes
		}
		change_variable = {
			name = bm_elevation_overflow_stacks
			add = var:bm_overflow_enlightenment
		}

			type = event_toast_effect_good
			title = bm_elevate_blood_archmage_tier_toast
			left_icon = root

				bm_recount_lifeforce_stacks_effect = yes
				remove_variable = bm_elevation_overflow_stacks
				remove_variable = bm_overflow_bloodlust
				remove_variable = bm_overflow_benediction

				remove_variable = bm_overflow_hematurgy
				remove_variable = bm_overflow_enlightenment
			}
		}

}

bm_apply_blood_mage_prevalence_retention_effect = {
	if = {
		limit = { has_game_rule = bm_prevalence_player_only }
		remove_trait = lifestyle_bm_acolyte
		remove_trait = lifestyle_blood_mage
		remove_trait = lifestyle_bm_archmage
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_10000 }
		random = {
			chance = 95
			remove_trait = lifestyle_bm_acolyte
			remove_trait = lifestyle_blood_mage
			remove_trait = lifestyle_bm_archmage
		}
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_1000 }
		random = {
			chance = 50
			remove_trait = lifestyle_bm_acolyte
			remove_trait = lifestyle_blood_mage
			remove_trait = lifestyle_bm_archmage
		}
	}
	if = {
		limit = { is_blood_mage_any_tier_trigger = yes }
		add_character_flag = bm_blood_mage_prevalence_reviewed
		bm_ensure_blood_mage_story_effect = yes
	}
	else = {
		if = {
			limit = {
				has_trait = lifestyle_blood_power
			}
			remove_trait = lifestyle_blood_power
		}
		if = {
			limit = {
				NOT = { has_trait = shieldmaiden }
				has_character_modifier = ignores_gender_army_rules
			}
			remove_character_modifier = ignores_gender_army_rules
		}
	}
}
