bm_ensure_blood_mage_story_effect = {
	if = {
		limit = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
			NOT = { any_owned_story = { story_type = bm_blood_mage_story } }
		}
		create_story = bm_blood_mage_story
	}
	if = {
		limit = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
			NOT = { has_trait = lifestyle_blood_power }
		}
		add_trait = lifestyle_blood_power
	}
	if = {
		limit = {
			has_trait = lifestyle_blood_knight
			NOT = { has_character_modifier = ignores_gender_army_rules }
		}
		add_character_modifier = ignores_gender_army_rules
	}
}

bm_become_blood_mage_effect = {
	if = {
		limit = {
			NOT = { has_trait = lifestyle_blood_mage }
			NOT = { has_trait = lifestyle_blood_knight }
		}
		add_trait = lifestyle_blood_mage
		if = {
			limit = { NOT = { has_trait = lifestyle_blood_power } }
			add_trait = lifestyle_blood_power
		}
	}
	bm_ensure_blood_mage_story_effect = yes
	if = {
		limit = { has_trait = lifestyle_blood_mage }
		add_character_flag = bm_blood_mage_prevalence_reviewed
	}
}

bm_apply_blood_mage_prevalence_retention_effect = {
	if = {
		limit = { has_game_rule = bm_prevalence_player_only }
		remove_trait = lifestyle_blood_mage
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_10000 }
		random = {
			chance = 95
			remove_trait = lifestyle_blood_mage
		}
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_1000 }
		random = {
			chance = 50
			remove_trait = lifestyle_blood_mage
		}
	}
	if = {
		limit = { has_trait = lifestyle_blood_mage }
		add_character_flag = bm_blood_mage_prevalence_reviewed
		bm_ensure_blood_mage_story_effect = yes
	}
	else = {
		if = {
			limit = {
				NOT = { has_trait = lifestyle_blood_knight }
				has_trait = lifestyle_blood_power
			}
			remove_trait = lifestyle_blood_power
		}
	}
}

bm_transmute_to_blood_knight_effect = {
	set_variable = {
		name = bm_transmute_ancient_xp
		value = bm_blood_mage_ancient_xp_value
	}
	set_variable = {
		name = bm_transmute_enlightenment_xp
		value = bm_blood_mage_enlightenment_xp_value
	}
	set_variable = {
		name = bm_transmute_benediction_xp
		value = bm_blood_mage_benediction_xp_value
	}
	remove_character_modifier = lifeforce_modifier_superior
	blood_magic_used_superior = yes
	add_character_flag = {
		flag = bm_swap_class_cooldown
		years = 5
	}
	remove_trait = lifestyle_blood_mage
	bm_become_blood_knight_effect = yes
	if = {
		limit = { bm_transmute_ancient_xp_value > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_knight
			track = ancient
			value = bm_transmute_ancient_xp_value
		}
	}
	if = {
		limit = { bm_transmute_enlightenment_xp_value > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_knight
			track = enlightenment
			value = bm_transmute_enlightenment_xp_value
		}
	}
	if = {
		limit = { bm_transmute_benediction_xp_value > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_knight
			track = benediction
			value = bm_transmute_benediction_xp_value
		}
	}
	remove_variable = bm_transmute_ancient_xp
	remove_variable = bm_transmute_enlightenment_xp
	remove_variable = bm_transmute_benediction_xp
	hidden_effect = {
		bm_recount_lifeforce_stacks_effect = yes
	}
}
