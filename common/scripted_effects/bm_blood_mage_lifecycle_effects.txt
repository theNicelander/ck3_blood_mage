bm_ensure_blood_mage_story_effect = {
	if = {
		limit = {
			is_blood_mage_any_tier_trigger = yes
			NOT = { any_owned_story = { story_type = bm_blood_mage_story } }
		}
		create_story = bm_blood_mage_story
	}
	if = {
		limit = {
			is_blood_mage_any_tier_trigger = yes
			NOT = { has_trait = lifestyle_blood_power }
		}
		add_trait = lifestyle_blood_power
	}
	if = {
		limit = {
			OR = {
				is_blood_mage_tier_2_or_higher_trigger = yes
				bm_has_track_xp_any_tier_trigger = { TRACK = bloodlust VALUE = 20 }
			}
			NOT = { has_character_modifier = ignores_gender_army_rules }
		}
		add_character_modifier = ignores_gender_army_rules
	}
}

bm_become_blood_acolyte_effect = {
	if = {
		limit = {
			NOT = { is_blood_mage_any_tier_trigger = yes }
		}
		add_trait = lifestyle_bm_acolyte
		if = {
			limit = { NOT = { has_trait = lifestyle_blood_power } }
			add_trait = lifestyle_blood_power
		}
	}
	bm_ensure_blood_mage_story_effect = yes
	if = {
		limit = { is_blood_mage_any_tier_trigger = yes }
		add_character_flag = bm_blood_mage_prevalence_reviewed
	}
}

bm_become_blood_mage_tier_2_effect = {
	if = {
		limit = { has_trait = lifestyle_bm_acolyte }
		remove_trait = lifestyle_bm_acolyte
	}
	if = {
		limit = { NOT = { has_trait = lifestyle_blood_mage } }
		add_trait = lifestyle_blood_mage
	}
	if = {
		limit = { NOT = { has_trait = lifestyle_blood_power } }
		add_trait = lifestyle_blood_power
	}
	bm_ensure_blood_mage_story_effect = yes
	add_character_flag = bm_blood_mage_prevalence_reviewed
}

bm_become_blood_archmage_effect = {
	if = {
		limit = { has_trait = lifestyle_blood_mage }
		remove_trait = lifestyle_blood_mage
	}
	if = {
		limit = { has_trait = lifestyle_bm_acolyte }
		remove_trait = lifestyle_bm_acolyte
	}
	if = {
		limit = { NOT = { has_trait = lifestyle_bm_archmage } }
		add_trait = lifestyle_bm_archmage
	}
	if = {
		limit = { NOT = { has_trait = lifestyle_blood_power } }
		add_trait = lifestyle_blood_power
	}
	bm_ensure_blood_mage_story_effect = yes
	add_character_flag = bm_blood_mage_prevalence_reviewed
}

bm_become_blood_mage_effect = {
	bm_become_blood_acolyte_effect = yes
}

bm_elevate_tier_effect = {
	if = {
		limit = { has_trait = lifestyle_bm_acolyte }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_elevate_blood_mage_tier_toast
			left_icon = root
			if = {
				limit = { bm_elevation_overflow_lifeforce_stacks_value > 0 }
				while = {
					count = bm_elevation_overflow_lifeforce_stacks_value
					add_character_modifier = lifeforce_modifier_minor
				}
			}
			remove_trait = lifestyle_bm_acolyte
			add_trait = lifestyle_blood_mage
			hidden_effect = {
				bm_ensure_blood_mage_story_effect = yes
			}
		}
	}
	else_if = {
		limit = { has_trait = lifestyle_blood_mage }
		send_interface_toast = {
			type = event_toast_effect_good
			title = bm_elevate_blood_archmage_tier_toast
			left_icon = root
			if = {
				limit = { bm_elevation_overflow_lifeforce_stacks_value > 0 }
				while = {
					count = bm_elevation_overflow_lifeforce_stacks_value
					add_character_modifier = lifeforce_modifier_major
				}
			}
			remove_trait = lifestyle_blood_mage
			add_trait = lifestyle_bm_archmage
			hidden_effect = {
				bm_ensure_blood_mage_story_effect = yes
			}
		}
	}
}

bm_apply_blood_mage_prevalence_retention_effect = {
	if = {
		limit = { has_game_rule = bm_prevalence_player_only }
		remove_trait = lifestyle_bm_acolyte
		remove_trait = lifestyle_blood_mage
		remove_trait = lifestyle_bm_archmage
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_10000 }
		random = {
			chance = 95
			remove_trait = lifestyle_bm_acolyte
			remove_trait = lifestyle_blood_mage
			remove_trait = lifestyle_bm_archmage
		}
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_1000 }
		random = {
			chance = 50
			remove_trait = lifestyle_bm_acolyte
			remove_trait = lifestyle_blood_mage
			remove_trait = lifestyle_bm_archmage
		}
	}
	if = {
		limit = { is_blood_mage_any_tier_trigger = yes }
		add_character_flag = bm_blood_mage_prevalence_reviewed
		bm_ensure_blood_mage_story_effect = yes
	}
	else = {
		if = {
			limit = {
				has_trait = lifestyle_blood_power
			}
			remove_trait = lifestyle_blood_power
		}
		if = {
			limit = {
				NOT = { has_trait = shieldmaiden }
				has_character_modifier = ignores_gender_army_rules
			}
			remove_character_modifier = ignores_gender_army_rules
		}
	}
}
