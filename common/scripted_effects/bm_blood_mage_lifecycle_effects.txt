bm_ensure_blood_mage_story_effect = {
	if = {
		limit = {
			has_trait = lifestyle_blood_mage
			NOT = { any_owned_story = { story_type = bm_blood_mage_story } }
		}
		create_story = bm_blood_mage_story
	}
}

bm_become_blood_mage_effect = {
	if = {
		limit = { NOT = { has_trait = lifestyle_blood_mage } }
		add_trait = lifestyle_blood_mage
	}
	bm_ensure_blood_mage_story_effect = yes
	if = {
		limit = { has_trait = lifestyle_blood_mage }
		add_character_flag = bm_blood_mage_prevalence_reviewed
	}
}

bm_apply_blood_mage_prevalence_retention_effect = {
	if = {
		limit = { has_game_rule = bm_prevalence_player_only }
		remove_trait = lifestyle_blood_mage
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_10000 }
		random = {
			chance = 95
			remove_trait = lifestyle_blood_mage
		}
	}
	else_if = {
		limit = { has_game_rule = bm_prevalence_one_in_1000 }
		random = {
			chance = 50
			remove_trait = lifestyle_blood_mage
		}
	}
	if = {
		limit = { has_trait = lifestyle_blood_mage }
		add_character_flag = bm_blood_mage_prevalence_reviewed
		bm_ensure_blood_mage_story_effect = yes
	}
}

bm_elevate_blood_knight_to_blood_mage_effect = {
	set_variable = { name = bm_transferred_ancient_xp value = bm_blood_knight_ancient_xp_value }
	set_variable = { name = bm_transferred_benediction_xp value = bm_blood_knight_benediction_xp_value }
	set_variable = { name = bm_transferred_hematurgy_xp value = bm_converted_hematurgy_xp_value }
	set_variable = { name = bm_transferred_bloodline_xp value = bm_converted_bloodline_xp_value }
	set_variable = { name = bm_transferred_enlightenment_xp value = bm_blood_knight_enlightenment_xp_value }
	remove_trait = lifestyle_blood_knight
	bm_become_blood_mage_effect = yes
	if = {
		limit = { var:bm_transferred_ancient_xp > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_mage
			track = ancient
			value = var:bm_transferred_ancient_xp
		}
	}
	if = {
		limit = { var:bm_transferred_benediction_xp > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_mage
			track = benediction
			value = var:bm_transferred_benediction_xp
		}
	}
	if = {
		limit = { var:bm_transferred_hematurgy_xp > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_mage
			track = hematurgy
			value = var:bm_transferred_hematurgy_xp
		}
	}
	if = {
		limit = { var:bm_transferred_bloodline_xp > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_mage
			track = bloodline
			value = var:bm_transferred_bloodline_xp
		}
	}
	if = {
		limit = { var:bm_transferred_enlightenment_xp > 0 }
		add_trait_xp = {
			trait = lifestyle_blood_mage
			track = enlightenment
			value = var:bm_transferred_enlightenment_xp
		}
	}
	remove_variable = bm_transferred_ancient_xp
	remove_variable = bm_transferred_benediction_xp
	remove_variable = bm_transferred_hematurgy_xp
	remove_variable = bm_transferred_bloodline_xp
	remove_variable = bm_transferred_enlightenment_xp
}
