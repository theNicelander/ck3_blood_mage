bm_synchronize_character_skills_effect = {
	# Diplomacy
	set_variable = {
		name = bm_skill_diff
		value = 20
	}
	change_variable = {
		name = bm_skill_diff
		subtract = diplomacy
	}
	if = {
		limit = { var:bm_skill_diff > 0 }
		add_diplomacy_skill = var:bm_skill_diff
	}
	else_if = {
		limit = { var:bm_skill_diff < 0 }
		add_diplomacy_skill = var:bm_skill_diff
	}

	# Martial
	set_variable = {
		name = bm_skill_diff

	}

		value = 20
	}

		name = bm_skill_diff
		subtract = learning
	}
	if = {
		limit = { var:bm_skill_diff > 0 }
		add_learning_skill = var:bm_skill_diff
	}
	else_if = {
		limit = { var:bm_skill_diff < 0 }
		add_learning_skill = var:bm_skill_diff
	}

	# Prowess
	set_variable = {
		name = bm_skill_diff
		value = root.prowess
	}
	change_variable = {
		name = bm_skill_diff
		subtract = prowess
	}
	if = {
		limit = { var:bm_skill_diff > 0 }

	}
	else_if = {
		limit = { var:bm_skill_diff < 0 }
		add_prowess_skill = var:bm_skill_diff
	}

	remove_variable = bm_skill_diff
	force_character_skill_recalculation = yes
}
