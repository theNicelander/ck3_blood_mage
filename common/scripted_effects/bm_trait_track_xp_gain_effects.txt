bm_add_blood_mage_xp_effect = {
	if = {
		limit = {
			has_trait = lifestyle_bm_acolyte
		}
		add_trait_xp = {
			trait = lifestyle_bm_acolyte
			track = $TRACK$
			value = $XP$
		}
	}
	else_if = {
		limit = {
			has_trait = lifestyle_blood_mage
		}
		add_trait_xp = {
			trait = lifestyle_blood_mage
			track = $TRACK$
			value = $XP$
		}
	}
	else_if = {
		limit = {
			has_trait = lifestyle_bm_archmage
		}
		add_trait_xp = {
			trait = lifestyle_bm_archmage
			track = $TRACK$
			value = $XP$
		}
	}
}
