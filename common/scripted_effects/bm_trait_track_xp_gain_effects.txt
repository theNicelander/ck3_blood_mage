bm_add_blood_mage_xp_effect = {
    if = {
        limit = {
            has_trait = lifestyle_blood_mage
        }
        add_trait_xp = {
            trait = lifestyle_blood_mage
            track = $TRACK$
            value = $XP$
        }
    }
}
