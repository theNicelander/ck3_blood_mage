# Blood Runes & Galdrastafir Scripted Effects

bm_strip_all_stave_modifiers_effect = {
	if = { limit = { has_character_modifier = bm_rune_vegvisir_minor } remove_character_modifier = bm_rune_vegvisir_minor }
	if = { limit = { has_character_modifier = bm_rune_vegvisir_major } remove_character_modifier = bm_rune_vegvisir_major }
	if = { limit = { has_character_modifier = bm_rune_vegvisir_superior } remove_character_modifier = bm_rune_vegvisir_superior }

	if = { limit = { has_character_modifier = bm_rune_aegishjalmur_minor } remove_character_modifier = bm_rune_aegishjalmur_minor }
	if = { limit = { has_character_modifier = bm_rune_aegishjalmur_major } remove_character_modifier = bm_rune_aegishjalmur_major }
	if = { limit = { has_character_modifier = bm_rune_aegishjalmur_superior } remove_character_modifier = bm_rune_aegishjalmur_superior }

	if = { limit = { has_character_modifier = bm_rune_feingur_minor } remove_character_modifier = bm_rune_feingur_minor }
	if = { limit = { has_character_modifier = bm_rune_feingur_major } remove_character_modifier = bm_rune_feingur_major }
	if = { limit = { has_character_modifier = bm_rune_feingur_superior } remove_character_modifier = bm_rune_feingur_superior }

	if = { limit = { has_character_modifier = bm_rune_nabrokarstafur_minor } remove_character_modifier = bm_rune_nabrokarstafur_minor }
	if = { limit = { has_character_modifier = bm_rune_nabrokarstafur_major } remove_character_modifier = bm_rune_nabrokarstafur_major }
	if = { limit = { has_character_modifier = bm_rune_nabrokarstafur_superior } remove_character_modifier = bm_rune_nabrokarstafur_superior }

	if = { limit = { has_character_modifier = bm_rune_hulinhjalmur_minor } remove_character_modifier = bm_rune_hulinhjalmur_minor }
	if = { limit = { has_character_modifier = bm_rune_hulinhjalmur_major } remove_character_modifier = bm_rune_hulinhjalmur_major }
	if = { limit = { has_character_modifier = bm_rune_hulinhjalmur_superior } remove_character_modifier = bm_rune_hulinhjalmur_superior }
}

bm_apply_selected_stave_at_current_tier_effect = {
	hidden_effect = {
		bm_strip_all_stave_modifiers_effect = yes
	}
	if = {
		limit = { has_character_modifier = superior_blood_rune_modifier }
		add_character_modifier = { modifier = bm_rune_$STAVE$_superior }
	}
	else_if = {
		limit = { has_character_modifier = major_blood_rune_modifier }
		add_character_modifier = { modifier = bm_rune_$STAVE$_major }
	}
	else = {
		hidden_effect = {
			if = {
				limit = { NOT = { has_character_modifier = minor_blood_rune_modifier } }
				add_character_modifier = { modifier = minor_blood_rune_modifier }
			}
		}
		add_character_modifier = { modifier = bm_rune_$STAVE$_minor }
	}
}

bm_upgrade_blood_rune_to_major_effect = {
	remove_character_modifier = minor_blood_rune_modifier
	add_character_modifier = { modifier = major_blood_rune_modifier }
	if = {
		limit = { has_character_modifier = bm_rune_vegvisir_minor }
		remove_character_modifier = bm_rune_vegvisir_minor
		add_character_modifier = { modifier = bm_rune_vegvisir_major }
	}
	else_if = {
		limit = { has_character_modifier = bm_rune_aegishjalmur_minor }
		remove_character_modifier = bm_rune_aegishjalmur_minor
		add_character_modifier = { modifier = bm_rune_aegishjalmur_major }
	}
	else_if = {
		limit = { has_character_modifier = bm_rune_feingur_minor }
		remove_character_modifier = bm_rune_feingur_minor
		add_character_modifier = { modifier = bm_rune_feingur_major }
	}
	else_if = {
		limit = { has_character_modifier = bm_rune_nabrokarstafur_minor }
		remove_character_modifier = bm_rune_nabrokarstafur_minor
		add_character_modifier = { modifier = bm_rune_nabrokarstafur_major }
	}
	else_if = {
		limit = { has_character_modifier = bm_rune_hulinhjalmur_minor }
		remove_character_modifier = bm_rune_hulinhjalmur_minor
		add_character_modifier = { modifier = bm_rune_hulinhjalmur_major }
	}
	else = {
		save_scope_as = rune_target
		trigger_event = { id = bm_blood_runes.001 }
	}
}
