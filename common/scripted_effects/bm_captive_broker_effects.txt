bm_purchase_broker_captive_effect = {
	remove_short_term_gold = $TARGET$.$PRICE$
	imprison = {
		target = $TARGET$
		type = dungeon
	}
	send_interface_toast = {
		type = $TOAST_TYPE$
		title = $TOAST_TITLE$
		left_icon = $TARGET$
	}
	hidden_effect = {
		add_character_flag = {
			flag = bm_bought_captive_session
			days = 30
		}
	}
}

bm_cleanup_unbought_captives_effect = {
	hidden_effect = {
		if = {
			limit = {
				exists = scope:captive_1
				scope:captive_1 = {
					is_alive = yes
					is_imprisoned = no
				}
			}
			scope:captive_1 = { silent_disappearance_effect = yes }
		}
		if = {
			limit = {
				exists = scope:captive_2
				scope:captive_2 = {
					is_alive = yes
					is_imprisoned = no
				}
			}
			scope:captive_2 = { silent_disappearance_effect = yes }
		}
		if = {
			limit = {
				exists = scope:captive_3
				scope:captive_3 = {
					is_alive = yes
					is_imprisoned = no
				}
			}
			scope:captive_3 = { silent_disappearance_effect = yes }
		}
	}
}

bm_finalize_captive_broker_cooldown_effect = {
	hidden_effect = {
		if = {
			limit = { has_character_flag = bm_bought_captive_session }
			remove_character_flag = bm_bought_captive_session
			add_character_flag = {
				flag = bm_buy_captives_cooldown
				years = 1
			}
		}
		else = {
			add_character_flag = {
				flag = bm_buy_captives_cooldown
				days = 90
			}
		}
	}
}
