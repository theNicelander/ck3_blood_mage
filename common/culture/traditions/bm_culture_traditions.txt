bm_tradition_althingi = {
	category = realm

	layers = {
		0 = diplo
		1 = western
		4 = council.dds
	}

	is_shown = {
		OR = {
			has_cultural_pillar = heritage_north_germanic
			culture_is_or_has_ancestor_trigger = { CULTURE = culture:norse }
			this = culture:bm_icelandic
		}
	}

	can_pick = {
		trigger_if = {
			limit = {
				OR = {
					NOT = { exists = scope:replacing }
					NOT = { scope:replacing = culture_tradition:tradition_things }
				}
			}
			NOT = { has_cultural_tradition = tradition_things }
		}
	}

	parameters = {
		scandinavian_elective_enabled = yes
		has_access_to_trials_by_combat = yes
	}

	character_modifier = {
		character_capital_county_monthly_prosperity_growth_add = 0.05
		cultural_head_fascination_mult = 0.15
	}

	county_modifier = {
		prosperity_growth_factor = 0.2
		control_baseline_add = 5
		monthly_control_change_when_below_baseline_add = 0.20
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_communal = { is_in_list = traits }
						culture_pillar:ethos_bureaucratic = { is_in_list = traits }
						culture_pillar:ethos_bellicose = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = bm_not_communal_bureaucratic_or_bellicose_desc
				}
			}
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 100
		if = {
			limit = {
				has_cultural_pillar = heritage_north_germanic
			}
			multiply = 2
		}
	}
}

bm_tradition_verbudir = {
	category = realm

	layers = {
		0 = steward
		1 = western
		4 = boat.dds
	}

	is_shown = {
		OR = {
			has_cultural_pillar = heritage_north_germanic
			culture_is_or_has_ancestor_trigger = { CULTURE = culture:norse }
			this = culture:bm_icelandic
		}
	}

	can_pick = {
		custom_description = {
			text = culture_on_coast_desc
			any_culture_county = {
				has_coastal_province = yes
			}
		}
		trigger_if = {
			limit = {
				OR = {
					NOT = { exists = scope:replacing }
					NOT = { scope:replacing = culture_tradition:tradition_maritime_mercantilism }
				}
			}
			NOT = { has_cultural_tradition = tradition_maritime_mercantilism }
		}
		trigger_if = {
			limit = {
				OR = {
					NOT = { exists = scope:replacing }
					NOT = { scope:replacing = culture_tradition:tradition_fishermen }
				}
			}
			NOT = { has_cultural_tradition = tradition_fishermen }
		}
	}

	parameters = {
		next_level_port_districts = yes
		port_district_enabled_for_tribals = yes
		maritime_mercantilism_coastal_districts = yes
		can_travel_along_rivers = yes
		fishermen_productivity_bonus = yes
	}

	character_modifier = {
		embarkation_cost_mult = -0.5
		sea_travel_danger = medium_sea_danger_reduction
		coastal_sea_travel_danger = low_sea_danger_reduction
		max_trade_goods_on_route = 1
		trade_capacity_mult = 0.1
		trade_route_length_add = 3
		trade_category_meat_xp_gain_mult = 0.2
		trade_good_fish_sell_price_mult = 0.02
	}

	county_modifier = {
		trade_good_fish_production_mult = 0.2
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_communal = { is_in_list = traits }
						culture_pillar:ethos_stoic = { is_in_list = traits }
						culture_pillar:ethos_bureaucratic = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_communal_bureaucratic_or_stoic_desc
				}
			}
			if = {
				limit = {
					NOT = {
						any_culture_county = {
							percent >= 0.3
							has_coastal_province = yes
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = coastal_percentage_desc
				}
			}
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 100
		if = {
			limit = {
				has_cultural_pillar = heritage_north_germanic
			}
			multiply = 2
		}
	}
}
