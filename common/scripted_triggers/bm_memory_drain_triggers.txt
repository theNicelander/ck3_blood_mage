# Scripted triggers for Drain Blood Memory (curated lifestyle, commander, and martial fame traits)

bm_can_drain_memory_no_track_trigger = {
	scope:recipient = { has_trait = $TRAIT$ }
	NOT = { has_trait = $TRAIT$ }
}

bm_can_drain_memory_single_track_trigger = {
	scope:recipient = { has_trait = $TRAIT$ }
	OR = {
		NOT = { has_trait = $TRAIT$ }
		has_trait_xp = {
			trait = $TRAIT$
			value < 100
		}
	}
}

bm_can_drain_memory_multi_track_trigger = {
	scope:recipient = { has_trait = $TRAIT$ }
	OR = {
		NOT = { has_trait = $TRAIT$ }
		has_trait_xp = {
			trait = $TRAIT$
			track = $TRACK_A$
			value < 100
		}
		has_trait_xp = {
			trait = $TRAIT$
			track = $TRACK_B$
			value < 100
		}
	}
}

bm_can_drain_memory_quad_track_trigger = {

		}

}

bm_can_drain_memory_from_target_trigger = {
	custom_description = {
		text = bm_can_drain_memory_from_target_trigger_tt
		subject = scope:recipient
		OR = {
			bm_can_drain_memory_lifestyle_trigger = yes
			bm_can_drain_memory_commander_trigger = yes
			bm_can_drain_memory_fame_trigger = yes
			bm_compat_can_drain_memory_from_target_trigger = yes
		}
	}
}

		amount >= 2

		bm_can_drain_memory_fame_trigger = yes
		bm_compat_can_drain_memory_from_target_trigger = yes
	}
}

bm_can_afford_memory_drain_no_track_trigger = {
	bm_can_drain_memory_no_track_trigger = { TRAIT = $TRAIT$ }
	piety >= bm_memory_drain_piety_cost
}

bm_can_afford_memory_drain_single_track_trigger = {
	bm_can_drain_memory_single_track_trigger = { TRAIT = $TRAIT$ }
	OR = {
		AND = {
			has_trait = $TRAIT$
			piety >= bm_memory_drain_xp_piety_cost
		}
		piety >= bm_memory_drain_piety_cost
	}
}
