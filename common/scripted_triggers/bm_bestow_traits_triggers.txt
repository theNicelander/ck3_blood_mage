bm_bestow_traits_can_bestow_intellect_trigger = {
	scope:actor = { has_trait = intellect_good }
	OR = {
		AND = {
			scope:actor = { has_trait = intellect_good_3 }
			NOT = { scope:recipient = { has_trait = intellect_good_3 } }
		}
		AND = {
			scope:actor = { has_trait = intellect_good_2 }
			NOT = {
				scope:recipient = {
					OR = {
						has_trait = intellect_good_2
						has_trait = intellect_good_3
					}
				}
			}
		}
		AND = {
			scope:actor = { has_trait = intellect_good_1 }
			NOT = { scope:recipient = { has_trait = intellect_good } }
		}
	}
}
