# Elder Kings 2 (2887120253) Trait Drain Compatibility Triggers
# All EK2-only references are isolated here and gated behind bm_is_ek2_loaded_trigger.

# Shared EK2 helpers
bm_compat_ek2_has_transformation_trigger = {
	OR = {
		has_trait = vampire_character
		has_trait = lycan_character
		has_trait = lich_character
		has_trait = briarheart
		has_trait = hagraven
	}
}

bm_compat_ek2_can_drain_ranked_trigger = {
	bm_is_ek2_loaded_trigger = yes
	NOT = { has_trait_rank = { trait = $TRAIT$ rank >= $MAX_RANK$ } }
	scope:recipient = { has_trait = $TRAIT$ }
}

bm_compat_ek2_can_afford_tier4_trigger = {
	trigger_if = {
		limit = { has_trait_rank = { trait = $TRAIT$ rank >= 3 } }
		piety >= 500
		piety_level >= 2
	}
	trigger_else_if = {
		limit = { has_trait_rank = { trait = $TRAIT$ rank >= 2 } }
		piety >= 350
	}
	trigger_else = {
		piety >= 200
	}
}

bm_compat_ek2_can_afford_hist_trigger = {
	trigger_if = {
		limit = { has_trait_rank = { trait = $TRAIT$ rank >= 2 } }
		piety >= 350
	}
	trigger_else_if = {
		limit = { has_trait = $TRAIT$ }
		piety >= 250
	}
	trigger_else = {
		piety >= 150
	}
}
