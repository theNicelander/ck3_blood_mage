# Elder Kings 2 (2887120253) Trait Drain Compatibility Triggers
# All EK2-only references are isolated here and gated behind bm_is_ek2_loaded_trigger.

bm_compat_ek2_can_drain_arcana_trigger = {
	bm_is_ek2_loaded_trigger = yes
	NOT = { has_trait_rank = { trait = arcana_good rank >= 3 } }
	scope:recipient = { has_trait = arcana_good }
}

bm_compat_ek2_can_drain_magickal_trigger = {
	bm_is_ek2_loaded_trigger = yes
	NOR = {
		has_trait = magickal
		has_trait = arcana_bad
	}
	scope:recipient = { has_trait = magickal }
}
