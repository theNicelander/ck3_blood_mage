bm_is_agot_loaded_trigger = {
	# AGOT (2962333032)
	has_global_variable = AGOT_is_loaded
}

bm_is_lotr_loaded_trigger = {
	# Lord of the Rings: Realms in Exile (2291024373)
	OR = {
		has_global_variable = tedjin_reactionaries_triggered
		has_global_variable = sauron_destroyed
	}
}

bm_is_ek2_loaded_trigger = {
	# Elder Kings 2 (2887120253)
	OR = {
		has_global_variable = morrowind_high_councilor
		has_global_variable = almsivi_almalexia_power
		has_global_variable = almsivi_sotha_power
		has_global_variable = almsivi_vivec_power
	}
}

is_playing_overhaul_mod = {
	OR = {
		bm_is_lotr_loaded_trigger = yes
		bm_is_ek2_loaded_trigger = yes
		bm_is_agot_loaded_trigger = yes
	}
}

bm_has_superior_stave_trigger = {
	has_character_modifier = superior_blood_rune_modifier
	has_character_modifier = bm_stave_$STAVE$
}

bm_has_any_stave_trigger = {
	OR = {
		has_character_modifier = bm_stave_vegvisir
		has_character_modifier = bm_stave_aegishjalmur
		has_character_modifier = bm_stave_feingur
		has_character_modifier = bm_stave_nabrokarstafur
		has_character_modifier = bm_stave_hulinhjalmur
	}
}

bm_has_blood_rune_trigger = {
	OR = {
		has_character_modifier = minor_blood_rune_modifier
		has_character_modifier = major_blood_rune_modifier
		has_character_modifier = superior_blood_rune_modifier
	}
}

bm_is_valid_positive_magic_target_trigger = {
	OR = {
		is_courtier_of = $ACTOR$
		is_vassal_of = $ACTOR$
		is_child_of = $ACTOR$
		is_close_family_of = $ACTOR$
		is_extended_family_of = $ACTOR$
		dynasty ?= $ACTOR$.dynasty
		is_spouse_of = $ACTOR$
		is_consort_of = $ACTOR$
		has_relation_lover = $ACTOR$
		has_relation_soulmate = $ACTOR$
		has_relation_friend = $ACTOR$
		has_relation_best_friend = $ACTOR$
		is_knight_of = $ACTOR$
		is_councillor_of = $ACTOR$
		employer ?= $ACTOR$
		has_relation_guardian = $ACTOR$
		has_relation_ward = $ACTOR$
		bm_is_blood_golem_trigger = yes
		trigger_if = {
			limit = { exists = $ACTOR$.liege }
			OR = {
				this = $ACTOR$.liege
				is_courtier_of = $ACTOR$.liege
			}
		}
	}
}

bm_has_stress_coping_trait_trigger = {
	OR = {
		has_trait = drunkard
		has_trait = hashishiyah
		has_trait = rakish
		has_trait = reclusive
		has_trait = irritable
		has_trait = flagellant
		has_trait = profligate
		has_trait = improvident
		has_trait = contrite
		has_trait = comfort_eater
		has_trait = inappetetic
	}
}

is_blood_mage_any_tier_trigger = {
	OR = {
		has_trait = lifestyle_bm_acolyte
		has_trait = lifestyle_blood_mage
		has_trait = lifestyle_bm_archmage
	}
}

is_blood_mage_tier_2_or_higher_trigger = {
	OR = {
		has_trait = lifestyle_blood_mage
		has_trait = lifestyle_bm_archmage
	}
}

is_blood_mage_archmage_trigger = {
	has_trait = lifestyle_bm_archmage
}

bm_is_blood_golem_trigger = {
	OR = {
		has_trait = bm_blood_golem
		house ?= house:bm_house_golem
	}
}

bm_has_lifeforce_shield_trigger = {
	is_blood_mage_any_tier_trigger = yes
	OR = {
		has_character_modifier = lifeforce_modifier_superior
		has_character_modifier = lifeforce_modifier_major
		has_character_modifier = lifeforce_modifier_minor
	}
}

is_blood_mage_acolyte_trigger = {
	has_trait = lifestyle_bm_acolyte
}

bm_has_track_xp_any_tier_trigger = {
	OR = {
		has_trait_xp = {
			trait = lifestyle_bm_acolyte
			track = $TRACK$
			value >= $VALUE$
		}
		has_trait_xp = {
			trait = lifestyle_blood_mage
			track = $TRACK$
			value >= $VALUE$
		}
		has_trait_xp = {
			trait = lifestyle_bm_archmage
			track = $TRACK$
			value >= $VALUE$
		}
	}
}
