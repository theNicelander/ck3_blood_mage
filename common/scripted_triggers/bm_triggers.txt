is_playing_overhaul_mod = {
    OR = {
        # Lord of the Rings (2291024373)
        has_global_variable  = sauron_destroyed
        # Elder Kings (2887120253)
        OR = {
            has_global_variable = almsivi_almalexia_power
            has_global_variable = almsivi_sotha_power
            has_global_variable = almsivi_vivec_power
        }
        # AGOT (2962333032)
        has_global_variable = AGOT_is_loaded
    }
}

bm_has_superior_stave_trigger = {
	has_character_modifier = superior_blood_rune_modifier
	has_character_modifier = bm_stave_$STAVE$
}

bm_has_any_stave_trigger = {
	OR = {
		has_character_modifier = bm_stave_vegvisir
		has_character_modifier = bm_stave_aegishjalmur
		has_character_modifier = bm_stave_feingur
		has_character_modifier = bm_stave_nabrokarstafur
		has_character_modifier = bm_stave_hulinhjalmur
	}
}

bm_has_blood_rune_trigger = {
	OR = {
		has_character_modifier = minor_blood_rune_modifier
		has_character_modifier = major_blood_rune_modifier
		has_character_modifier = superior_blood_rune_modifier
	}
}

bm_is_valid_positive_magic_target_trigger = {
	OR = {
		is_courtier_of = $ACTOR$
		is_vassal_of = $ACTOR$
		is_child_of = $ACTOR$
		is_close_family_of = $ACTOR$
		is_extended_family_of = $ACTOR$
		dynasty ?= $ACTOR$.dynasty
		is_spouse_of = $ACTOR$
		is_consort_of = $ACTOR$
		has_relation_lover = $ACTOR$
		has_relation_soulmate = $ACTOR$
		has_relation_friend = $ACTOR$
		has_relation_best_friend = $ACTOR$
		is_knight_of = $ACTOR$
		is_councillor_of = $ACTOR$
		has_relation_guardian = $ACTOR$
		has_relation_ward = $ACTOR$
		house ?= house:bm_house_golem
		trigger_if = {
			limit = { exists = $ACTOR$.liege }
			OR = {
				this = $ACTOR$.liege
				is_courtier_of = $ACTOR$.liege
			}
		}
	}
}
