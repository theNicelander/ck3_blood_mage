bm_has_minor_ailment_trigger = {
	OR = {
		has_trait = wounded_1
		has_trait = ill
		has_trait = lovers_pox
		has_trait = gout_ridden
		has_trait = scarred
	}
}

bm_has_major_ailment_trigger = {
	OR = {
		has_trait = wounded_2
		has_trait = incapable
		has_trait = pneumonic
		has_trait = early_great_pox
		has_trait = maimed
		has_trait = consumption
		has_trait = typhus
		has_trait = measles
		has_trait = blind
		has_trait = one_eyed
		has_trait = weak
		has_trait = lunatic_1
		has_trait = possessed_1
		has_trait = dull
		has_trait = depressed_1
	}
}

bm_has_old_age_ailment_trigger = {
	OR = {
		has_trait = infirm
		has_trait = withering_mind
		has_trait = clouded_eyes
		has_trait = fragile_bones
		has_trait = faltering_heart
	}
}

bm_has_deadly_ailment_trigger = {
	OR = {
		has_trait = wounded_3
		has_trait = sickly
		has_trait = great_pox
		has_trait = cancer
		has_trait = bubonic_plague
		has_trait = smallpox
		has_trait = leper
		has_trait = impotent
		has_trait = dysentery
		has_trait = ergotism
		has_trait = disfigured
		has_trait = one_legged
	}
}

bm_has_non_old_age_ailment_trigger = {
	OR = {
		bm_has_minor_ailment_trigger = yes
		bm_has_major_ailment_trigger = yes
		bm_has_deadly_ailment_trigger = yes
		bm_has_stress_coping_trait_trigger = yes
	}
}

bm_has_any_ailment_trigger = {
	OR = {
		bm_has_minor_ailment_trigger = yes
		bm_has_major_ailment_trigger = yes
		bm_has_deadly_ailment_trigger = yes
		bm_has_old_age_ailment_trigger = yes
		bm_has_stress_coping_trait_trigger = yes
	}
}

bm_actor_can_heal_target_ailments_trigger = {
	# TARGET = target character
	# Evaluated in actor scope
	has_trait = lifestyle_blood_mage
	$TARGET$ = { bm_has_any_ailment_trigger = yes }
}

bm_actor_has_lifeforce_to_heal_target_trigger = {
	# TARGET = target character
	# Evaluated in actor scope
	OR = {
		AND = {
			$TARGET$ = { bm_has_minor_ailment_trigger = yes }
			has_character_modifier = lifeforce_modifier_minor
		}
		AND = {
			$TARGET$ = { bm_has_major_ailment_trigger = yes }
			has_character_modifier = lifeforce_modifier_major
		}
		AND = {
			has_trait = lifestyle_blood_mage
			$TARGET$ = { bm_has_deadly_ailment_trigger = yes }
			has_character_modifier = lifeforce_modifier_superior
		}
		AND = {
			$TARGET$ = { bm_has_old_age_ailment_trigger = yes }
			OR = {
				has_character_modifier = lifeforce_modifier_minor
				has_character_modifier = lifeforce_modifier_major
			}
		}
		AND = {
			$TARGET$ = { bm_has_stress_coping_trait_trigger = yes }
			has_character_modifier = lifeforce_modifier_major
		}
	}
}

bm_can_heal_character_trigger = {
	is_alive = yes
	trigger_if = {
		limit = { exists = scope:actor }
		scope:actor = {
			bm_actor_can_heal_target_ailments_trigger = { TARGET = prev }
		}
	}
	trigger_else = {
		root = {
			bm_actor_can_heal_target_ailments_trigger = { TARGET = prev }
		}
	}
}
