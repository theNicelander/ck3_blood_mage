# Scripted triggers for Blood Mage trait and location requirements

bm_can_drain_traits_from_target_trigger = {
	custom_description = {
		text = bm_can_drain_traits_from_target_trigger_tt
		OR = {
			AND = {
				NOT = { has_trait_rank = { trait = intellect_good rank >= 5 } }
				scope:recipient = { has_trait = intellect_good }
			}
			AND = {
				NOT = { has_trait_rank = { trait = physique_good rank >= 5 } }
				scope:recipient = { has_trait = physique_good }
			}
			AND = {
				NOT = { has_trait_rank = { trait = beauty_good rank >= 5 } }
				scope:recipient = { has_trait = beauty_good }
			}
			AND = {
				NOT = { has_trait = fecund }
				scope:recipient = { has_trait = fecund }
			}
			AND = {
				NOT = { has_trait = giant }
				scope:recipient = { has_trait = giant }
			}
			AND = {
				NOT = { has_trait = pure_blooded }
				scope:recipient = { has_trait = pure_blooded }
			}
		}
	}
}

	}
}

		AND = {
			bm_has_rank_1_education_trigger = yes
			piety_level >= 1
		}

			bm_has_rank_2_education_trigger = yes
			piety_level >= 2
		}
