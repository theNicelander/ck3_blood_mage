# Dynamic UI Labels and Status Headers for Empower Dynasty Bloodline Covenants

# 1. Covenant of Majesty
BM_COVENANT_CHARISMA_BUTTON_LABEL = {
	type = character
	text = {
		trigger = { house ?= { has_house_modifier = dynasty_blood_charisma_modifier_2 } }
		localization_key = bm_covenant_charisma_label_major
	}
	text = {
		trigger = { house ?= { has_house_modifier = dynasty_blood_charisma_modifier_1 } }
		localization_key = bm_covenant_charisma_label_minor
	}
	text = {
		fallback = yes
		localization_key = bm_covenant_charisma_label_none
	}
}

BM_COVENANT_CHARISMA_STATUS_HEADER = {
	type = character
	text = {
		trigger = { house ?= { has_house_modifier = dynasty_blood_charisma_modifier_2 } }
		localization_key = bm_covenant_status_major_to_superior
	}
	text = {
		trigger = { house ?= { has_house_modifier = dynasty_blood_charisma_modifier_1 } }
		localization_key = bm_covenant_status_minor_to_major
	}
	text = {
		fallback = yes
		localization_key = bm_covenant_status_none_to_minor
	}
}
