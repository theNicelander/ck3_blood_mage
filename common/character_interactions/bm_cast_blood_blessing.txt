						scope:blessing_ancient_minor = yes

			else_if = {
				limit = {
					scope:blessing_inscribe_rune = yes

				limit = {
					scope:blessing_inscribe_rune = yes

						NOT = { has_character_modifier = minor_blood_rune_modifier }
						NOT = { has_character_modifier = major_blood_rune_modifier }
						NOT = { has_character_modifier = superior_blood_rune_modifier }
					}
				}
				value = 100
			}
			else_if = {
				limit = { scope:blessing_grant_lifeforce_minor = yes }
				value = lifedrain_piety_cost_minor
			}
			else_if = {
				limit = { scope:blessing_grant_lifeforce_major = yes }
				value = lifedrain_piety_cost_major
			}
			else_if = {
				limit = { scope:blessing_grant_lifeforce_superior = yes }
				value = 100
			}
			else_if = {
				limit = { scope:blessing_grant_lifeforce_ancient = yes }
				value = lifedrain_piety_cost_major
			}
			else = {
				value = 0
			}
		}
	}

	is_shown = {

		}

			is_alive = yes
