bm_bless_bloodline_interaction = {
	category = interaction_category_bm_blood_magic
	auto_accept = yes
	common_interaction = yes
	desc = bm_bless_bloodline_interaction_desc
	interface_priority = 195

	is_highlighted = {
		scope:recipient = { is_child_of = scope:actor }
	}

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			is_alive = yes
			OR = {
				is_child_of = scope:actor
				dynasty ?= scope:actor.dynasty
				is_knight_of = scope:actor
				AND = {
					is_courtier_of = scope:actor
					has_trait = lifestyle_blood_knight
				}
			}
			OR = {
				NOR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
				}
				AND = {
					scope:actor = { bm_has_superior_stave_trigger = { STAVE = feingur } }
					OR = {
						is_child_of = scope:actor
						dynasty ?= scope:actor.dynasty
					}
					NOT = { has_character_modifier = bm_feingur_blessed_blood_modifier }
				}
			}
		}
	}

	cost = {
		piety = {
			if = {
				limit = { scope:bless_feingur = yes }
				value = 50
			}
			else = {
				value = 35
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { has_character_modifier = lifeforce_modifier_minor }
	}

	send_options_exclusive = yes

	send_option = {
		flag = bless_sovereign
		localization = bm_bless_bloodline_interaction.bless_sovereign
		current_description = bm_bless_bloodline_interaction.bless_sovereign_desc
		is_shown = {
			scope:recipient = {
				OR = {
					is_child_of = scope:actor
					dynasty ?= scope:actor.dynasty
				}
				NOR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
				}
			}
		}
		is_valid = { always = yes }
		starts_enabled = {
			scope:recipient = {
				OR = {
					is_child_of = scope:actor
					dynasty ?= scope:actor.dynasty
				}
				NOR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
				}
				OR = {
					is_primary_heir_of = scope:actor
					is_ruler = yes
				}
			}
		}
	}

	send_option = {
		flag = bless_iron
		localization = bm_bless_bloodline_interaction.bless_iron
		current_description = bm_bless_bloodline_interaction.bless_iron_desc
		is_shown = {
			scope:recipient = {
				NOR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
				}
			}
		}
		is_valid = { always = yes }
		starts_enabled = {
			scope:recipient = {
				NOR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
				}
				OR = {
					NOR = {
						is_child_of = scope:actor
						dynasty ?= scope:actor.dynasty
					}
					AND = {
						NOR = {
							is_primary_heir_of = scope:actor
							is_ruler = yes
						}
						OR = {
							is_knight = yes
							martial >= 12
						}
					}
				}
			}
		}
	}

	send_option = {
		flag = bless_crucible
		localization = bm_bless_bloodline_interaction.bless_crucible
		current_description = bm_bless_bloodline_interaction.bless_crucible_desc
		is_shown = {
			scope:recipient = {
				OR = {
					is_child_of = scope:actor
					dynasty ?= scope:actor.dynasty
				}
				NOR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
				}
			}
		}
		is_valid = { always = yes }
		starts_enabled = {
			scope:recipient = {
				OR = {
					is_child_of = scope:actor
					dynasty ?= scope:actor.dynasty
				}
				NOR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
					is_primary_heir_of = scope:actor
					is_ruler = yes
					is_knight = yes
					martial >= 12
				}
			}
		}
	}

	send_option = {
		flag = bless_feingur
		localization = bm_bless_bloodline_interaction.bless_feingur
		current_description = bm_bless_bloodline_interaction.bless_feingur_desc
		is_shown = {
			scope:actor = { bm_has_superior_stave_trigger = { STAVE = feingur } }
			scope:recipient = {
				OR = {
					is_child_of = scope:actor
					dynasty ?= scope:actor.dynasty
				}
				NOT = { has_character_modifier = bm_feingur_blessed_blood_modifier }
			}
		}
		is_valid = { always = yes }
		starts_enabled = {
			scope:actor = { bm_has_superior_stave_trigger = { STAVE = feingur } }
			scope:recipient = {
				OR = {
					is_child_of = scope:actor
					dynasty ?= scope:actor.dynasty
				}
				NOT = { has_character_modifier = bm_feingur_blessed_blood_modifier }
				OR = {
					has_character_modifier = bm_blessed_blood_modifier
					has_character_modifier = bm_blessed_blood_sovereign_modifier
					has_character_modifier = bm_blessed_blood_iron_modifier
					has_character_modifier = bm_blessed_blood_crucible_modifier
					has_character_modifier = bm_anointed_champion_modifier
				}
			}
		}
	}

	on_accept = {
		debug_log = "BloodMageInteraction: bless_bloodline | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_bless_bloodline_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				bloodline_magic_cast_effect_minor = yes
				scope:recipient = {
					if = {
						limit = { scope:bless_feingur = yes }
						add_character_modifier = {
							modifier = bm_feingur_blessed_blood_modifier
						}
						add_opinion = {
							target = scope:actor
							modifier = magic_opinion_positive
							opinion = 40
						}
					}
					else_if = {
						limit = { scope:bless_sovereign = yes }
						add_character_modifier = {
							modifier = bm_blessed_blood_sovereign_modifier
							years = 10
						}
						add_opinion = {
							target = scope:actor
							modifier = magic_opinion_positive
							opinion = 25
						}
					}
					else_if = {
						limit = { scope:bless_iron = yes }
						add_character_modifier = {
							modifier = bm_blessed_blood_iron_modifier
							years = 10
						}
						add_opinion = {
							target = scope:actor
							modifier = magic_opinion_positive
							opinion = 25
						}
					}
					else = {
						add_character_modifier = {
							modifier = bm_blessed_blood_crucible_modifier
							years = 10
						}
						add_opinion = {
							target = scope:actor
							modifier = magic_opinion_positive
							opinion = 25
						}
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = children
		ai_recipients = dynasty
		ai_recipients = courtiers
		ai_recipients = vassals
	}
	ai_potential = { always = yes }
	ai_frequency = 12
	ai_will_do = {
		base = 10
		ai_willingness_to_do_positive_magic = yes

		modifier = {
			scope:recipient = { is_child_of = scope:actor }
			add = 50
		}
		modifier = {
			scope:recipient = { is_primary_heir_of = scope:actor }
			add = 50
		}
		modifier = {
			scope:recipient = { is_knight_of = scope:actor }
			add = 25
		}
		modifier = {
			scope:actor = {
				NOT = { has_character_modifier = lifeforce_modifier_minor }
			}
			factor = 0
		}
	}
}
