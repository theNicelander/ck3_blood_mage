grant_crimson_knight_interaction = {
    category = interaction_category_friendly
    auto_accept = yes
    common_interaction = yes
    desc = grant_crimson_knight_interaction
    interface_priority = 190

    cost = { piety = lifedrain_piety_cost_minor }

    is_shown = {
        scope:actor = { has_trait = lifestyle_blood_mage }
        scope:recipient = {
            OR = {
                is_courtier_of = scope:actor
                this = scope:actor
            }
            NOT = { has_trait = lifestyle_crimson_knight }
        }
    }
    is_valid_showing_failures_only = {
        scope:actor = { has_character_modifier = lifeforce_modifier_minor }
        trigger_if = {
            limit = { scope:recipient != scope:actor }
            scope:recipient = { is_knight_of = scope:actor }
        }
    }

    on_accept = {
        debug_log = "BloodMageInteraction: grant_crimson_knight | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        scope:actor = { benediction_magic_cast_effect_minor = yes }
        scope:recipient = {
            add_trait = lifestyle_crimson_knight
            if = {
                limit = { this != scope:actor }
                add_opinion = {
                    target = scope:actor
                    modifier = magic_opinion_positive
                    opinion = 20
                }
            }
        }
    }
    ai_potential = { always = yes }
    ai_targets = {
        ai_recipients = self
        ai_recipients = dynasty
        ai_recipients = spouses
        ai_recipients = family
        ai_recipients = children
        ai_recipients = courtiers
        ai_recipients = councillors
    }
    ai_frequency = 6
    ai_will_do = {
        base = 33
        ai_willingness_to_do_postitive_magic = yes
        modifier = {
            scope:recipient = { prowess >= 12 }
            add = 30
        }
    }
}
