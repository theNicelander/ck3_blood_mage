bm_cull_champion_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_cull_champion_interaction
	auto_accept = yes
	common_interaction = yes
	desc = bm_cull_champion_interaction_desc
	interface_priority = 170

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			is_alive = yes
			trigger_if = {
				limit = {
					NOT = { is_imprisoned_by = scope:actor }
					NOR = {
						has_relation_rival = scope:actor
						has_relation_nemesis = scope:actor
					}
				}
				NOR = {
					is_child_of = scope:actor
					is_primary_heir_of = scope:actor
					is_spouse_of = scope:actor
					is_consort_of = scope:actor
					has_relation_soulmate = scope:actor
				}
			}
			prowess >= 20
			OR = {
				# Eligible prisoner
				AND = {
					is_imprisoned_by = scope:actor
				}
				# Fallen or critically broken own follower
				AND = {
					OR = {
						is_courtier_of = scope:actor
						is_knight_of = scope:actor
					}
					OR = {
						has_trait = maimed
						has_trait = incapable
						has_trait = infirm
						has_trait = wounded_3
					}
				}
			}
		}
	}

	cost = {
		piety = 50
	}

	on_accept = {
		debug_log = "BloodMageInteraction: cull_champion | Actor: [actor.GetShortUINameNoTooltip] -> Recipient: [recipient.GetShortUINameNoTooltip]"
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_cull_champion_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				add_character_modifier = {
					modifier = lifeforce_modifier_major
				}
				if = {
					limit = { has_trait = lifestyle_blood_knight }
					add_blood_knight_xp = {
						TRACK = slaughter
						XP = 3
					}
				}
				else = {
					add_blood_mage_xp = {
						TRACK = hematurgy
						XP = 3
					}
				}
				random = {
					chance = bm_cull_champion_superior_chance_value
					add_character_modifier = {
						modifier = lifeforce_modifier_superior
					}
				}
				scope:recipient = {
					death = {
						death_reason = death_lifedrain_reason
						killer = scope:actor
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
			}
		}
	}

	ai_frequency = 12
	ai_potential = {
		OR = {
			has_trait = lifestyle_blood_mage
			has_trait = lifestyle_blood_knight
		}
	}
	ai_targets = {
		ai_recipients = prisoners
		ai_recipients = courtiers
		ai_recipients = vassals
	}
	ai_will_do = {
		base = 10
		ai_willingness_to_use_hostile_magic = yes
		modifier = {
			factor = 2
			ai_compassion < 0
		}
		modifier = {
			factor = 0
			scope:recipient = {
				OR = {
					is_child_of = scope:actor
					is_spouse_of = scope:actor
					is_close_family_of = scope:actor
					is_primary_heir_of = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_lover = scope:actor
					has_relation_soulmate = scope:actor
				}
			}
		}
	}
}
