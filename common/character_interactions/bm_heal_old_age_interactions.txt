bm_heal_old_age_trait = {
	interface_priority = 209
	category = interaction_category_bm_blood_magic
	desc = bm_heal_old_age_trait_desc
	common_interaction = yes

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		scope:recipient = { bm_has_old_age_ailment_trigger = yes }
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = IN_COURT_OR_IMPRISONED
			OR = {
				scope:actor = scope:recipient
				scope:recipient = {
					bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
				}
			}
		}
		custom_tooltip = {
			text = bm_heal_old_age_resource_requirement_tt
			OR = {
				AND = {
					scope:actor = {
						has_character_modifier = lifeforce_modifier_minor
						piety >= heal_old_age_cost_minor
					}
				}
				AND = {
					scope:actor = {
						has_character_modifier = lifeforce_modifier_major
						piety >= heal_old_age_cost_major
					}
				}
			}
		}
	}

	on_accept = {
		debug_log = "BloodMageInteraction: bm_heal_old_age_trait | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		scope:actor = {
			trigger_event = bm_heal_old_age.0001
		}
	}

	auto_accept = yes
	ai_accept = { base = 100 }
	ai_potential = { always = yes }
	ai_targets = {
		ai_recipients = self
		ai_recipients = dynasty
		ai_recipients = spouses
		ai_recipients = family
		ai_recipients = children
		ai_recipients = courtiers
		ai_recipients = councillors
		ai_recipients = liege
	}
	ai_frequency = 6
	ai_will_do = {
		base = -20
		modifier = {
			scope:actor = scope:recipient
			add = 100
		}
		modifier = {
			scope:recipient = { is_close_family_of = scope:actor }
			add = 50
		}
		ai_willingness_to_do_positive_magic = yes
	}
}
