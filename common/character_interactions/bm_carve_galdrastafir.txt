			always = yes
		}
		is_valid = {
			scope:actor = {

			}
		}
		starts_enabled = {

					type = bm_event_toast_effect_good
					title = bm_carve_galdrastafir_toast

						add_xp_bm_dynamic = { TRACK = wisdom XP = 5 }

				}
			}
		}

			limit = { scope:bm_carve_aegishjalmur = yes }

					bm_set_capstone_stave_effect = { STAVE = nabrokarstafur }
				}
			}
		}
		else_if = {
			limit = { scope:bm_carve_hulinhjalmur = yes }

		is_blood_mage_archmage_trigger = yes
		has_character_modifier = superior_blood_rune_modifier

		ai_recipients = self

		base = 10

		modifier = {
			scope:actor = { bm_has_any_stave_trigger = no }
			add = 100
		}
		modifier = {
			scope:actor = { health <= fine_health }
			add = -100
		}
		modifier = {
			scope:bm_carve_vegvisir = yes
			scope:actor = { has_trait = lifestyle_traveler }
			add = 20
		}
		modifier = {
			scope:bm_carve_aegishjalmur = yes
			scope:actor = {
				bm_has_track_xp_any_tier_trigger = { TRACK = fury VALUE = 25 }
			}
			add = 25
		}
		modifier = {
			scope:bm_carve_feingur = yes
			scope:actor = { NOT = { any_child = { count >= 2 } } }
			add = 25
		}
		modifier = {
			scope:bm_carve_nabrokarstafur = yes
			scope:actor = { gold < 100 }
			add = 20
		}
		modifier = {
			scope:bm_carve_hulinhjalmur = yes
			scope:actor = { has_lifestyle = intrigue_lifestyle }
			add = 20
		}

		# Do not swap once a stave is carved
		modifier = {
			scope:actor = { bm_has_any_stave_trigger = yes }
			factor = 0
		}

		# Absolute veto: must select at least one valid send option
		modifier = {
			NOR = {
				scope:bm_carve_vegvisir = yes
				scope:bm_carve_aegishjalmur = yes
				scope:bm_carve_feingur = yes
				scope:bm_carve_nabrokarstafur = yes
				scope:bm_carve_hulinhjalmur = yes
			}
