make_blood_knight_interaction = {
    category = interaction_category_friendly
    auto_accept = yes
    common_interaction = yes
    desc = make_blood_knight_interaction_desc
    interface_priority = 190

    cost = {
        piety = {
            if = {
                limit = { scope:mage_initiation = yes }
                value = blood_knight_creation_piety_cost
            }
        }
        prestige = {
            if = {
                limit = { scope:knight_minor_initiation = yes }
                value = blood_knight_creation_prestige_cost
            }
        }
    }

    is_shown = {
        scope:actor = {
            OR = {
                has_trait = lifestyle_blood_mage
                has_trait = lifestyle_blood_knight
            }
        }
        scope:recipient = {
            is_courtier_of = scope:actor
            NOT = { this = scope:actor }
            NOT = { has_trait = lifestyle_blood_mage }
            NOT = { has_trait = lifestyle_blood_knight }
        }
    }
    is_valid_showing_failures_only = {
        custom_description = {
            text = recipient_must_be_knight_tt
            scope:recipient = { is_knight_of = scope:actor }
        }
        trigger_if = {
            limit = {
                scope:actor = {
                    has_trait = lifestyle_blood_mage
                    NOT = { has_trait = lifestyle_blood_knight }
                }
            }
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
            }
        }
        trigger_else = {
            custom_description = {
                text = make_blood_knight_interaction_knight_requirement
                scope:actor = {
                    OR = {
                        AND = {
                            has_character_modifier = lifeforce_modifier_minor
                            prestige >= blood_knight_creation_prestige_cost
                        }
                        has_character_modifier = lifeforce_modifier_major
                    }
                }
            }
        }
    }

    send_options_exclusive = yes

    send_option = {
        flag = mage_initiation
        localization = make_blood_knight_interaction.mage
        current_description = make_blood_knight_interaction.mage_desc
        is_shown = {
            scope:actor = { has_trait = lifestyle_blood_mage }
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
            }
        }
        starts_enabled = {
            scope:actor = {
                has_trait = lifestyle_blood_mage
                OR = {
                    NOT = { has_trait = lifestyle_blood_knight }
                    AND = {
                        NOT = { has_character_modifier = lifeforce_modifier_minor }
                        NOT = { has_character_modifier = lifeforce_modifier_major }
                    }
                }
            }
        }
    }

    send_option = {
        flag = knight_minor_initiation
        localization = make_blood_knight_interaction.knight_minor
        current_description = make_blood_knight_interaction.knight_minor_desc
        is_shown = {
            scope:actor = { has_trait = lifestyle_blood_knight }
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                prestige >= blood_knight_creation_prestige_cost
            }
        }
        starts_enabled = {
            scope:actor = {
                has_trait = lifestyle_blood_knight
                has_character_modifier = lifeforce_modifier_minor
                prestige >= blood_knight_creation_prestige_cost
            }
        }
    }

    send_option = {
        flag = knight_major_initiation
        localization = make_blood_knight_interaction.knight_major
        current_description = make_blood_knight_interaction.knight_major_desc
        is_shown = {
            scope:actor = { has_trait = lifestyle_blood_knight }
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
            }
        }
        starts_enabled = {
            scope:actor = {
                has_trait = lifestyle_blood_knight
                has_character_modifier = lifeforce_modifier_major
                OR = {
                    NOT = { has_character_modifier = lifeforce_modifier_minor }
                    prestige < blood_knight_creation_prestige_cost
                }
            }
        }
    }

    on_accept = {
        debug_log = "BloodMageInteraction: make_blood_knight | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        if = {
            limit = { scope:knight_minor_initiation = yes }
            scope:actor = {
                remove_character_modifier = lifeforce_modifier_minor
                add_blood_knight_xp = {
                    TRACK = vanguard
                    XP = 2
                }
            }
        }
        else_if = {
            limit = { scope:knight_major_initiation = yes }
            scope:actor = {
                remove_character_modifier = lifeforce_modifier_major
                add_blood_knight_xp = {
                    TRACK = vanguard
                    XP = 4
                }
            }
        }
        else = {
            scope:actor = { benediction_magic_cast_effect_major = yes }
        }
        scope:recipient = {
            bm_become_blood_knight_effect = yes
            add_opinion = {
                target = scope:actor
                modifier = magic_opinion_positive
                opinion = 20
            }
        }
    }
    ai_potential = { always = yes }
    ai_targets = {
        ai_recipients = dynasty
        ai_recipients = spouses
        ai_recipients = family
        ai_recipients = children
        ai_recipients = courtiers
        ai_recipients = councillors
    }
    ai_frequency = 6
    ai_will_do = {
        base = 33
        ai_willingness_to_do_positive_magic = yes
        bm_blood_knight_prevalence_ai_acquisition_modifier = yes
        modifier = {
            scope:recipient = { prowess >= 12 }
            add = 30
        }
    }
}

empower_blood_knight_interaction = {
    category = interaction_category_friendly
    auto_accept = yes
    desc = empower_blood_knight_interaction_desc
    interface_priority = 189

    is_shown = {
        scope:actor = { has_trait = lifestyle_blood_knight }
        scope:recipient = {
            is_courtier_of = scope:actor
            NOT = { this = scope:actor }
            has_trait = lifestyle_blood_knight
        }
    }
    is_valid_showing_failures_only = {
        custom_description = {
            text = "empower_blood_knight_interaction_requirement"
            scope:actor = {
                OR = {
                    has_character_modifier = lifeforce_modifier_major
                    has_character_modifier = lifeforce_modifier_minor
                }
            }
        }
    }

    send_option = {
        flag = empower_major
        localization = empower_blood_knight_interaction.major
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
            }
        }
    }

    send_option = {
        flag = empower_minor
        localization = empower_blood_knight_interaction.minor
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
            }
        }
        starts_enabled = {
            scope:actor = {
                NOT = { has_character_modifier = lifeforce_modifier_major }
                has_character_modifier = lifeforce_modifier_minor
            }
        }
    }

    send_option = {
        flag = empower_cancel
        localization = empower_blood_knight_interaction.cancel
    }

    on_accept = {
        debug_log = "BloodKnightInteraction: empower_blood_knight | Actor: [actor.GetShortUINameNoTooltip] -> Recipient: [recipient.GetShortUINameNoTooltip]"
        if = {
            limit = { scope:empower_major = yes }
            scope:actor = {
                remove_character_modifier = lifeforce_modifier_major
                add_blood_knight_xp = { TRACK = benediction XP = 4 }
            }
            scope:recipient = {
                add_blood_knight_xp = { TRACK = vanguard XP = 10 }
                add_blood_knight_xp = { TRACK = slaughter XP = 10 }
                add_opinion = {
                    target = scope:actor
                    modifier = magic_opinion_positive
                    opinion = 30
                }
            }
        }
        else_if = {
            limit = { scope:empower_minor = yes }
            scope:actor = {
                remove_character_modifier = lifeforce_modifier_minor
                add_blood_knight_xp = { TRACK = benediction XP = 2 }
            }
            scope:recipient = {
                add_blood_knight_xp = { TRACK = vanguard XP = 5 }
                add_blood_knight_xp = { TRACK = slaughter XP = 5 }
                add_opinion = {
                    target = scope:actor
                    modifier = magic_opinion_positive
                    opinion = 15
                }
            }
        }
    }
    ai_potential = { always = yes }
    ai_frequency = 6
    ai_targets = {
        ai_recipients = dynasty
        ai_recipients = spouses
        ai_recipients = family
        ai_recipients = children
        ai_recipients = courtiers
        ai_recipients = councillors
    }
    ai_will_do = {
        base = 15
        ai_willingness_to_do_positive_magic = yes
        modifier = {
            scope:recipient = { prowess >= 14 }
            add = 25
        }
    }
}
