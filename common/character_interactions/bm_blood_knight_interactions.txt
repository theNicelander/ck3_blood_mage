make_blood_knight_interaction = {
    category = interaction_category_bm_blood_magic
    auto_accept = yes
    common_interaction = yes
    desc = make_blood_knight_interaction_desc
    interface_priority = 190

    cost = {
        piety = {
            if = {
                limit = { scope:knight_major_initiation = yes }
                value = bm_initiation_major_piety_cost
            }
            if = {
                limit = { scope:knight_minor_initiation = yes }
                value = bm_initiation_minor_piety_cost
            }
        }
    }

    is_shown = {
        scope:actor = {
            OR = {
                has_trait = lifestyle_blood_knight
                has_trait = lifestyle_blood_mage
            }
        }
        scope:recipient = {
            NOT = { this = scope:actor }
            NOT = { has_trait = lifestyle_blood_mage }
            NOT = { has_trait = lifestyle_blood_knight }
            bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
        }
    }
    is_valid_showing_failures_only = {
        custom_description = {
            text = make_blood_knight_interaction_knight_requirement
            scope:actor = {
                OR = {
                    AND = {
                        has_character_modifier = lifeforce_modifier_minor
                        piety >= bm_initiation_minor_piety_cost
                    }
                    AND = {
                        has_character_modifier = lifeforce_modifier_major
                        piety >= bm_initiation_major_piety_cost
                    }
                }
            }
        }
    }

    send_options_exclusive = yes

    send_option = {
        flag = knight_minor_initiation
        localization = make_blood_knight_interaction.knight_minor
        current_description = make_blood_knight_interaction.knight_minor_desc
        can_invalidate_interaction = yes
        is_shown = {
            always = yes
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= bm_initiation_minor_piety_cost
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= bm_initiation_minor_piety_cost
                OR = {
                    NOT = { has_character_modifier = lifeforce_modifier_major }
                    piety < bm_initiation_major_piety_cost
                }
            }
        }
    }

    send_option = {
        flag = knight_major_initiation
        localization = make_blood_knight_interaction.knight_major
        current_description = make_blood_knight_interaction.knight_major_desc
        can_invalidate_interaction = yes
        is_shown = {
            always = yes
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= bm_initiation_major_piety_cost
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= bm_initiation_major_piety_cost
            }
        }
    }

    on_accept = {
        if = {
            limit = { scope:knight_minor_initiation = yes }
            debug_log = "BloodMageInteraction: make_blood_knight (minor) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = make_blood_knight_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    remove_character_modifier = lifeforce_modifier_minor
                    if = {
                        limit = { has_trait = lifestyle_blood_knight }
                        add_blood_knight_xp = {
                            TRACK = vanguard
                            XP = 1
                        }
                    }
                    else_if = {
                        limit = { has_trait = lifestyle_blood_mage }
                        add_xp_bm_dynamic = {
                            TRACK = benediction
                            XP = 1
                        }
                    }
                    scope:recipient = {
                        bm_become_blood_knight_effect = yes
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 20
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                    }
                }
            }
        }
        else_if = {
            limit = { scope:knight_major_initiation = yes }
            debug_log = "BloodMageInteraction: make_blood_knight (major) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = make_blood_knight_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    remove_character_modifier = lifeforce_modifier_major
                    if = {
                        limit = { has_trait = lifestyle_blood_knight }
                        if = {
                            limit = {
                                scope:recipient = {
                                    OR = {
                                        is_child_of = scope:actor
                                        dynasty ?= scope:actor.dynasty
                                    }
                                }
                            }
                            add_blood_knight_xp = {
                                TRACK = vanguard
                                XP = 2
                            }
                        }
                        else = {
                            add_blood_knight_xp = {
                                TRACK = vanguard
                                XP = 1
                            }
                        }
                    }
                    else_if = {
                        limit = { has_trait = lifestyle_blood_mage }
                        if = {
                            limit = {
                                scope:recipient = {
                                    OR = {
                                        is_child_of = scope:actor
                                        dynasty ?= scope:actor.dynasty
                                    }
                                }
                            }
                            add_xp_bm_dynamic = {
                                TRACK = benediction
                                XP = 2
                            }
                        }
                        else = {
                            add_xp_bm_dynamic = {
                                TRACK = benediction
                                XP = 1
                            }
                        }
                    }
                    scope:recipient = {
                        bm_become_blood_knight_effect = yes
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 20
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                    }
                }
            }
        }
        else = {
            debug_log = "BloodMageInteraction: make_blood_knight (NO OPTION) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        }
    }
    ai_potential = { always = yes }
    ai_targets = {
        ai_recipients = spouses
        ai_recipients = children
        ai_recipients = councillors
        ai_recipients = family
        ai_recipients = dynasty
        ai_recipients = courtiers
        ai_recipients = vassals
    }
    ai_frequency = 3
    ai_will_do = {
        base = 0

        # Score send options
        modifier = {
            scope:knight_major_initiation = yes
            add = 50
        }
        modifier = {
            scope:knight_minor_initiation = yes
            add = 25
        }

        # If actor is Blood Mage: save spouse/consort/primary heir for Blood Mage initiation
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_mage }
            scope:recipient = {
                OR = {
                    is_spouse_of = scope:actor
                    is_consort_of = scope:actor
                    is_primary_heir_of = scope:actor
                }
            }
            factor = 0
        }
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_mage }
            scope:recipient = { is_knight_of = scope:actor }
            add = 100
        }
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_mage }
            scope:recipient = { prowess >= 12 }
            add = 50
        }

        # If actor is Blood Knight: standard hierarchy (spouse > consort > heir > child > knight > prowess)
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_knight }
            scope:recipient = { is_spouse_of = scope:actor }
            add = 1000
        }
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_knight }
            scope:recipient = { is_consort_of = scope:actor }
            add = 500
        }
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_knight }
            scope:recipient = { is_primary_heir_of = scope:actor }
            add = 800
        }
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_knight }
            scope:recipient = { is_child_of = scope:actor }
            add = 100
        }
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_knight }
            scope:recipient = { is_knight_of = scope:actor }
            add = 50
        }
        modifier = {
            scope:actor = { has_trait = lifestyle_blood_knight }
            scope:recipient = { prowess >= 12 }
            add = 30
        }

        # Shared priorities
        modifier = {
            scope:recipient = { is_councillor_of = scope:actor }
            add = 25
        }
        modifier = {
            scope:recipient = {
                exists = dynasty
                exists = scope:actor.dynasty
                dynasty = scope:actor.dynasty
            }
            add = 25
        }

        bm_blood_knight_prevalence_ai_acquisition_modifier = yes
        ai_willingness_to_do_positive_magic = yes

        # Absolute veto: must select at least one valid send option
        modifier = {
            NOR = {
                scope:knight_minor_initiation = yes
                scope:knight_major_initiation = yes
            }
            factor = 0
        }
    }
}
