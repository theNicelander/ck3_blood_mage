make_blood_knight_interaction = {
    category = interaction_category_bm_blood_magic
    auto_accept = yes
    common_interaction = yes
    desc = make_blood_knight_interaction_desc
    interface_priority = 190

    cost = {
        piety = {
            if = {
                limit = { scope:knight_major_initiation = yes }
                value = bm_initiation_major_piety_cost
            }
            if = {
                limit = { scope:knight_minor_initiation = yes }
                value = bm_initiation_minor_piety_cost
            }
        }
    }

    is_shown = {
        scope:actor = {
            has_trait = lifestyle_blood_knight
        }
        scope:recipient = {
            NOT = { this = scope:actor }
            NOT = { has_trait = lifestyle_blood_mage }
            NOT = { has_trait = lifestyle_blood_knight }
            bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
        }
    }
    is_valid_showing_failures_only = {
        custom_description = {
            text = make_blood_knight_interaction_knight_requirement
            scope:actor = {
                OR = {
                    AND = {
                        has_character_modifier = lifeforce_modifier_minor
                        piety >= bm_initiation_minor_piety_cost
                    }
                    AND = {
                        has_character_modifier = lifeforce_modifier_major
                        piety >= bm_initiation_major_piety_cost
                    }
                }
            }
        }
    }

    send_options_exclusive = yes

    send_option = {
        flag = knight_minor_initiation
        localization = make_blood_knight_interaction.knight_minor
        current_description = make_blood_knight_interaction.knight_minor_desc
        is_shown = {
            always = yes
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= bm_initiation_minor_piety_cost
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= bm_initiation_minor_piety_cost
                OR = {
                    NOT = { has_character_modifier = lifeforce_modifier_major }
                    piety < bm_initiation_major_piety_cost
                }
            }
        }
    }

    send_option = {
        flag = knight_major_initiation
        localization = make_blood_knight_interaction.knight_major
        current_description = make_blood_knight_interaction.knight_major_desc
        is_shown = {
            always = yes
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= bm_initiation_major_piety_cost
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= bm_initiation_major_piety_cost
            }
        }
    }

    on_accept = {
        debug_log = "BloodMageInteraction: make_blood_knight | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        if = {
            limit = { scope:knight_minor_initiation = yes }
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = make_blood_knight_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    remove_character_modifier = lifeforce_modifier_minor
                    add_blood_knight_xp = {
                        TRACK = vanguard
                        XP = 1
                    }
                    scope:recipient = {
                        bm_become_blood_knight_effect = yes
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 20
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                    }
                }
            }
        }
        else_if = {
            limit = { scope:knight_major_initiation = yes }
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = make_blood_knight_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    remove_character_modifier = lifeforce_modifier_major
                    if = {
                        limit = {
                            scope:recipient = {
                                OR = {
                                    is_child_of = scope:actor
                                    dynasty ?= scope:actor.dynasty
                                }
                            }
                        }
                        add_blood_knight_xp = {
                            TRACK = vanguard
                            XP = 2
                        }
                    }
                    else = {
                        add_blood_knight_xp = {
                            TRACK = vanguard
                            XP = 1
                        }
                    }
                    scope:recipient = {
                        bm_become_blood_knight_effect = yes
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 20
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                    }
                }
            }
        }
    }
    ai_potential = { always = yes }
    ai_targets = {
        ai_recipients = spouses
        ai_recipients = children
        ai_recipients = councillors
        ai_recipients = family
        ai_recipients = dynasty
        ai_recipients = courtiers
        ai_recipients = vassals
    }
    ai_frequency = 3
    ai_will_do = {
        base = 0
        ai_willingness_to_do_positive_magic = yes

        # Prioritisation hierarchy: spouse > consort > primary heir > children > knights > prowess > councillors > dynasty
        modifier = {
            scope:recipient = { is_spouse_of = scope:actor }
            add = 1000
        }
        modifier = {
            scope:recipient = { is_consort_of = scope:actor }
            add = 500
        }
        modifier = {
            scope:recipient = { is_primary_heir_of = scope:actor }
            add = 800
        }
        modifier = {
            scope:recipient = { is_child_of = scope:actor }
            add = 100
        }
        modifier = {
            scope:recipient = { is_knight_of = scope:actor }
            add = 50
        }
        modifier = {
            scope:recipient = { prowess >= 12 }
            add = 30
        }
        modifier = {
            scope:recipient = { is_councillor_of = scope:actor }
            add = 25
        }
        modifier = {
            scope:recipient = {
                exists = dynasty
                exists = scope:actor.dynasty
                dynasty = scope:actor.dynasty
            }
            add = 25
        }

        bm_blood_knight_prevalence_ai_acquisition_modifier = yes
    }
}

empower_blood_knight_interaction = {
    category = interaction_category_bm_blood_magic
    icon = empower_blood_knight_interaction
    auto_accept = yes
    desc = empower_blood_knight_interaction_desc
    interface_priority = 189

    is_shown = {
        scope:actor = { has_trait = lifestyle_blood_knight }
        scope:recipient = {
            NOT = { this = scope:actor }
            has_trait = lifestyle_blood_knight
            bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
        }
    }
    is_valid_showing_failures_only = {
        custom_description = {
            text = "empower_blood_knight_interaction_requirement"
            scope:actor = {
                OR = {
                    has_character_modifier = lifeforce_modifier_major
                    has_character_modifier = lifeforce_modifier_minor
                }
            }
        }
    }

    send_options_exclusive = yes

    send_option = {
        flag = empower_major
        localization = empower_blood_knight_interaction.major
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
            }
        }
    }

    send_option = {
        flag = empower_minor
        localization = empower_blood_knight_interaction.minor
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
            }
        }
        starts_enabled = {
            scope:actor = {
                NOT = { has_character_modifier = lifeforce_modifier_major }
                has_character_modifier = lifeforce_modifier_minor
            }
        }
    }

    on_accept = {
        debug_log = "BloodKnightInteraction: empower_blood_knight | Actor: [actor.GetShortUINameNoTooltip] -> Recipient: [recipient.GetShortUINameNoTooltip]"
        if = {
            limit = { scope:empower_major = yes }
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = empower_blood_knight_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    blood_magic_used_major = yes
                    add_blood_knight_xp = { TRACK = benediction XP = 2 }
                    scope:recipient = {
                        add_blood_knight_xp = { TRACK = vanguard XP = 3 }
                        add_blood_knight_xp = { TRACK = slaughter XP = 3 }
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 30
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                    }
                }
            }
        }
        else_if = {
            limit = { scope:empower_minor = yes }
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = empower_blood_knight_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    blood_magic_used_minor = yes
                    add_blood_knight_xp = { TRACK = benediction XP = 1 }
                    scope:recipient = {
                        add_blood_knight_xp = { TRACK = vanguard XP = 1 }
                        add_blood_knight_xp = { TRACK = slaughter XP = 1 }
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 15
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                    }
                }
            }
        }
    }
    ai_potential = { always = yes }
    ai_frequency = 6
    ai_targets = {
        ai_recipients = dynasty
        ai_recipients = spouses
        ai_recipients = family
        ai_recipients = children
        ai_recipients = courtiers
        ai_recipients = councillors
    }
    ai_will_do = {
        base = 15
        ai_willingness_to_do_positive_magic = yes
        modifier = {
            scope:recipient = { prowess >= 14 }
            add = 25
        }
    }
}
