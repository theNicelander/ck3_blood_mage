	interface_priority = 888

	cost = { piety = 75 }

	is_shown = {
		scope:actor = { has_trait = lifestyle_blood_mage }

			remove_character_modifier = lifeforce_modifier_minor

			custom_tooltip = channel_lifeforce_enlightenment_minor.effects_list
		}
	}

	ai_targets = { ai_recipients = self }
	ai_potential = { always = yes }
	ai_frequency = 6
	ai_will_do = {
		base = 10
		modifier = {
			add = -100
			scope:actor = { health <= fine_health }
		}
	}
}

bm_cast_blood_magic_self_manifest_lifeforce = {
	category = interaction_category_friendly
	auto_accept = yes
	common_interaction = yes
	interface_priority = 888
	send_name = bm_cast_blood_magic_self_manifest_lifeforce
	desc = bm_cast_blood_magic_self_manifest_lifeforce_desc

	cost = { piety = 100 }
	cooldown = { years = 1 }

	is_shown = {
		scope:actor = { has_trait = lifestyle_blood_mage }
		scope:actor = scope:recipient
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_description = {
				text = bm_cooldown_active_tt
				NOT = { has_character_flag = bm_manifest_lifeforce_cooldown }
			}
			piety_level >= 2
		}
	}

	on_accept = {
		debug_log = debug_log.blood_mage_log
		scope:actor = {
			add_character_flag = { flag = bm_manifest_lifeforce_cooldown years = 1 }
			add_xp_bm_dynamic = {
				TRACK = enlightenment
				XP = 5
			}
			duel = {
				skill = learning
				value = 0

				# Success - major lifeforce
				15 = {
					desc = manifest_lifeforce_major_success_tt
					compare_modifier = {
						value = scope:duel_value
						multiplier = 1
					}
					send_interface_toast = {
						title = manifest_lifeforce_major_success_tt
						left_icon = root
						add_character_modifier = { modifier = lifeforce_modifier_major }
					}
				}

				# Partial success - minor lifeforce
				40 = {
					desc = manifest_lifeforce_minor_success_tt
					compare_modifier = {
						value = scope:duel_value
						multiplier = 2
					}
					send_interface_toast = {
						title = manifest_lifeforce_minor_success_tt
						left_icon = root
						add_character_modifier = { modifier = lifeforce_modifier_minor }
					}
				}

				# Failure - nothing happens
				40 = {
					desc = manifest_lifeforce_minor_failure_tt
					send_interface_toast = {
						title = manifest_lifeforce_minor_failure_tt
						left_icon = root
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}
				}

				# Critical failure
				15 = {
					desc = manifest_lifeforce_major_failure_tt
					send_interface_toast = {
						title = manifest_lifeforce_major_failure_tt
						left_icon = root
						temporary_negative_lifeforce_effect_minor = yes
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.25
					}
				}
			}
		}
	}

	ai_targets = { ai_recipients = self }
	ai_potential = { always = yes }
	ai_frequency = 12
	ai_will_do = {
		base = 10
		modifier = {
			scope:actor = { piety > 2000 }
			add = 1000
		}
		modifier = {
			scope:actor = { NOT = { has_character_modifier = lifeforce_modifier_major } }
			add = 50
		}
	}
}
