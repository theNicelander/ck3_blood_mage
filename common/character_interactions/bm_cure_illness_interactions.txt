bm_heal_ailment = {
	interface_priority = 210
	category = interaction_category_friendly
	desc = bm_heal_ailment_desc
	common_interaction = yes

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		OR = {
			AND = {
				scope:actor = { has_trait = lifestyle_blood_mage }
				scope:recipient = { bm_has_any_ailment_trigger = yes }
			}
			AND = {
				scope:actor = {
					has_trait = lifestyle_blood_knight
					NOT = { has_trait = lifestyle_blood_mage }
				}
				scope:recipient = { bm_has_minor_ailment_trigger = yes }
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = IN_COURT_OR_IMPRISONED
			OR = { # self, courtiers, liege, family, spouse, knights
				scope:actor = scope:recipient
				scope:recipient = { is_courtier_of = scope:actor }
				trigger_if = {
					limit = { exists = scope:actor.liege }
					scope:recipient = scope:actor.liege
					scope:recipient = { is_courtier_of = scope:actor.liege }
				}
				scope:recipient = { is_close_family_of = scope:actor }
				scope:recipient = { is_spouse_of = scope:actor }
				scope:recipient = { is_knight_of = scope:actor }
			}
		}
		custom_tooltip = {
			text = bm_heal_ailment_resource_requirement_tt
			OR = {
				AND = {
					scope:recipient = { bm_has_minor_ailment_trigger = yes }
					scope:actor = {
						has_character_modifier = lifeforce_modifier_minor
						piety >= 15
					}
				}
				AND = {
					scope:actor = { has_trait = lifestyle_blood_mage }
					scope:recipient = { bm_has_major_ailment_trigger = yes }
					scope:actor = {
						has_character_modifier = lifeforce_modifier_major
						piety >= 35
					}
				}
				AND = {
					scope:actor = { has_trait = lifestyle_blood_mage }
					scope:recipient = { bm_has_deadly_ailment_trigger = yes }
					scope:actor = {
						has_character_modifier = lifeforce_modifier_minor
						has_character_modifier = lifeforce_modifier_major
						piety >= 100
					}
				}
			}
		}
	}

	on_accept = {
		debug_log = "BloodMageInteraction: bm_heal_ailment | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		scope:actor = {
			trigger_event = bm_heal_ailment.0001
		}
	}

	auto_accept = yes
	ai_accept = { base = 100 }
	ai_potential = { always = yes }
	ai_targets = {
		ai_recipients = self
		ai_recipients = dynasty
		ai_recipients = spouses
		ai_recipients = family
		ai_recipients = children
		ai_recipients = courtiers
		ai_recipients = councillors
	}
	ai_frequency = 6
	ai_will_do = {
		base = -20
		modifier = { # want to cure myself
			scope:actor = scope:recipient
			add = 100
		}
		modifier = {
			scope:recipient = { is_close_family_of = scope:actor }
			add = 50
		}
		ai_willingness_to_do_positive_magic = yes
	}
}
