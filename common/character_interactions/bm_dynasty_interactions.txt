bm_anoint_heir_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_anoint_heir_interaction
	auto_accept = yes
	common_interaction = yes
	desc = bm_anoint_heir_interaction_desc
	interface_priority = 196

	is_highlighted = {
		scope:recipient = { is_primary_heir_of = scope:actor }
	}

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			is_alive = yes
			NOT = { has_character_modifier = bm_anointed_heir_modifier }
			OR = {
				is_primary_heir_of = scope:actor
				is_child_of = scope:actor
			}
		}
	}

	cost = { piety = 250 }

	is_valid_showing_failures_only = {
		scope:actor = { has_character_modifier = lifeforce_modifier_major }
	}

	on_accept = {
		debug_log = "BloodMageInteraction: anoint_heir | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		scope:actor = {
			if = {
				limit = {
					exists = var:bm_anointed_heir
					var:bm_anointed_heir = {
						is_alive = yes
						has_character_modifier = bm_anointed_heir_modifier
						NOT = { this = scope:recipient }
					}
				}
				var:bm_anointed_heir = {
					remove_character_modifier = bm_anointed_heir_modifier
				}
			}
			every_child = {
				limit = {
					has_character_modifier = bm_anointed_heir_modifier
					NOT = { this = scope:recipient }
				}
				remove_character_modifier = bm_anointed_heir_modifier
			}
			if = {
				limit = {
					exists = primary_heir
					primary_heir = {
						has_character_modifier = bm_anointed_heir_modifier
						NOT = { this = scope:recipient }
					}
				}
				primary_heir = {
					remove_character_modifier = bm_anointed_heir_modifier
				}
			}
			set_variable = {
				name = bm_anointed_heir
				value = scope:recipient
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_anoint_heir_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				blood_magic_used_major = yes
				if = {
					limit = { has_trait = lifestyle_blood_mage }
					add_trait_xp = {
						trait = lifestyle_blood_mage
						track = bloodline
						value = 2
					}
				}
				if = {
					limit = {
						has_trait = lifestyle_blood_knight
						NOT = { has_trait = lifestyle_blood_mage }
					}
					add_blood_knight_xp = {
						TRACK = benediction
						XP = 2
					}
				}
				scope:recipient = {
					add_character_modifier = {
						modifier = bm_anointed_heir_modifier
					}
					add_opinion = {
						target = scope:actor
						modifier = magic_opinion_positive
						opinion = 30
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = children
		ai_recipients = family
	}
	ai_potential = { always = yes }
	ai_frequency = 24
	ai_will_do = {
		base = 20
		ai_willingness_to_do_positive_magic = yes

		modifier = {
			scope:recipient = { is_primary_heir_of = scope:actor }
			add = 100
		}
		modifier = {
			scope:actor = {
				NOT = { has_character_modifier = lifeforce_modifier_major }
			}
			factor = 0
		}
	}
}

bm_demand_blood_tithe_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_demand_blood_tithe_interaction
	auto_accept = yes
	common_interaction = yes
	desc = bm_demand_blood_tithe_interaction_desc
	interface_priority = 197
	cooldown = { years = 2 }

	is_shown = {
		scope:actor = {
			exists = house
			house.house_head = scope:actor
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			is_alive = yes
			is_adult = yes
			house ?= scope:actor.house
		}
	}

	cost = { piety = 50 }

	is_valid_showing_failures_only = {
		custom_description = {
			text = "bm_recipient_recently_lifedrained"
			NOT = { scope:recipient = { has_character_modifier = recently_lifedrained_modifier } }
		}
	}

	on_accept = {
		debug_log = "BloodMageInteraction: demand_blood_tithe | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_demand_blood_tithe_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				if = {
					limit = {
						scope:recipient = {
							OR = {
								has_trait = lifestyle_blood_mage
								has_trait = lifestyle_blood_knight
							}
						}
					}
					add_character_modifier = {
						modifier = lifeforce_modifier_major
					}
				}
				else = {
					add_character_modifier = {
						modifier = lifeforce_modifier_minor
					}
				}
				if = {
					limit = { has_trait = lifestyle_blood_mage }
					add_trait_xp = {
						trait = lifestyle_blood_mage
						track = bloodline
						value = 1
					}
				}
				if = {
					limit = {
						has_trait = lifestyle_blood_knight
						NOT = { has_trait = lifestyle_blood_mage }
					}
					add_blood_knight_xp = {
						TRACK = benediction
						XP = 1
					}
				}
				scope:recipient = {
					add_character_modifier = {
						modifier = recently_lifedrained_modifier
						years = 1
					}
					add_character_modifier = {
						modifier = lifeforce_modifier_negative_minor
						years = 1
					}
					if = {
						limit = {
							NOT = { bm_is_blood_cult_faith_trigger = yes }
							NOT = { has_trait = loyal }
						}
						add_opinion = {
							target = scope:actor
							modifier = magic_opinion_negative
							opinion = -25
						}
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = dynasty
		ai_recipients = family
		ai_recipients = courtiers
	}
	ai_potential = { always = yes }
	ai_frequency = 12
	ai_will_do = {
		base = 15
		ai_willingness_to_use_hostile_magic = yes
		modifier = {
			scope:actor = {
				NOT = { has_character_modifier = lifeforce_modifier_minor }
				NOT = { has_character_modifier = lifeforce_modifier_major }
			}
			add = 35
		}
		modifier = {
			scope:recipient = {
				has_opinion_modifier = {
					target = scope:actor
					modifier = magic_opinion_negative
				}
			}
			factor = 0.5
		}
		modifier = {
			factor = 0
			scope:recipient = {
				OR = {
					is_primary_heir_of = scope:actor
					is_spouse_of = scope:actor
					has_relation_lover = scope:actor
					has_relation_soulmate = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
				}
			}
		}
	}
}
