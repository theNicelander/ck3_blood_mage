lifedrain_prisoner_event_interaction = {
    common_interaction = yes
    interface_priority = 300
    category = interaction_category_bm_blood_magic
    desc = lifedrain_prisoner_interaction_desc
    is_highlighted = { always = yes }
    is_shown = {
        scope:actor = { has_trait = lifestyle_blood_mage }
        scope:recipient = { is_imprisoned_by = scope:actor }
    }

    cost = { piety = lifedrain_piety_cost_major }
    is_valid_showing_failures_only = {
        scope:recipient = { NOT = { has_character_modifier = lifedrained_modifier } }
        trigger_if = {
            limit = { scope:actor = { is_ai = yes } }
            NOT = {
                scope:actor = {
                    OR = {
                        is_spouse_of = scope:recipient
                        is_consort_of = scope:recipient
                    }
                }
            }
            NOT = { scope:recipient = { is_child_of = scope:actor } }
        }
    }

    on_accept = {
        debug_log = "BloodMageInteraction: lifedrain_prisoner | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        scope:actor = { hematurgy_magic_cast_effect_major = yes }
        lifedrain_execution_effect = {
            VICTIM = scope:recipient
            EXECUTIONER = scope:actor
        }
    }

    # AI
    auto_accept = yes
    ai_targets = { ai_recipients = prisoners }
    ai_potential = { always = yes }
    ai_frequency = 3
    ai_will_do = {
        base = 50
        ai_willingness_to_lifedrain = yes
    }
}

lifedrain_courtier_event_interaction = {
    common_interaction = yes
    interface_priority = 300
    category = interaction_category_bm_blood_magic
    desc = lifedrain_prisoner_interaction_desc
    is_shown = {
        scope:actor = {
            OR = {
                has_trait = lifestyle_blood_mage
                has_trait = lifestyle_blood_knight
            }
        }
        scope:recipient = {
            OR = {
                is_imprisoned_by = scope:actor
                AND = {
                    is_courtier_of = scope:actor
                    trigger_if = {
                        limit = {
                            NOR = {
                                has_relation_rival = scope:actor
                                has_relation_nemesis = scope:actor
                            }
                        }
                        NOR = {
                            is_close_family_of = scope:actor
                            is_child_of = scope:actor
                            is_primary_heir_of = scope:actor
                            is_spouse_of = scope:actor
                            is_consort_of = scope:actor
                            has_relation_soulmate = scope:actor
                        }
                    }
                    NOT = {
                        AND = {
                            is_adult = yes
                            exists = scope:actor.house
                            scope:actor.house.house_head ?= scope:actor
                            house ?= scope:actor.house
                        }
                    }
                }
            }
        }
    }
    is_valid_showing_failures_only = {
        scope:recipient = { NOT = { has_character_modifier = recently_lifedrained_modifier } }
        trigger_if = {
            limit = { scope:actor = { is_ai = yes } }
            NOT = {
                scope:actor = {
                    OR = {
                        is_spouse_of = scope:recipient
                        is_consort_of = scope:recipient
                        is_close_family_of = scope:recipient
                    }
                }
            }
            NOT = { scope:recipient = { is_child_of = scope:actor } }
            NOT = { scope:recipient = { is_knight_of = scope:actor } }
            NOT = { scope:recipient = { is_councillor_of = scope:actor } }
            NOT = { scope:recipient = { any_court_position_employer = { this = scope:actor } } }
            NOT = { scope:recipient = { house ?= house:bm_house_golem } }
        }
    }

    cost = { piety = lifedrain_piety_cost_minor }
    cooldown = { months = 3 }

    on_accept = {
        debug_log = "BloodMageInteraction: lifedrain_courtier | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        scope:recipient = {
            add_character_modifier =  { modifier = lifedrained_modifier }
            temporary_negative_recently_lifedrained = yes
        }
        scope:actor ={
            if = {
                limit = {
                    has_trait = lifestyle_blood_knight
                    NOT = { has_trait = lifestyle_blood_mage }
                }
                duel = {
                    skill = prowess
                    target = scope:recipient
                    50 = {
                        desc = bm_lifedrain_courtier.001.a.success_desc
                        compare_modifier = {
                            value = scope:duel_value
                            multiplier = 3
                        }
                        send_interface_toast  = {
                            title = bm_lifedrain_courtier.001.a.success_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                            hematurgy_magic_cast_effect_minor = yes
                            hidden_effect = {
                                bm_recount_lifeforce_stacks_effect = yes
                            }
                        }
                    }
                    50 = {
                        desc = bm_lifedrain_courtier.001.a.fail_desc
                        compare_modifier = {
                            value = scope:duel_value
                            multiplier = -2
                        }
                        send_interface_toast  = {
                            title = bm_lifedrain_courtier.001.a.fail_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                            scope:recipient = {
                                add_trait = wounded_1
                                if = {
                                    limit = {
                                        NOT = { scope:actor = { bm_has_superior_stave_trigger = { STAVE = hulinhjalmur } } }
                                    }
                                    add_opinion = {
                                        target = scope:actor
                                        modifier = lifedrained_me
                                        opinion = -50
                                    }
                                }
                            }
                            temporary_negative_lifeforce_effect_minor = yes
                        }
                    }
                    10 = {
                        desc = bm_lifedrain_courtier.001.a.critical_fail_desc
                        compare_modifier = {
                            value = scope:duel_value
                            multiplier = -0.5
                        }
                        send_interface_toast  = {
                            title = bm_lifedrain_courtier.001.a.critical_fail_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                            temporary_negative_lifeforce_effect_minor = yes
                            add_trait = wounded_1
                            scope:recipient = {
                                add_trait = wounded_2
                                if = {
                                    limit = {
                                        NOT = { scope:actor = { bm_has_superior_stave_trigger = { STAVE = hulinhjalmur } } }
                                    }
                                    add_opinion = {
                                        target = scope:actor
                                        modifier = lifedrained_me
                                        opinion = -50
                                    }
                                }
                            }
                        }
                    }
                }
            }
            else = {
                duel = {
                    skills = { intrigue learning }
                    target = scope:recipient
                    50 = {
                        desc = bm_lifedrain_courtier.001.a.success_desc
                        compare_modifier = {
                            value = scope:duel_value
                            multiplier = 3
                        }
                        send_interface_toast  = {
                            title = bm_lifedrain_courtier.001.a.success_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                            hematurgy_magic_cast_effect_minor = yes
                            hidden_effect = {
                                bm_recount_lifeforce_stacks_effect = yes
                            }
                        }
                    }
                    50 = {
                        desc = bm_lifedrain_courtier.001.a.fail_desc
                        compare_modifier = {
                            value = scope:duel_value
                            multiplier = -2
                        }
                        send_interface_toast  = {
                            title = bm_lifedrain_courtier.001.a.fail_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                            scope:recipient = {
                                add_trait = wounded_1
                                if = {
                                    limit = {
                                        NOT = { scope:actor = { bm_has_superior_stave_trigger = { STAVE = hulinhjalmur } } }
                                    }
                                    add_opinion = {
                                        target = scope:actor
                                        modifier = lifedrained_me
                                        opinion = -50
                                    }
                                }
                            }
                            temporary_negative_lifeforce_effect_minor = yes
                        }
                    }
                    10 = {
                        desc = bm_lifedrain_courtier.001.a.critical_fail_desc

                        compare_modifier = {
                            value = scope:duel_value
                            multiplier = -0.5
                        }
                        send_interface_toast  = {
                            title = bm_lifedrain_courtier.001.a.critical_fail_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                            temporary_negative_lifeforce_effect_minor = yes
                            add_trait = wounded_1
                            scope:recipient = {
                                add_trait = wounded_2
                                if = {
                                    limit = {
                                        NOT = { scope:actor = { bm_has_superior_stave_trigger = { STAVE = hulinhjalmur } } }
                                    }

    }

        ai_recipients = courtiers

        ai_recipients = prisoners
    }

        ai_willingness_to_lifedrain = yes
    }
