bm_bleed_own_vein_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_bleed_own_vein_interaction
	auto_accept = yes
	common_interaction = yes
	desc = bm_bleed_own_vein_interaction_desc
	interface_priority = 185

	cooldown = {
		years = 1
	}

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = bm_bleed_own_vein_interaction_requirement_tt
			scope:actor = {
				NOT = { has_trait = wounded_3 }
				NOT = { has_trait = maimed }
			}
		}
	}

	send_options_exclusive = yes

	send_option = {
		flag = minor_bleed
		localization = bm_bleed_own_vein_interaction.minor_bleed
		current_description = bm_bleed_own_vein_interaction.minor_bleed_desc
		is_shown = {
			always = yes
		}
		is_valid = {
			scope:actor = {
				NOT = { has_trait = wounded_2 }
				NOT = { has_trait = wounded_3 }
				NOT = { has_trait = maimed }
			}
		}
		starts_enabled = {
			scope:actor = {
				NOT = { has_trait = wounded_2 }
				NOT = { has_trait = wounded_3 }
				NOT = { has_trait = maimed }
			}
		}
	}

	send_option = {
		flag = major_bleed
		localization = bm_bleed_own_vein_interaction.major_bleed
		current_description = bm_bleed_own_vein_interaction.major_bleed_desc
		is_shown = {
			always = yes
		}
		is_valid = {
			scope:actor = {
				NOT = { has_trait = wounded_3 }
				NOT = { has_trait = maimed }
				NOT = { has_character_modifier = lifedrained_modifier }
			}
		}
		starts_enabled = {
			scope:actor = {
				has_trait = wounded_2
				NOT = { has_trait = wounded_3 }
				NOT = { has_trait = maimed }
				NOT = { has_character_modifier = lifedrained_modifier }
			}
		}
	}

	on_accept = {
		debug_log = "BloodMageSelfCast: bleed_own_vein | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')])"

		if = {
			limit = { scope:minor_bleed = yes }
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = bm_bleed_own_vein_interaction.minor_bleed.toast
					left_icon = scope:actor

					if = {
						limit = { has_trait = wounded_1 }
						remove_trait = wounded_1
						add_trait = wounded_2
					}
					else = {
						add_trait = wounded_1
					}
					add_character_modifier = {
						modifier = lifeforce_modifier_minor
					}
					if = {
						limit = { has_trait = lifestyle_blood_mage }
						add_trait_xp = {
							trait = lifestyle_blood_mage
							track = hematurgy
							value = 2
						}
					}
					if = {
						limit = {
							has_trait = lifestyle_blood_knight
							NOT = { has_trait = lifestyle_blood_mage }
						}
						add_blood_knight_xp = {
							TRACK = slaughter
							XP = 2
						}
					}
					hidden_effect = {
						bm_recount_lifeforce_stacks_effect = yes
					}
				}
			}
		}

		if = {
			limit = { scope:major_bleed = yes }
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = bm_bleed_own_vein_interaction.major_bleed.toast
					left_icon = scope:actor

					if = {
						limit = { has_trait = wounded_1 }
						remove_trait = wounded_1
					}
					add_trait = wounded_2
					add_character_modifier = {
						modifier = lifedrained_modifier
					}
					add_stress = 25
					add_character_modifier = {
						modifier = lifeforce_modifier_major
					}
					if = {
						limit = { has_trait = lifestyle_blood_mage }
						add_trait_xp = {
							trait = lifestyle_blood_mage
							track = hematurgy
							value = 4
						}
					}
					if = {
						limit = {
							has_trait = lifestyle_blood_knight
							NOT = { has_trait = lifestyle_blood_mage }
						}
						add_blood_knight_xp = {
							TRACK = slaughter
							XP = 4
						}
					}
					hidden_effect = {
						bm_recount_lifeforce_stacks_effect = yes
					}
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = self
	}
	ai_potential = { always = yes }
	ai_frequency = 24
	ai_will_do = {
		base = 0
		modifier = {
			scope:actor = {
				NOT = { has_character_modifier = lifeforce_modifier_minor }
				NOT = { has_character_modifier = lifeforce_modifier_major }
				NOT = { has_character_modifier = lifeforce_modifier_superior }
			health >= 4
			}
			add = 10
		}
	}
}

bm_purge_coping_traits_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_purge_coping_traits_interaction
	auto_accept = yes
	common_interaction = yes
	desc = bm_purge_coping_traits_interaction_desc
	interface_priority = 185

	cost = {
		piety = 250
	}

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		scope:recipient = {
			is_alive = yes
			bm_has_stress_coping_trait_trigger = yes
			OR = {
				this = scope:actor
				bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			has_character_modifier = lifeforce_modifier_major
		}
	}

	on_accept = {
		debug_log = "BloodMageInteraction: purge_coping_traits | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_purge_coping_traits_interaction.toast
				left_icon = scope:actor
				right_icon = scope:recipient

				blood_magic_used_major = yes
				if = {
					limit = { scope:recipient = scope:actor }
					add_shared_bm_bk_xp = {
						TRACK = enlightenment
						XP = 4
					}
				}
				else = {
					add_shared_bm_bk_xp = {
						TRACK = benediction
						XP = 4
					}
				}
				scope:recipient = {
					if = { limit = { has_trait = drunkard } remove_trait = drunkard }
					if = { limit = { has_trait = hashishiyah } remove_trait = hashishiyah }
					if = { limit = { has_trait = rakish } remove_trait = rakish }
					if = { limit = { has_trait = reclusive } remove_trait = reclusive }
					if = { limit = { has_trait = irritable } remove_trait = irritable }
					if = { limit = { has_trait = flagellant } remove_trait = flagellant }
					if = { limit = { has_trait = profligate } remove_trait = profligate }
					if = { limit = { has_trait = improvident } remove_trait = improvident }
					if = { limit = { has_trait = contrite } remove_trait = contrite }
					if = { limit = { has_trait = comfort_eater } remove_trait = comfort_eater }
					if = { limit = { has_trait = inappetetic } remove_trait = inappetetic }
					add_stress = -50
					if = {
						limit = { NOT = { this = scope:actor } }
						add_opinion = {
							target = scope:actor
							modifier = magic_opinion_positive
							opinion = 25
						}
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = self
		ai_recipients = family
		ai_recipients = spouses
		ai_recipients = courtiers
	}
	ai_potential = { always = yes }
	ai_frequency = 24
	ai_will_do = {
		base = 0
		modifier = {
			scope:actor = {
				has_character_modifier = lifeforce_modifier_major
				piety >= 500
			}
			add = 30
		}
		modifier = {
			scope:recipient = scope:actor
			add = 20
		}
		ai_willingness_to_do_positive_magic = yes
	}
}
