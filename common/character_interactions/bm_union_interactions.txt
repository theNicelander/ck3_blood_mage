bm_consecrate_union_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_consecrate_union_interaction
	desc = bm_consecrate_union_interaction_desc
	common_interaction = yes
	auto_accept = yes

	is_highlighted = {
		scope:recipient = {
			OR = {
				is_spouse_of = scope:actor
				has_relation_soulmate = scope:actor
			}
		}
	}

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		scope:recipient = {
			is_alive = yes
			OR = {
				this = scope:actor
				is_spouse_of = scope:actor
				is_consort_of = scope:actor
				scope:actor = { is_consort_of = scope:recipient }
				has_relation_lover = scope:actor
				has_relation_soulmate = scope:actor
			}
		}
	}

	cost = {
		piety = 100
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = bm_consecrate_union_interaction_requirement
			scope:actor = {
				OR = {
					has_character_modifier = lifeforce_modifier_minor
					has_character_modifier = lifeforce_modifier_major
				}
			}
		}
		custom_description = {
			text = bm_consecrated_union_already_consecrated
			subject = scope:recipient
			NOT = {
				scope:recipient = {
					has_character_modifier = bm_consecrated_union_modifier
				}
			}
		}
	}

	send_options_exclusive = yes

	send_option = {
		flag = union_minor
		localization = bm_consecrate_union_interaction.minor
		current_description = bm_consecrate_union_interaction.minor_desc
		is_shown = {
			always = yes
		}
		is_valid = {
			scope:actor = {
				has_character_modifier = lifeforce_modifier_minor
			}
		}
		starts_enabled = {
			scope:actor = {
				has_character_modifier = lifeforce_modifier_minor
				NOT = { has_character_modifier = lifeforce_modifier_major }
			}
		}
	}

	send_option = {
		flag = union_major
		localization = bm_consecrate_union_interaction.major
		current_description = bm_consecrate_union_interaction.major_desc
		is_shown = {
			always = yes
		}
		is_valid = {
			scope:actor = {
				has_character_modifier = lifeforce_modifier_major
			}
		}
		starts_enabled = {
			scope:actor = {
				has_character_modifier = lifeforce_modifier_major
			}
		}
	}

	on_accept = {
		debug_log = "BloodMageInteraction: consecrate_union | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		scope:actor = {
			if = {
				limit = { scope:union_minor = yes }
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_consecrate_union_toast
					left_icon = scope:actor
					right_icon = scope:recipient

					blood_magic_used_minor = yes
					if = {
						limit = { has_trait = lifestyle_blood_mage }
						add_xp_bm_dynamic = {
							TRACK = bloodline
							XP = 1
						}
					}
					else = {
						add_blood_knight_xp = {
							TRACK = benediction
							XP = 1
						}
					}
					scope:recipient = {
						add_character_modifier = {
							modifier = bm_consecrated_union_modifier
							years = 3
						}
						if = {
							limit = { NOT = { this = scope:actor } }
							add_opinion = {
								target = scope:actor
								modifier = magic_opinion_positive
								opinion = 20
							}
						}
					}
					hidden_effect = {
						bm_recount_lifeforce_stacks_effect = yes
					}
				}
			}
			else_if = {
				limit = { scope:union_major = yes }
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_consecrate_union_toast
					left_icon = scope:actor
					right_icon = scope:recipient

					blood_magic_used_major = yes
					if = {
						limit = { has_trait = lifestyle_blood_mage }
						add_xp_bm_dynamic = {
							TRACK = bloodline
							XP = 2
						}
					}
					else = {
						add_blood_knight_xp = {
							TRACK = benediction
							XP = 2
						}
					}
					scope:recipient = {
						add_character_modifier = {
							modifier = bm_consecrated_union_modifier
							years = 10
						}
						if = {
							limit = { NOT = { this = scope:actor } }
							add_opinion = {
								target = scope:actor
								modifier = magic_opinion_positive
								opinion = 35
							}
						}
					}
					hidden_effect = {
						bm_recount_lifeforce_stacks_effect = yes
					}
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = self
		ai_recipients = spouses
		ai_recipients = scripted_relations
	}
	ai_potential = { always = yes }
	ai_frequency = 12
	ai_will_do = {
		base = 20
		ai_willingness_to_do_positive_magic = yes

		modifier = {
			scope:actor = {
				NOR = {
					has_character_modifier = lifeforce_modifier_minor
					has_character_modifier = lifeforce_modifier_major
				}
			}
			factor = 0
		}
		modifier = {
			scope:recipient = {
				is_spouse_of = scope:actor
			}
			add = 30
		}
		modifier = {
			scope:recipient = {
				has_relation_soulmate = scope:actor
			}
			add = 40
		}
	}
}
