bm_blood_drain_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_blood_drain_interaction
	auto_accept = yes
	common_interaction = yes
	desc = bm_blood_drain_interaction_desc
	interface_priority = 300

	cost = {
		piety = {
			if = {
				limit = { scope:drain_major = yes }
				value = lifedrain_piety_cost_major
			}
			else_if = {
				limit = { scope:drain_minor = yes }
				value = lifedrain_piety_cost_minor
			}
			else_if = {
				limit = { scope:drain_stored = yes }
				value = 75
			}
			else_if = {
				limit = { scope:drain_tithe = yes }
				value = 50
			}
			else = {
				value = 0
			}
		}
	}

	is_shown = {
		scope:actor = {
			is_blood_mage_any_tier_trigger = yes
		}
		scope:recipient = {
			is_alive = yes
			NOT = { this = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = bm_blood_drain_interaction_requirement
			scope:recipient = { is_alive = yes }
		}
	}

	send_options_exclusive = yes

	# 1. Major Life drain
	send_option = {
		flag = drain_major
		localization = bm_blood_drain_option_major
		current_description = bm_blood_drain_option_major_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:actor = { is_blood_mage_any_tier_trigger = yes }
			scope:recipient = { is_imprisoned_by = scope:actor }
		}
		is_valid = {
			scope:recipient = {
				NOT = { has_character_modifier = lifedrained_modifier }
			}
		}
		starts_enabled = {
			scope:recipient = {
				is_imprisoned_by = scope:actor
				NOT = { has_character_modifier = lifedrained_modifier }
			}
		}
	}

	# 2. Minor Life drain
	send_option = {
		flag = drain_minor
		localization = bm_blood_drain_option_minor
		current_description = bm_blood_drain_option_minor_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:recipient = {
				OR = {
					is_imprisoned_by = scope:actor
					is_courtier_of = scope:actor
				}
			}
		}
		is_valid = {
			scope:recipient = {
				NOT = { has_character_modifier = recently_lifedrained_modifier }
			}
			scope:actor = {
				NOT = { has_character_flag = bm_minor_lifedrain_cooldown }
			}
		}
		starts_enabled = {
			scope:recipient = {
				is_courtier_of = scope:actor
				NOT = { is_imprisoned_by = scope:actor }
			}
		}
	}

	# 3. Drain Stored Lifeforce (renamed from Siphon)
	send_option = {
		flag = drain_stored
		localization = bm_blood_drain_option_stored
		current_description = bm_blood_drain_option_stored_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:recipient = {
				OR = {
					has_character_modifier = lifeforce_modifier_minor
					has_character_modifier = lifeforce_modifier_major
					has_character_modifier = lifeforce_modifier_superior
				}
			}
		}
		is_valid = {
			scope:actor = {
				NOT = { has_character_flag = bm_siphon_reserves_cooldown }
			}
		}
		starts_enabled = {
			scope:recipient = {
				OR = {
					has_character_modifier = lifeforce_modifier_minor
					has_character_modifier = lifeforce_modifier_major
					has_character_modifier = lifeforce_modifier_superior
				}
			}
		}
	}

	# 4. Demand Blood Tithe
	send_option = {
		flag = drain_tithe
		localization = bm_blood_drain_option_tithe
		current_description = bm_blood_drain_option_tithe_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:actor = {
				exists = house
				house.house_head = scope:actor
			}
			scope:recipient = {
				is_adult = yes
				house ?= scope:actor.house
			}
		}
		is_valid = {
			scope:recipient = {
				NOT = { has_character_modifier = recently_lifedrained_modifier }
			}
			scope:actor = {
				NOT = { has_character_flag = bm_demand_blood_tithe_cooldown }
			}
		}
		starts_enabled = {
			scope:actor = {
				exists = house
				house.house_head = scope:actor
			}
			scope:recipient = { house ?= scope:actor.house }
		}
	}

	on_accept = {
		if = {
			limit = { scope:drain_major = yes }
			debug_log = "BloodMageInteraction: bm_blood_drain (major) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
			scope:actor = { hematurgy_magic_cast_effect_major = yes }
			lifedrain_execution_effect = {
				VICTIM = scope:recipient
				EXECUTIONER = scope:actor
			}
		}
		else_if = {
			limit = { scope:drain_minor = yes }
			debug_log = "BloodMageInteraction: bm_blood_drain (minor) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
			scope:actor = {
				add_character_flag = {
					flag = bm_minor_lifedrain_cooldown
					months = 3
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_blood_drain_toast
					left_icon = scope:actor
					right_icon = scope:recipient
					add_character_modifier = { modifier = lifeforce_modifier_minor }
					bm_add_blood_mage_xp_effect = {
						TRACK = hematurgy
						XP = 1
					}
					scope:recipient = {
						add_character_modifier = {
							modifier = recently_lifedrained_modifier
							years = 1
						}
						add_character_modifier = {
							modifier = lifeforce_modifier_negative_minor
							years = 1
						}
						if = {
							limit = {
								NOT = { bm_is_blood_cult_faith_trigger = yes }
								NOT = { has_trait = loyal }
							}
							add_opinion = {
								target = scope:actor
								modifier = magic_opinion_negative
								opinion = -25
							}
						}
					}
				}
			}
		}
		else_if = {
			limit = { scope:drain_stored = yes }
			debug_log = "BloodMageInteraction: bm_blood_drain (stored) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
			scope:actor = {
				add_character_flag = {
					flag = bm_siphon_reserves_cooldown
					years = 1
				}
				if = {
					limit = { scope:recipient = { is_imprisoned_by = scope:actor } }
					send_interface_toast = {
						type = event_toast_effect_good
						title = bm_blood_drain_toast
						left_icon = scope:actor
						right_icon = scope:recipient
						if = {
							limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_superior } }
							scope:recipient = { remove_character_modifier = lifeforce_modifier_superior }
							add_character_modifier = { modifier = lifeforce_modifier_superior }
						}
						else_if = {
							limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_major } }
							scope:recipient = { remove_character_modifier = lifeforce_modifier_major }
							add_character_modifier = { modifier = lifeforce_modifier_major }
						}
						else_if = {
							limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_minor } }
							scope:recipient = { remove_character_modifier = lifeforce_modifier_minor }
							add_character_modifier = { modifier = lifeforce_modifier_minor }
						}
						if = {
							limit = { NOT = { scope:actor = { bm_has_superior_stave_trigger = { STAVE = hulinhjalmur } } } }
							scope:recipient = {
								add_opinion = {
									target = scope:actor
									modifier = lifedrained_me
									opinion = -50
								}
							}
						}
						add_xp_bm_dynamic = { TRACK = hematurgy XP = 3 }
					}
				}
				else = {
					duel = {
						skills = { learning prowess }
						target = scope:recipient
						50 = {
							compare_modifier = { value = scope:duel_value multiplier = 2 }
							send_interface_toast = {
								type = event_toast_effect_good
								title = bm_blood_drain_toast
								left_icon = scope:actor
								right_icon = scope:recipient
								if = {
									limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_superior } }
									scope:recipient = { remove_character_modifier = lifeforce_modifier_superior }
									add_character_modifier = { modifier = lifeforce_modifier_superior }
								}
								else_if = {
									limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_major } }
									scope:recipient = { remove_character_modifier = lifeforce_modifier_major }
									add_character_modifier = { modifier = lifeforce_modifier_major }
								}
								else_if = {
									limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_minor } }
									scope:recipient = { remove_character_modifier = lifeforce_modifier_minor }
									add_character_modifier = { modifier = lifeforce_modifier_minor }
								}
								add_xp_bm_dynamic = { TRACK = hematurgy XP = 3 }
							}
						}
						50 = {
							compare_modifier = { value = scope:duel_value multiplier = -2 }
							send_interface_toast = {
								type = event_toast_effect_bad
								title = bm_blood_drain_toast
								left_icon = scope:actor
								right_icon = scope:recipient
								add_stress = 20
							}
						}
					}
				}
			}
		}
		else_if = {
			limit = { scope:drain_tithe = yes }
			debug_log = "BloodMageInteraction: bm_blood_drain (tithe) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
			scope:actor = {
				add_character_flag = {
					flag = bm_demand_blood_tithe_cooldown
					years = 2
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = bm_blood_drain_toast
					left_icon = scope:actor
					right_icon = scope:recipient
					if = {
						limit = {
							scope:recipient = {
								is_blood_mage_any_tier_trigger = yes
							}
						}
						add_character_modifier = { modifier = lifeforce_modifier_major }
					}
					else = {
						add_character_modifier = { modifier = lifeforce_modifier_minor }
					}
					bm_add_blood_mage_xp_effect = {
						TRACK = bloodline
						XP = 2
					}
					scope:recipient = {
						add_character_modifier = {
							modifier = recently_lifedrained_modifier
							years = 1
						}
						add_character_modifier = {
							modifier = lifeforce_modifier_negative_minor
							years = 1
						}
						if = {
							limit = {
								NOT = { bm_is_blood_cult_faith_trigger = yes }
								NOT = { has_trait = loyal }
							}
							add_opinion = {
								target = scope:actor
								modifier = magic_opinion_negative
								opinion = -25
							}
						}
					}
				}
			}
		}
		else = {
			debug_log = "BloodMageInteraction: bm_blood_drain (NO OPTION) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
		}
	}

	ai_potential = { always = yes }
	ai_frequency = 6
	ai_targets = {
		ai_recipients = prisoners
		ai_recipients = courtiers
		ai_recipients = dynasty
	}
	ai_will_do = {
		base = 25

		# Score send options
		modifier = {
			scope:drain_stored = yes
			add = 50
		}
		modifier = {
			scope:drain_major = yes
			add = 40
		}
		modifier = {
			scope:drain_minor = yes
			add = 25
		}
		modifier = {
			scope:drain_tithe = yes
			add = 15
		}

		# Personality, relationship, and protective vetoes (evaluated last)
		ai_willingness_to_lifedrain = yes

		# Absolute veto: must select at least one valid send option
		modifier = {
			NOR = {
				scope:drain_major = yes
				scope:drain_minor = yes
				scope:drain_stored = yes
				scope:drain_tithe = yes
			}
			factor = 0
		}
	}
}
