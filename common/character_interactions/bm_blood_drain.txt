bm_blood_drain_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_blood_drain_interaction
	auto_accept = yes
	common_interaction = yes
	desc = bm_blood_drain_interaction_desc
	interface_priority = 300

	cost = {
		piety = {
			if = {
				limit = { scope:drain_major = yes }
				value = lifedrain_piety_cost_major
			}
			else_if = {
				limit = { scope:drain_minor = yes }
				value = bm_drain_minor_piety_cost
			}
			else_if = {
				limit = { scope:drain_minor_empowered = yes }
				value = bm_drain_minor_empowered_piety_cost
			}
			else_if = {
				limit = { scope:drain_stored = yes }
				value = 75
			}
			else_if = {
				limit = { scope:drain_tithe = yes }
				value = 50
			}
			else = {
				value = 0
			}
		}
	}

	is_shown = {
		scope:actor = {
			is_blood_mage_any_tier_trigger = yes
		}
		scope:recipient = {
			is_alive = yes
			NOT = { this = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = bm_blood_drain_interaction_requirement
			scope:recipient = { is_alive = yes }
		}
	}

	send_options_exclusive = yes

	# 1. Major Life drain
	send_option = {
		flag = drain_major
		localization = bm_blood_drain_option_major
		current_description = bm_blood_drain_option_major_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:actor = { is_blood_mage_any_tier_trigger = yes }
			scope:recipient = { is_imprisoned_by = scope:actor }
		}
		is_valid = {
			scope:recipient = {
				NOT = { has_character_modifier = lifedrained_modifier }
			}
		}
		starts_enabled = {
			scope:recipient = {
				is_imprisoned_by = scope:actor
				NOT = { has_character_modifier = lifedrained_modifier }
			}
		}
	}

	# 2. Minor Life drain
	send_option = {
		flag = drain_minor
		localization = bm_blood_drain_option_minor
		current_description = bm_blood_drain_option_minor_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:recipient = {
				OR = {
					is_imprisoned_by = scope:actor
					is_courtier_of = scope:actor
					is_vassal_of = scope:actor
					AND = {
						exists = scope:actor.liege
						this = scope:actor.liege
					}
				}
			}
		}
		is_valid = {
			scope:recipient = {
				custom_tooltip = {
					text = bm_minor_lifedrain_victim_cooldown_tt
					NOT = { has_character_flag = bm_minor_lifedrain_cooldown }
				}
			}
		}
		starts_enabled = {
			scope:recipient = {
				OR = {
					is_courtier_of = scope:actor
					is_vassal_of = scope:actor
					AND = {
						exists = scope:actor.liege
						this = scope:actor.liege
					}
				}
				NOT = { is_imprisoned_by = scope:actor }
				NOT = { has_character_flag = bm_minor_lifedrain_cooldown }
			}
		}
	}

	# 2b. Empowered Minor Life drain (100 Piety, 80% base chance)
	send_option = {
		flag = drain_minor_empowered
		localization = bm_blood_drain_option_minor_empowered
		current_description = bm_blood_drain_option_minor_empowered_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:recipient = {
				NOT = { is_imprisoned_by = scope:actor }
				OR = {
					is_courtier_of = scope:actor
					is_vassal_of = scope:actor
					AND = {
						exists = scope:actor.liege
						this = scope:actor.liege
					}
				}
			}
		}
		is_valid = {
			scope:recipient = {
				custom_tooltip = {
					text = bm_minor_lifedrain_victim_cooldown_tt
					NOT = { has_character_flag = bm_minor_lifedrain_cooldown }
				}
			}
		}
		starts_enabled = { always = no }
	}

	# 3. Drain Stored Lifeforce (renamed from Siphon)
	send_option = {
		flag = drain_stored
		localization = bm_blood_drain_option_stored
		current_description = bm_blood_drain_option_stored_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:recipient = {
				OR = {
					has_character_modifier = lifeforce_modifier_minor
					has_character_modifier = lifeforce_modifier_major
					has_character_modifier = lifeforce_modifier_superior
				}
			}
		}
		is_valid = {
			scope:actor = {
				NOT = { has_character_flag = bm_siphon_reserves_cooldown }
			}
		}
		starts_enabled = {
			scope:actor = {
				NOT = { has_character_flag = bm_siphon_reserves_cooldown }
			}
			scope:recipient = {
				OR = {
					has_character_modifier = lifeforce_modifier_minor
					has_character_modifier = lifeforce_modifier_major
					has_character_modifier = lifeforce_modifier_superior
				}
			}
		}
	}

	# 4. Demand Blood Tithe
	send_option = {
		flag = drain_tithe
		localization = bm_blood_drain_option_tithe
		current_description = bm_blood_drain_option_tithe_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:actor = {
				exists = house
				house.house_head = scope:actor
			}
			scope:recipient = {
				is_adult = yes
				house ?= scope:actor.house
			}
		}
		is_valid = {
			scope:recipient = {
				NOT = { has_character_modifier = recently_lifedrained_modifier }
			}
			scope:actor = {
				NOT = { has_character_flag = bm_demand_blood_tithe_cooldown }
			}
		}
		starts_enabled = {
			scope:actor = {
				exists = house
				house.house_head = scope:actor
				NOT = { has_character_flag = bm_demand_blood_tithe_cooldown }
			}
			scope:recipient = {
				house ?= scope:actor.house
				NOT = { has_character_modifier = recently_lifedrained_modifier }
			}
		}
	}

	# 5. Drain Positive Traits
	send_option = {
		flag = drain_trait
		localization = bm_blood_drain_option_trait
		current_description = bm_blood_drain_option_trait_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:actor = { bm_can_drain_traits_from_target_trigger = yes }
		}
		is_valid = {
			scope:actor = { bm_can_drain_traits_from_target_tier_valid_trigger = yes }
			scope:recipient = { is_imprisoned_by = scope:actor }

				custom_tooltip = {
					text = bm_trait_drain_caster_cooldown_tt

				}
			}
			scope:recipient = {
				custom_tooltip = {
					text = bm_trait_drain_victim_cooldown_tt
					NOT = { has_character_flag = bm_trait_drain_victim_cooldown }
				}
			}
		}
		starts_enabled = { always = no }
	}

	# 6. Drain Blood Memory

		flag = drain_memory
		localization = bm_blood_drain_option_memory
		current_description = bm_blood_drain_option_memory_desc
		can_invalidate_interaction = yes
		is_shown = {
			scope:actor = { bm_can_drain_memory_from_target_trigger = yes }
		}
		is_valid = {
			scope:recipient = {
				is_imprisoned_by = scope:actor
				NOT = { has_character_modifier = lifedrained_modifier }
			}
			scope:actor = {
				has_character_modifier = lifeforce_modifier_major
				piety >= bm_memory_drain_piety_cost
				custom_tooltip = {
					text = bm_memory_drain_caster_cooldown_tt
					NOT = { has_character_flag = bm_memory_drain_cooldown }
				}
			}
			scope:recipient = {
				custom_tooltip = {
					text = bm_memory_drain_victim_cooldown_tt
					NOT = { has_character_flag = bm_memory_drain_victim_cooldown }
				}
			}
		}
		starts_enabled = { always = no }
	}

					skills = { intrigue learning }

							value = scope:duel_value
							multiplier = 2.67
							min = -70
							max = 15

							add = -20
							scope:recipient = {
								NOT = { is_courtier_of = scope:actor }
							}
						}

						title = bm_blood_drain_toast
						left_icon = scope:actor

						if = {

							scope:recipient = { remove_character_modifier = lifeforce_modifier_superior }
							add_character_modifier = { modifier = lifeforce_modifier_superior }
						}
						else_if = {
							limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_major } }

							limit = { scope:recipient = { has_character_modifier = lifeforce_modifier_minor } }
							scope:recipient = { remove_character_modifier = lifeforce_modifier_minor }
