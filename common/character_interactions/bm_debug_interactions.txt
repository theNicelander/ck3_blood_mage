bm_debug_character_interaction = {
	category = interaction_debug_main
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	auto_accept = yes
	icon = grant_blood_magic_interaction

	desc = bm_debug_character_interaction_desc

	is_shown = {
		debug_only = yes
	}

	send_option = {
		flag = make_blood_mage
		localization = bm_debug_option_make_blood_mage
	}
	send_option = {
		flag = make_blood_knight
		localization = bm_debug_option_make_blood_knight
	}
	send_option = {
		flag = convert_to_blodtru
		localization = bm_debug_option_convert_to_blodtru
	}

	on_accept = {
		scope:recipient = {
			switch = {
				trigger = yes
				scope:make_blood_mage = {
					remove_trait = lifestyle_blood_knight
					bm_become_blood_mage_effect = yes
				}
				scope:make_blood_knight = {
					remove_trait = lifestyle_blood_mage
					bm_become_blood_knight_effect = yes
				}
				scope:convert_to_blodtru = {
					set_character_faith = faith:blodtru_faith
				}
			}
		}
	}
}
