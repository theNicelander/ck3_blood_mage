grant_lifeforce_interaction = {
    category = interaction_category_bm_blood_magic
    auto_accept = yes
    common_interaction = yes
    desc = grant_lifeforce_interaction_desc
    interface_priority = 190

    cost = {
        piety = {
            if = {
                limit = { scope:grant_major = yes }
                value = lifedrain_piety_cost_major
            }
            else = {
                value = lifedrain_piety_cost_minor
            }
        }
    }

    is_shown = {
        scope:actor = {
            OR = {
                has_trait = lifestyle_blood_mage
                has_trait = lifestyle_blood_knight
            }
        }
        OR = {
            AND = {
                scope:actor = scope:recipient
                scope:actor = { has_character_modifier = lifedrained_modifier }
            }
            AND = {
                NOT = { scope:actor = scope:recipient }
                scope:recipient = {
                    OR = {
                        has_character_modifier = lifedrained_modifier
                        bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
                        is_imprisoned_by = scope:actor
                    }
                }
            }
        }
    }
    is_valid_showing_failures_only = {
        trigger_if = {
            limit = {
                scope:recipient = { has_character_modifier = lifedrained_modifier }
            }
            custom_description = {
                text = "grant_lifeforce_reverse_lifedrain_requirement"
                scope:actor = {
                    has_character_modifier = lifeforce_modifier_major
                    piety >= lifedrain_piety_cost_minor
                }
            }
        }
        trigger_else = {
            custom_description = {
                text = "grant_lifeforce_interaction_requirement"
                scope:actor = {
                    OR = {
                        AND = {
                            has_character_modifier = lifeforce_modifier_minor
                            piety >= lifedrain_piety_cost_minor
                        }
                        AND = {
                            has_character_modifier = lifeforce_modifier_major
                            piety >= lifedrain_piety_cost_minor
                        }
                    }
                }
            }
        }
    }

    send_options_exclusive = yes

    send_option = { # Give minor
        localization = grant_lifeforce_interaction.grant_minor
        flag = grant_minor
        is_shown = {
            NOT = { scope:actor = scope:recipient }
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= lifedrain_piety_cost_minor
            }
        }
        starts_enabled = {
            scope:recipient = {
                NOT = { has_character_modifier = lifedrained_modifier }
            }
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= lifedrain_piety_cost_minor
            }
        }
    }
    send_option = { # Give major
        localization = grant_lifeforce_interaction.grant_major
        flag = grant_major
        is_shown = {
            NOT = { scope:actor = scope:recipient }
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= lifedrain_piety_cost_major
            }
        }
        starts_enabled = {
            scope:recipient = {
                NOT = { has_character_modifier = lifedrained_modifier }
            }
            scope:actor = {
                NOT = { has_character_modifier = lifeforce_modifier_minor }
                has_character_modifier = lifeforce_modifier_major
                piety >= lifedrain_piety_cost_major
            }
        }
    }
    send_option = { # Reverse lifedrain
        localization = grant_lifeforce_interaction.reverse_lifedrain
        current_description = grant_lifeforce_interaction_reversed_desc
        flag = reverse_lifedrain
        is_shown = {
            scope:recipient = {
                has_character_modifier = lifedrained_modifier
            }
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= lifedrain_piety_cost_minor
            }
        }
        starts_enabled = {
            scope:recipient = {
                has_character_modifier = lifedrained_modifier
            }
        }
    }

    on_accept = {
        debug_log = "BloodMageInteraction: grant_lifeforce | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"

        # Reverse lifedrained
        if = {
            limit = { scope:reverse_lifedrain = yes }
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = grant_lifeforce_interaction_reversed
                    left_icon = scope:actor
                    right_icon = scope:recipient

                    scope:recipient = {
                        remove_character_modifier = lifedrained_modifier
                        remove_character_modifier = recently_lifedrained_modifier
                        add_character_modifier = lifeforce_modifier_minor
                        trigger_if = {
                            limit = { NOT = { this = scope:actor } }
                            add_opinion = {
                                target = scope:actor
                                modifier = magic_opinion_positive
                                opinion = 20
                            }
                        }
                    }
                    remove_character_modifier = lifeforce_modifier_major
                    add_shared_bm_bk_xp = {
                        TRACK = benediction
                        XP = 4
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                        trigger_if = {
                            limit = { NOT = { scope:recipient = scope:actor } }
                            scope:recipient = {
                                bm_recount_lifeforce_stacks_effect = yes
                            }
                        }
                    }
                }
            }
        }
        # Grant minor lifeforce
        else_if = {
            limit = { scope:grant_minor = yes }
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_neutral
                    title = grant_lifeforce_interaction.toast
                    left_icon = scope:actor
                    right_icon = scope:recipient

                    remove_character_modifier = lifeforce_modifier_minor
                    add_shared_bm_bk_xp = {
                        TRACK = enlightenment
                        XP = 1
                    }
                    scope:recipient = {
                        add_character_modifier = lifeforce_modifier_minor
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 10
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                        scope:recipient = {
                            bm_recount_lifeforce_stacks_effect = yes
                        }
                    }
                }
            }
        }
        # Grant major lifeforce
        else_if = {
            limit = { scope:grant_major = yes }
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_neutral
                    title = grant_lifeforce_interaction.toast
                    left_icon = scope:actor
                    right_icon = scope:recipient

                    remove_character_modifier = lifeforce_modifier_major
                    add_shared_bm_bk_xp = {
                        TRACK = enlightenment
                        XP = 2
                    }
                    scope:recipient = {
                        add_character_modifier = lifeforce_modifier_major
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 25
                        }
                    }
                    hidden_effect = {
                        bm_recount_lifeforce_stacks_effect = yes
                        scope:recipient = {
                            bm_recount_lifeforce_stacks_effect = yes
                        }
                    }
                }
            }
        }
    }
    # AI restores lifedrained positive targets
    ai_targets = {
        ai_recipients = self
        ai_recipients = dynasty
        ai_recipients = spouses
        ai_recipients = family
        ai_recipients = children
        ai_recipients = courtiers
        ai_recipients = councillors
        ai_recipients = liege
    }
    ai_potential = { always = yes }
    ai_frequency = 12
    ai_will_do = {
        base = -10
        ai_willingness_to_do_positive_magic = yes
        modifier = {
            factor = 0
            NOT = { scope:reverse_lifedrain = yes }
        }
    }
}
