bm_inscribe_rune_other_interaction = {
	category = interaction_category_bm_blood_magic
	icon = bm_inscribe_rune_other_interaction
	desc = bm_inscribe_rune_other_interaction_desc
	common_interaction = yes

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
		}
		scope:recipient = {
			is_alive = yes
			NOT = { this = scope:actor }
			NOT = { has_character_modifier = superior_blood_rune_modifier }
			trigger_if = {
				limit = {
					scope:actor = {
						NOT = { has_trait = lifestyle_blood_mage }
						has_trait = lifestyle_blood_knight
					}
				}
				NOT = { has_character_modifier = major_blood_rune_modifier }
			}
			bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_adult = yes
			is_imprisoned = no
		}
		scope:actor = {
			custom_description = {
				text = bm_blood_rune_other_cooldown_tt
				NOT = { has_character_flag = bm_blood_rune_other_cooldown }
			}
			custom_tooltip = {
				text = bm_inscribe_rune_other_requirement_tt
				OR = {
					# Tier 1: Target has no rune -> Minor
					AND = {
						scope:recipient = {
							NOT = { has_character_modifier = minor_blood_rune_modifier }
							NOT = { has_character_modifier = major_blood_rune_modifier }
							NOT = { has_character_modifier = superior_blood_rune_modifier }
						}
						piety_level >= 2
						OR = {
							scope:actor.bm_total_blood_mage_xp >= 25
							scope:actor.bm_total_blood_knight_xp >= 25
						}
						has_character_modifier = lifeforce_modifier_minor
					}
					# Tier 2: Target has Minor rune -> Major
					AND = {
						scope:recipient = {
							has_character_modifier = minor_blood_rune_modifier
							NOT = { has_character_modifier = major_blood_rune_modifier }
							NOT = { has_character_modifier = superior_blood_rune_modifier }
						}
						piety_level >= 3
						OR = {
							scope:actor.bm_total_blood_mage_xp >= 50
							scope:actor.bm_total_blood_knight_xp >= 50
						}
						piety >= 200
						has_character_modifier = lifeforce_modifier_major
					}
					# Tier 3: Target has Major rune -> Superior (Blood Mage only)
					AND = {
						scope:recipient = {
							has_character_modifier = major_blood_rune_modifier
							NOT = { has_character_modifier = superior_blood_rune_modifier }
						}
						has_trait = lifestyle_blood_mage
						piety_level >= 4
						scope:actor.bm_total_blood_mage_xp >= 100
						piety >= 400
						has_character_modifier = lifeforce_modifier_superior
					}
				}
			}
		}
	}

	on_accept = {
		scope:actor = {
			add_character_flag = {
				flag = bm_blood_rune_other_cooldown
				years = 5
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_inscribe_rune_other_interaction
				left_icon = scope:actor
				right_icon = scope:recipient

				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = magic_opinion_positive
						opinion = 20
					}
				}
				if = {
					limit = {
						scope:recipient = {
							NOT = { has_character_modifier = minor_blood_rune_modifier }
							NOT = { has_character_modifier = major_blood_rune_modifier }
							NOT = { has_character_modifier = superior_blood_rune_modifier }
						}
					}
					blood_magic_used_minor = yes
					if = {
						limit = { has_trait = lifestyle_blood_mage }
						add_trait_xp = {
							trait = lifestyle_blood_mage
							track = benediction
							value = 4
						}
					}
					if = {
						limit = {
							has_trait = lifestyle_blood_knight
							NOT = { has_trait = lifestyle_blood_mage }
						}
						add_blood_knight_xp = {
							TRACK = benediction
							XP = 4
						}
					}
					scope:recipient = {
						add_character_modifier = {
							modifier = minor_blood_rune_modifier
						}
					}
				}
				else_if = {
					limit = {
						scope:recipient = {
							has_character_modifier = minor_blood_rune_modifier
							NOT = { has_character_modifier = major_blood_rune_modifier }
							NOT = { has_character_modifier = superior_blood_rune_modifier }
						}
					}
					add_piety = -200
					blood_magic_used_major = yes
					if = {
						limit = { has_trait = lifestyle_blood_mage }
						add_trait_xp = {
							trait = lifestyle_blood_mage
							track = benediction
							value = 5
						}
					}
					if = {
						limit = {
							has_trait = lifestyle_blood_knight
							NOT = { has_trait = lifestyle_blood_mage }
						}
						add_blood_knight_xp = {
							TRACK = benediction
							XP = 5
						}
					}
					scope:recipient = {
						bm_upgrade_blood_rune_to_major_effect = yes
					}
				}
				else_if = {
					limit = {
						scope:recipient = {
							has_character_modifier = major_blood_rune_modifier
							NOT = { has_character_modifier = superior_blood_rune_modifier }
						}
					}
					add_piety = -400
					blood_magic_used_superior = yes
					add_trait_xp = {
						trait = lifestyle_blood_mage
						track = benediction
						value = 10
					}
					scope:recipient = {
						bm_upgrade_blood_rune_to_superior_effect = yes
					}
				}
				hidden_effect = {
					bm_recount_lifeforce_stacks_effect = yes
				}
			}
		}
	}

	auto_accept = yes
	ai_accept = { base = 100 }
	ai_potential = { always = yes }
	ai_targets = {
		ai_recipients = family
		ai_recipients = spouses
		ai_recipients = children
		ai_recipients = courtiers
		ai_recipients = liege
	}
	ai_frequency = 60
	ai_will_do = {
		base = 0
		modifier = {
			scope:actor = { piety < 500 }
			factor = 0
		}
		modifier = {
			scope:recipient = { is_spouse_of = scope:actor }
			add = 60
		}
		modifier = {
			scope:recipient = {
				OR = {
					is_primary_heir_of = scope:actor
					is_child_of = scope:actor
				}
			}
			add = 50
		}
		modifier = {
			scope:recipient = { has_trait = lifestyle_blood_knight }
			add = 40
		}
		modifier = {
			scope:recipient = { house ?= house:bm_house_golem }
			add = 40
		}
		ai_willingness_to_do_positive_magic = yes
	}
}
