# Trigger event
trait_drain_prisoner_event_interaction = {
    common_interaction = yes
    interface_priority = 300
    category = interaction_category_hostile
    desc = generic_drain_prisoner_trait_interaction_desc

    is_highlighted = { always = yes }
    cooldown = { days = 60 }
    cooldown_against_recipient = { days = 180 }

    is_valid_showing_failures_only = {
        scope:actor = {
            has_character_modifier = lifeforce_modifier_major
        }
    }

    is_shown = {
        scope:actor = {
            has_trait = lifestyle_blood_mage
            bm_can_drain_traits_from_target_trigger = yes
        }
        scope:recipient = { is_imprisoned_by = scope:actor }
    }

    on_accept = {
        debug_log = "BloodMageInteraction: trait_drain_prisoner | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        scope:actor = {
            trigger_event = bm_trait_drain.001
            remove_character_modifier = lifeforce_modifier_major
            custom_tooltip = generic_drain_prisoner_trait_interaction_desc
        }

    }

    # AI
    auto_accept = yes
    ai_potential = { always = yes }
    ai_targets = { ai_recipients = prisoners }
    ai_frequency = 3
    ai_will_do = {
        base = 20
        ai_willingness_to_lifedrain = yes
    }
}
