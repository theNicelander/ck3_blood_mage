grant_blood_magic_interaction = {
    category = interaction_category_bm_blood_magic
    auto_accept = yes
    common_interaction = yes
    desc = grant_blood_magic_interaction_desc
    interface_priority = 200

    is_highlighted = { always = yes }

    cost = {
        piety = {
            if = {
                limit = { scope:mage_major_initiation = yes }
                value = bm_initiation_major_piety_cost
            }
            if = {
                limit = { scope:mage_minor_initiation = yes }
                value = bm_initiation_minor_piety_cost
            }
        }
    }

    is_shown = {
        NOT = { scope:actor = scope:recipient }
        scope:actor = { is_blood_mage_any_tier_trigger = yes }
        scope:recipient = {
            NOT = { is_blood_mage_any_tier_trigger = yes }
            bm_is_valid_positive_magic_target_trigger = { ACTOR = scope:actor }
        }
    }
    is_valid_showing_failures_only = {
        custom_description = {
            text = grant_blood_magic_interaction_requirement
            scope:actor = {
                OR = {
                    AND = {
                        has_character_modifier = lifeforce_modifier_minor
                        piety >= bm_initiation_minor_piety_cost
                    }
                    AND = {
                        has_character_modifier = lifeforce_modifier_major
                        piety >= bm_initiation_major_piety_cost
                    }
                }
            }
        }
    }

    send_options_exclusive = yes

    send_option = {
        flag = mage_minor_initiation
        localization = grant_blood_magic_interaction.minor
        current_description = grant_blood_magic_interaction.minor_desc
        can_invalidate_interaction = yes
        is_shown = {
            always = yes
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= bm_initiation_minor_piety_cost
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_minor
                piety >= bm_initiation_minor_piety_cost
                OR = {
                    NOT = { has_character_modifier = lifeforce_modifier_major }
                    piety < bm_initiation_major_piety_cost
                }
            }
        }
    }

    send_option = {
        flag = mage_major_initiation
        localization = grant_blood_magic_interaction.major
        current_description = grant_blood_magic_interaction.major_desc
        can_invalidate_interaction = yes
        is_shown = {
            always = yes
        }
        is_valid = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= bm_initiation_major_piety_cost
            }
        }
        starts_enabled = {
            scope:actor = {
                has_character_modifier = lifeforce_modifier_major
                piety >= bm_initiation_major_piety_cost
            }
        }
    }

    on_accept = {
        if = {
            limit = { scope:mage_minor_initiation = yes }
            debug_log = "BloodMageInteraction: grant_blood_magic (minor) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = grant_blood_magic_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    remove_character_modifier = lifeforce_modifier_minor
                    add_xp_bm_dynamic = {
                        TRACK = bloodline
                        XP = 2
                    }
                    scope:recipient = {
                        bm_become_blood_acolyte_effect = yes
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 50
                        }
                    }
                }
            }
        }
        else_if = {
            limit = { scope:mage_major_initiation = yes }
            debug_log = "BloodMageInteraction: grant_blood_magic (major) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
            scope:actor = {
                send_interface_toast = {
                    type = event_toast_effect_good
                    title = grant_blood_magic_interaction
                    left_icon = scope:actor
                    right_icon = scope:recipient
                    remove_character_modifier = lifeforce_modifier_major
                    if = {
                        limit = {
                            scope:recipient = {
                                OR = {
                                    is_child_of = scope:actor
                                    dynasty ?= scope:actor.dynasty
                                }
                            }
                        }
                        add_xp_bm_dynamic = {
                            TRACK = bloodline
                            XP = 4
                        }
                    }
                    else = {
                        add_xp_bm_dynamic = {
                            TRACK = bloodline
                            XP = 3
                        }
                    }
                    scope:recipient = {
                        bm_become_blood_acolyte_effect = yes
                        add_opinion = {
                            target = scope:actor
                            modifier = magic_opinion_positive
                            opinion = 50
                        }
                    }
                }
            }
        }
        else = {
            debug_log = "BloodMageInteraction: grant_blood_magic (NO OPTION) | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        }
    }

    # AI
    ai_potential = { always = yes }
    ai_frequency = 3
    ai_targets = {
        ai_recipients = spouses
        ai_recipients = children
        ai_recipients = liege
        ai_recipients = councillors
        ai_recipients = family
        ai_recipients = dynasty
        ai_recipients = courtiers
        ai_recipients = vassals
    }
    ai_will_do = {
        base = 0

        # Score send options
        modifier = {
            scope:mage_major_initiation = yes
            add = 50
        }
        modifier = {
            scope:mage_minor_initiation = yes
            add = 25
        }

        # Prioritisation hierarchy: spouse > consort > primary heir > children > liege > councillors > knights > dynasty
        modifier = {
            scope:recipient = { is_spouse_of = scope:actor }
            add = 1000
        }
        modifier = {
            scope:recipient = { is_consort_of = scope:actor }
            add = 500
        }
        modifier = {
            scope:recipient = { is_primary_heir_of = scope:actor }
            add = 800
        }
        modifier = {
            scope:recipient = { is_child_of = scope:actor }
            add = 100
        }
        modifier = {
            scope:recipient = { scope:actor.liege ?= this }
            add = 50
        }
        modifier = {
            scope:recipient = { is_knight_of = scope:actor }
            add = 35
        }
        modifier = {
            scope:recipient = { is_councillor_of = scope:actor }
            add = 25
        }
        modifier = {
            scope:recipient = {
                exists = dynasty
                exists = scope:actor.dynasty
                dynasty = scope:actor.dynasty
            }
            add = 25
        }

        bm_blood_mage_prevalence_ai_acquisition_modifier = yes
        ai_willingness_to_do_positive_magic = yes

        # Absolute veto: must select at least one valid send option
        modifier = {
            NOR = {
                scope:mage_minor_initiation = yes
                scope:mage_major_initiation = yes
            }
            factor = 0
        }
    }
}
