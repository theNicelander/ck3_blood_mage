grant_blood_magic_interaction = {
    category = interaction_category_friendly
    auto_accept = yes
    common_interaction = yes
    desc = grant_blood_magic_interaction_desc
    interface_priority = 200

    is_highlighted = { always = yes }

    is_shown = {
        NOT = { scope:actor = scope:recipient }
        scope:actor = { has_trait = lifestyle_blood_mage }
        scope:recipient = { NOT = { has_trait = lifestyle_blood_mage } }
    }
    cost = { piety = 75 }
    is_valid_showing_failures_only = {
        scope:actor = { has_character_modifier = lifeforce_modifier_major }
    }

    on_accept = {
        debug_log = "BloodMageInteraction: grant_blood_magic | Actor: [actor.GetShortUINameNoTooltip] (ID:[actor.GetID], [Select_CString(actor.IsPlayer, 'Player', 'AI')]) -> Recipient: [recipient.GetShortUINameNoTooltip] (ID:[recipient.GetID], [Select_CString(recipient.IsPlayer, 'Player', 'AI')]) | Rel: [actor.Custom2('RelationToMeShort', recipient)] | Tag: [actor.Custom2('bm_debug_relation_tags', recipient)] | Opinion: [actor.GetOpinionOf( recipient )]"
        scope:recipient = {
            bm_become_blood_mage_effect = yes
            add_opinion = {
                target = scope:actor
                modifier = magic_opinion_positive
                opinion = 50
            }
        }
        scope:actor = { benediction_magic_cast_effect_major = yes }
    }

    # AI
    ai_potential = { always = yes }
    ai_frequency = 6
    ai_targets = {
        ai_recipients = spouses
        ai_recipients = children
        ai_recipients = liege
        ai_recipients = councillors
        ai_recipients = family
        ai_recipients = dynasty
        ai_recipients = courtiers
    }
    ai_will_do = {
        base = 0
        ai_willingness_to_do_positive_magic = yes

        # Prioritisation hierarchy: spouse > children > liege > councillors > knights > dynasty
        modifier = {
            scope:recipient = { is_spouse_of = scope:actor }
            add = 100
        }
        modifier = {
            scope:recipient = { is_child_of = scope:actor }
            add = 75
        }
        modifier = {
            scope:recipient = { scope:actor.liege ?= this }
            add = 50
        }
        modifier = {
            scope:recipient = { is_knight_of = scope:actor }
            add = 35
        }
        modifier = {
            scope:recipient = { is_councillor_of = scope:actor }
            add = 25
        }
        modifier = {
            scope:recipient = {
                exists = dynasty
                exists = scope:actor.dynasty
                dynasty = scope:actor.dynasty
            }
            add = 25
        }

        bm_blood_mage_prevalence_ai_acquisition_modifier = yes
    }
}
