bm_feingur_bless_dynasty_interaction = {
	category = interaction_category_bm_blood_magic
	auto_accept = yes
	common_interaction = yes
	desc = bm_feingur_bless_dynasty_interaction_desc
	interface_priority = 196

	is_highlighted = {
		scope:recipient = { is_child_of = scope:actor }
	}

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = lifestyle_blood_mage
				has_trait = lifestyle_blood_knight
			}
			bm_has_superior_stave_trigger = { STAVE = feingur }
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			OR = {
				is_child_of = scope:actor
				dynasty ?= scope:actor.dynasty
			}
		}
	}

	cost = { piety = 50 }

	is_valid_showing_failures_only = {
		scope:actor = { has_character_modifier = lifeforce_modifier_minor }
		custom_description = {
			text = "bm_recipient_already_feingur_blessed"
			NOT = { scope:recipient = { has_character_modifier = bm_feingur_blessed_blood_modifier } }
		}
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_feingur_bless_dynasty_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				bloodline_magic_cast_effect_minor = yes
			}
		}
		scope:recipient = {
			add_character_modifier = {
				modifier = bm_feingur_blessed_blood_modifier
			}
			add_opinion = {
				target = scope:actor
				modifier = magic_opinion_positive
				opinion = 40
			}
		}
	}

	ai_targets = {
		ai_recipients = children
		ai_recipients = dynasty
	}
	ai_potential = { always = yes }
	ai_frequency = 12
	ai_will_do = {
		base = 15
		ai_willingness_to_do_positive_magic = yes

		modifier = {
			scope:recipient = { is_child_of = scope:actor }
			add = 50
		}
		modifier = {
			scope:recipient = { is_primary_heir_of = scope:actor }
			add = 50
		}
		modifier = {
			scope:actor = {
				NOT = { has_character_modifier = lifeforce_modifier_minor }
			}
			factor = 0
		}
	}
}
