bm_action_can_heal = {
	combine_into_one = yes
	priority = 350

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				OR = {
					has_trait = lifestyle_blood_mage
					has_trait = lifestyle_blood_knight
				}
			}
			save_scope_as = actor

			# 1. Self
			if = {
				limit = {
					bm_can_heal_character_trigger = yes
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 2. Spouses
			every_spouse = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 3. Children
			every_child = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 4. Lovers and Soulmates
			every_relation = {
				type = lover
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}
			every_relation = {
				type = soulmate
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 5. Friends and Best Friends
			every_relation = {
				type = friend
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}
			every_relation = {
				type = best_friend
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 6. Liege
			if = {
				limit = {
					exists = liege
					liege != root
					liege = {
						bm_can_heal_character_trigger = yes
						NOT = { is_in_list = bm_heal_candidates }
					}
				}
				liege = {
					add_to_temporary_list = bm_heal_candidates
				}
			}

			# 7. Councillors
			every_councillor = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 8. Knights
			every_knight = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# Create important actions for candidate characters
			every_in_list = {
				list = bm_heal_candidates
				try_create_important_action = {
					important_action_type = bm_action_can_heal
					actor = scope:actor
					recipient = this
				}
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = bm_heal_ailment
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
