bm_action_can_heal = {
	icon = bm_blood_magic
	combine_into_one = yes
	priority = 350

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				is_blood_mage_any_tier_trigger = yes
			}
			save_scope_as = actor

			# 1. Self
			if = {
				limit = {
					bm_can_heal_character_trigger = yes
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 2. Spouses
			every_spouse = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 3. Children
			every_child = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 4. Close Family
			every_close_family_member = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 5. Lovers and Soulmates
			every_relation = {
				type = lover
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}
			every_relation = {
				type = soulmate
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 5. Friends and Best Friends
			every_relation = {
				type = friend
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}
			every_relation = {
				type = best_friend
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 6. Liege
			if = {
				limit = {
					exists = liege
					liege != root
					liege = {
						bm_can_heal_character_trigger = yes
						NOT = { is_in_list = bm_heal_candidates }
					}
				}
				liege = {
					add_to_temporary_list = bm_heal_candidates
				}
			}

			# 8. Councillors
			every_councillor = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 9. Court Positions
			every_court_position_holder = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# 10. Knights
			every_knight = {
				limit = {
					bm_can_heal_character_trigger = yes
					NOT = { is_in_list = bm_heal_candidates }
				}
				add_to_temporary_list = bm_heal_candidates
			}

			# Create important actions for candidate characters
			every_in_list = {
				list = bm_heal_candidates
				try_create_important_action = {
					important_action_type = bm_action_can_heal
					actor = scope:actor
					recipient = this
				}
			}
		}
	}

	unimportant = {
		NOT = {
			root = {
				bm_actor_has_lifeforce_to_heal_target_trigger = { TARGET = scope:recipient }
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = bm_heal_ailment
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}

bm_action_can_lifedrain_prisoner = {
	icon = bm_blood_magic
	combine_into_one = yes
	priority = 340

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				is_blood_mage_any_tier_trigger = yes
			}
			save_scope_as = actor

			every_prisoner = {
				limit = {
					save_temporary_scope_as = prisoner_target

						is_character_interaction_valid = {
							interaction = bm_blood_drain_interaction
							recipient = scope:prisoner_target
						}
					}
				}
				add_to_temporary_list = bm_lifedrain_candidates
			}

			every_in_list = {
				list = bm_lifedrain_candidates

					important_action_type = bm_action_can_lifedrain_prisoner
					actor = scope:actor
					recipient = this
				}
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = bm_blood_drain_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
