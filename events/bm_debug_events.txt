namespace = bm_debug

# 1. Debug: Experience Menu
bm_debug.0001 = {
	type = character_event
	title = bm_debug.0001.title
	desc = bm_debug.0001.desc
	theme = witchcraft

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Trait (20xp at a time)
	option = {
		name = bm_debug.0001.trait_xp
		send_interface_toast = {
			title = bm_debug.0001.toast
			left_icon = root
			if = {
				limit = { NOT = { is_blood_mage_any_tier_trigger = yes } }
				bm_become_blood_acolyte_effect = yes
			}
			bm_add_blood_mage_xp_effect = { TRACK = ancient XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = enlightenment XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = bloodline XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = benediction XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = hematurgy XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = bloodlust XP = 20 }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# Blood empowerment (20xp at a time)
	option = {
		name = bm_debug.0001.empowerment_xp
		send_interface_toast = {
			title = bm_debug.0001.toast
			left_icon = root
			if = {
				limit = { NOT = { has_trait = lifestyle_blood_power } }
				add_trait = lifestyle_blood_power
			}
			bm_add_blood_power_xp_effect = { TRACK = dynasty XP = 20 }
			bm_add_blood_power_xp_effect = { TRACK = warband XP = 20 }
			bm_add_blood_power_xp_effect = { TRACK = mastery XP = 20 }
			bm_add_blood_power_xp_effect = { TRACK = presence XP = 20 }
			bm_add_blood_power_xp_effect = { TRACK = prosperity XP = 20 }
			bm_add_blood_power_xp_effect = { TRACK = shadows XP = 20 }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 100 Ancient XP
	option = {
		name = bm_debug.0001.ancient_xp
		send_interface_toast = {
			title = bm_debug.0001.toast
			left_icon = root
			if = {
				limit = { NOT = { is_blood_mage_any_tier_trigger = yes } }
				bm_become_blood_acolyte_effect = yes
			}
			bm_add_blood_mage_xp_effect = { TRACK = ancient XP = 100 }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# Back
	option = {
		name = bm_debug.back
	}
}

# 2. Debug: Lifeforce Menu
bm_debug.0002 = {
	type = character_event
	title = bm_debug.0002.title
	desc = bm_debug.0002.desc
	theme = witchcraft

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Add Superior x5
	option = {
		name = bm_debug.0002.add_superior_5
		send_interface_toast = {
			title = bm_debug.0002.toast
			left_icon = root
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
		trigger_event = { id = bm_debug.0002 }
	}

	# Add major x5
	option = {
		name = bm_debug.0002.add_major_5
		send_interface_toast = {
			title = bm_debug.0002.toast
			left_icon = root
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
		trigger_event = { id = bm_debug.0002 }
	}

	# Add minor x5
	option = {
		name = bm_debug.0002.add_minor_5
		send_interface_toast = {
			title = bm_debug.0002.toast
			left_icon = root
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
		trigger_event = { id = bm_debug.0002 }
	}

	# Remove all lifeforces
	option = {
		name = bm_debug.0002.remove_all
		send_interface_toast = {
			title = bm_debug.0002.toast
			left_icon = root
			while = {
				limit = { has_character_modifier = lifeforce_modifier_superior }
				remove_character_modifier = lifeforce_modifier_superior
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_major }
				remove_character_modifier = lifeforce_modifier_major
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_minor }
				remove_character_modifier = lifeforce_modifier_minor
			}
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
		trigger_event = { id = bm_debug.0002 }
	}

	# Back
	option = {
		name = bm_debug.back
	}
}
