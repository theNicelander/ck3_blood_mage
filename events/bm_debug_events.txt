namespace = bm_debug

# Debug Menu (all debug actions in one persistent event)
bm_debug.0001 = {
	type = character_event
	title = bm_debug.0001.title
	desc = bm_debug.0001.desc
	theme = witchcraft

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# 1. Become Blood Mage
	option = {
		name = bm_debug.0001.become_blood_mage
		trigger = {
			NOT = { is_blood_mage_any_tier_trigger = yes }
		}
		send_interface_toast = {
			title = bm_debug.0001.become_blood_mage_toast
			left_icon = root
			bm_become_blood_mage_effect = yes
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 2. Give Superior Blood Rune
	option = {
		name = bm_debug.0001.superior_rune
		send_interface_toast = {
			title = bm_debug.0001.superior_rune_toast
			left_icon = root
			remove_character_modifier = minor_blood_rune_modifier
			remove_character_modifier = major_blood_rune_modifier
			add_character_modifier = { modifier = superior_blood_rune_modifier }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 3. Upgrade All Covenants
	option = {
		name = bm_debug.0001.upgrade_covenants
		trigger = {
			exists = house
		}
		send_interface_toast = {
			title = bm_debug.0001.upgrade_covenants_toast
			left_icon = root
			house = {
				# Radiance
				if = {
					limit = { has_house_modifier = dynasty_blood_radiance_modifier_2 }
					remove_house_modifier = dynasty_blood_radiance_modifier_2
					add_house_modifier = { modifier = dynasty_blood_radiance_modifier_3 }
				}
				else_if = {
					limit = { has_house_modifier = dynasty_blood_radiance_modifier_1 }
					remove_house_modifier = dynasty_blood_radiance_modifier_1
					add_house_modifier = { modifier = dynasty_blood_radiance_modifier_2 }
				}
				else_if = {
					limit = { NOT = { has_house_modifier = dynasty_blood_radiance_modifier_3 } }
					add_house_modifier = { modifier = dynasty_blood_radiance_modifier_1 }
				}

				# Fury
				if = {
					limit = { has_house_modifier = dynasty_blood_fury_modifier_2 }
					remove_house_modifier = dynasty_blood_fury_modifier_2
					add_house_modifier = { modifier = dynasty_blood_fury_modifier_3 }
				}
				else_if = {
					limit = { has_house_modifier = dynasty_blood_fury_modifier_1 }
					remove_house_modifier = dynasty_blood_fury_modifier_1
					add_house_modifier = { modifier = dynasty_blood_fury_modifier_2 }
				}
				else_if = {
					limit = { NOT = { has_house_modifier = dynasty_blood_fury_modifier_3 } }
					add_house_modifier = { modifier = dynasty_blood_fury_modifier_1 }
				}

				# Bloodline
				if = {
					limit = { has_house_modifier = dynasty_blood_fortune_modifier_2 }
					remove_house_modifier = dynasty_blood_fortune_modifier_2
					add_house_modifier = { modifier = dynasty_blood_fortune_modifier_3 }
				}
				else_if = {
					limit = { has_house_modifier = dynasty_blood_fortune_modifier_1 }
					remove_house_modifier = dynasty_blood_fortune_modifier_1
					add_house_modifier = { modifier = dynasty_blood_fortune_modifier_2 }
				}
				else_if = {
					limit = { NOT = { has_house_modifier = dynasty_blood_fortune_modifier_3 } }
					add_house_modifier = { modifier = dynasty_blood_fortune_modifier_1 }
				}

				# Cunning
				if = {
					limit = { has_house_modifier = dynasty_blood_cunning_modifier_2 }
					remove_house_modifier = dynasty_blood_cunning_modifier_2
					add_house_modifier = { modifier = dynasty_blood_cunning_modifier_3 }
				}
				else_if = {
					limit = { has_house_modifier = dynasty_blood_cunning_modifier_1 }
					remove_house_modifier = dynasty_blood_cunning_modifier_1
					add_house_modifier = { modifier = dynasty_blood_cunning_modifier_2 }
				}
				else_if = {
					limit = { NOT = { has_house_modifier = dynasty_blood_cunning_modifier_3 } }
					add_house_modifier = { modifier = dynasty_blood_cunning_modifier_1 }
				}

				# Wisdom
				if = {
					limit = { has_house_modifier = dynasty_blood_wisdom_modifier_2 }
					remove_house_modifier = dynasty_blood_wisdom_modifier_2
					add_house_modifier = { modifier = dynasty_blood_wisdom_modifier_3 }
				}
				else_if = {
					limit = { has_house_modifier = dynasty_blood_wisdom_modifier_1 }
					remove_house_modifier = dynasty_blood_wisdom_modifier_1
					add_house_modifier = { modifier = dynasty_blood_wisdom_modifier_2 }
				}
				else_if = {
					limit = { NOT = { has_house_modifier = dynasty_blood_wisdom_modifier_3 } }
					add_house_modifier = { modifier = dynasty_blood_wisdom_modifier_1 }
				}

				# Ancient
				if = {
					limit = { has_house_modifier = dynasty_blood_ancient_modifier_2 }
					remove_house_modifier = dynasty_blood_ancient_modifier_2
					add_house_modifier = { modifier = dynasty_blood_ancient_modifier_3 }
				}
				else_if = {
					limit = { has_house_modifier = dynasty_blood_ancient_modifier_1 }
					remove_house_modifier = dynasty_blood_ancient_modifier_1
					add_house_modifier = { modifier = dynasty_blood_ancient_modifier_2 }
				}
				else_if = {
					limit = { NOT = { has_house_modifier = dynasty_blood_ancient_modifier_3 } }
					add_house_modifier = { modifier = dynasty_blood_ancient_modifier_1 }
				}
			}
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 4. Trait (20 XP)
	option = {
		name = bm_debug.0001.trait_xp
		send_interface_toast = {
			title = bm_debug.0001.xp_toast
			left_icon = root
			if = {
				limit = { NOT = { is_blood_mage_any_tier_trigger = yes } }
				bm_become_blood_acolyte_effect = yes
			}
			bm_add_blood_mage_xp_effect = { TRACK = ancient XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = wisdom XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = fortune XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = radiance XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = cunning XP = 20 }
			bm_add_blood_mage_xp_effect = { TRACK = fury XP = 20 }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 5. 100 Ancient XP
	option = {
		name = bm_debug.0001.ancient_xp
		send_interface_toast = {
			title = bm_debug.0001.xp_toast
			left_icon = root
			if = {
				limit = { NOT = { is_blood_mage_any_tier_trigger = yes } }
				bm_become_blood_acolyte_effect = yes
			}
			bm_add_blood_mage_xp_effect = { TRACK = ancient XP = 100 }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 6. Add Superior x5
	option = {
		name = bm_debug.0001.add_superior_5
		send_interface_toast = {
			title = bm_debug.0001.lifeforce_toast
			left_icon = root
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
			add_character_modifier = { modifier = lifeforce_modifier_superior }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 7. Add Major x5
	option = {
		name = bm_debug.0001.add_major_5
		send_interface_toast = {
			title = bm_debug.0001.lifeforce_toast
			left_icon = root
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
			add_character_modifier = { modifier = lifeforce_modifier_major }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 8. Add Minor x5
	option = {
		name = bm_debug.0001.add_minor_5
		send_interface_toast = {
			title = bm_debug.0001.lifeforce_toast
			left_icon = root
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
			add_character_modifier = { modifier = lifeforce_modifier_minor }
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# 9. Remove All Lifeforce
	option = {
		name = bm_debug.0001.remove_all_lifeforce
		send_interface_toast = {
			title = bm_debug.0001.lifeforce_toast
			left_icon = root
			while = {
				limit = { has_character_modifier = lifeforce_modifier_superior }
				remove_character_modifier = lifeforce_modifier_superior
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_major }
				remove_character_modifier = lifeforce_modifier_major
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_minor }
				remove_character_modifier = lifeforce_modifier_minor
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_radiant }
				remove_character_modifier = lifeforce_modifier_radiant
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_furious }
				remove_character_modifier = lifeforce_modifier_furious
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_fortunate }
				remove_character_modifier = lifeforce_modifier_fortunate
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_cunning }
				remove_character_modifier = lifeforce_modifier_cunning
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_wise }
				remove_character_modifier = lifeforce_modifier_wise
			}
			while = {
				limit = { has_character_modifier = lifeforce_modifier_ancient }
				remove_character_modifier = lifeforce_modifier_ancient
			}
		}
		trigger_event = { id = bm_debug.0001 }
	}

	# Back
	option = {
		name = bm_debug.back
	}
}
