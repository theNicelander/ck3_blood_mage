namespace = bm_cast_blood_magic_minor

bm_cast_blood_magic_minor.001 = {
	type = character_event
	title = bm_cast_blood_magic_minor.001.title
	desc = bm_cast_blood_magic_minor.001.desc
	theme = witchcraft

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Channel Minor Lifeforce
	option = {
		name = bm_cast_blood_magic_minor.001.channel
		trigger = {
			piety >= 75
		}
		show_as_unavailable = {
			always = yes
		}
		add_piety = -75
		remove_character_modifier = lifeforce_modifier_minor
		if = {
			limit = { has_trait = lifestyle_blood_mage }
			add_xp_enlightenment = yes
		}
		if = {
			limit = {
				has_trait = lifestyle_blood_knight
				NOT = { has_trait = lifestyle_blood_mage }
			}
			add_blood_knight_xp = { TRACK = resilience XP = 1 }
		}
		custom_tooltip = channel_lifeforce_enlightenment_minor.effects_list
		trigger_event = { id = bm_channel_lifeforce_enlightenment_minor.001 }
		ai_chance = {
			base = 10
			modifier = {
				add = -100
				health <= fine_health
			}
		}
	}

	# Attune Lifeforce
	option = {
		name = bm_cast_blood_magic_minor.001.attune
		trigger = {
			piety >= 25
		}
		show_as_unavailable = {
			always = yes
		}
		add_piety = -25
		remove_character_modifier = lifeforce_modifier_minor
		custom_tooltip = bm_attune_lifeforce.effects_list
		trigger_event = { id = bm_attune_lifeforce.001 }
		ai_chance = {
			base = 10
			modifier = {
				add = -100
				health <= fine_health
			}
			modifier = {
				add = 100
				NOT = {
					has_character_modifier = ancient_attuned
					has_character_modifier = enlightenment_attuned
					has_character_modifier = bloodline_attuned
					has_character_modifier = benediction_attuned
					has_character_modifier = hematurgy_attuned
					has_character_modifier = vanguard_attuned
					has_character_modifier = slaughter_attuned
					has_character_modifier = resilience_attuned
				}
			}
		}
	}

	# Purge Bodily Impurities
	option = {
		name = bm_cast_blood_magic_minor.001.purge
		trigger = {
			piety >= 50
			OR = {
				has_trait = wounded_1
				has_trait = ill
				has_trait = lovers_pox
				has_trait = gout_ridden
				has_trait = scarred
				has_trait = beauty_bad
				has_trait = intellect_bad
				has_trait = physique_bad
				has_trait = dull
				has_trait = weak
			}
		}
		add_piety = -50
		remove_character_modifier = lifeforce_modifier_minor

		if = {
			limit = { has_trait = lifestyle_blood_mage }
			add_xp_bm_dynamic = {
				TRACK = benediction
				XP = 2
			}
		}
		else = {
			add_blood_knight_xp = {
				TRACK = benediction
				XP = 2
			}
			add_blood_knight_xp = {
				TRACK = resilience
				XP = 1
			}
		}

		# Cleanse traits
		remove_trait = wounded_1
		remove_trait = ill
		remove_trait = lovers_pox
		remove_trait = gout_ridden
		remove_trait = scarred

		remove_trait = beauty_bad_1
		remove_trait = beauty_bad_2
		remove_trait = beauty_bad_3
		remove_trait = beauty_bad

		remove_trait = intellect_bad_1
		remove_trait = intellect_bad_2
		remove_trait = intellect_bad_3
		remove_trait = intellect_bad
		remove_trait = dull

		remove_trait = physique_bad_1
		remove_trait = physique_bad_2
		remove_trait = physique_bad_3
		remove_trait = physique_bad
		remove_trait = weak

		ai_chance = {
			base = 50
			modifier = {
				add = 50
				OR = {
					has_trait = beauty_bad
					has_trait = intellect_bad
					has_trait = physique_bad
				}
			}
		}
	}

	# Cancel
	option = {
		name = bm_cast_blood_magic_minor.001.cancel
		trigger = {
			is_ai = no
		}
	}
}
