namespace = bm_cast_blood_magic_minor

bm_cast_blood_magic_minor.001 = {
	type = character_event
	title = bm_cast_blood_magic_minor.001.title
	desc = bm_cast_blood_magic_minor.001.desc
	theme = witchcraft

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Channel Minor Lifeforce
	option = {
		name = bm_cast_blood_magic_minor.001.channel
		trigger = {
			piety >= 75
		}
		show_as_unavailable = {
			always = yes
		}
		add_piety = -75
		remove_character_modifier = lifeforce_modifier_minor
		if = {
			limit = { has_trait = lifestyle_blood_mage }
			add_xp_enlightenment = yes
		}
		if = {
			limit = {
				has_trait = lifestyle_blood_knight
				NOT = { has_trait = lifestyle_blood_mage }
			}
			add_blood_knight_xp = { TRACK = enlightenment XP = 1 }
		}
		custom_tooltip = channel_lifeforce_enlightenment_minor.effects_list
		trigger_event = { id = bm_channel_lifeforce_enlightenment_minor.001 }
		ai_chance = {
			base = 10
			modifier = {
				add = -100
				health <= fine_health
			}
		}
	}

	# Attune Lifeforce
	option = {
		name = bm_cast_blood_magic_minor.001.attune
		trigger = {
			piety >= 25
		}
		show_as_unavailable = {
			always = yes
		}
		add_piety = -25
		remove_character_modifier = lifeforce_modifier_minor
		custom_tooltip = bm_attune_lifeforce.effects_list
		trigger_event = { id = bm_attune_lifeforce.001 }
		ai_chance = {
			base = 10
			modifier = {
				add = -100
				health <= fine_health
			}
			modifier = {
				add = 100
				NOT = {
					has_character_modifier = ancient_attuned
					has_character_modifier = enlightenment_attuned
					has_character_modifier = bloodline_attuned
					has_character_modifier = benediction_attuned
					has_character_modifier = hematurgy_attuned
					has_character_modifier = vanguard_attuned
					has_character_modifier = slaughter_attuned
				}
			}
		}
	}

	# Commune with Bloodline
	option = {
		name = bm_cast_blood_magic_minor.001.commune_bloodline
		trigger = {
			has_trait = lifestyle_blood_mage
			exists = house
			piety >= 50
			NOT = { has_character_flag = bm_commune_bloodline_cooldown }
		}
		show_as_unavailable = {
			has_trait = lifestyle_blood_mage
			exists = house
		}
		add_piety = -50
		add_character_flag = {
			flag = bm_commune_bloodline_cooldown
			years = 3
		}
		bloodline_magic_cast_effect_minor = yes
		add_character_modifier = {
			modifier = bm_bloodline_communion
			years = 5
		}
		custom_tooltip = bm_commune_bloodline.effects_list
		ai_chance = {
			base = 25
			modifier = {
				add = -100
				health <= fine_health
			}
		}
	}

	# Purge Bodily Impurities
	option = {
		name = bm_cast_blood_magic_minor.001.purge
		trigger = {
			piety >= 50
			OR = {
				has_trait = wounded_1
				has_trait = ill
				has_trait = lovers_pox
				has_trait = gout_ridden
				has_trait = scarred
				has_trait = beauty_bad
				has_trait = intellect_bad
				has_trait = physique_bad
				has_trait = dull
				has_trait = weak
			}
		}
		add_piety = -50
		remove_character_modifier = lifeforce_modifier_minor

		if = {
			limit = { has_trait = lifestyle_blood_mage }
			if = {
				limit = {
					OR = {
						has_trait = beauty_bad
						has_trait = intellect_bad
						has_trait = physique_bad
						has_trait = dull
						has_trait = weak
					}
				}
				add_xp_bm_dynamic = {
					TRACK = bloodline
					XP = 2
				}
			}
			else = {
				add_xp_bm_dynamic = {
					TRACK = benediction
					XP = 2
				}
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						has_trait = physique_bad
						has_trait = weak
					}
				}
				add_blood_knight_xp = {
					TRACK = enlightenment
					XP = 2
				}
			}
			else = {
				add_blood_knight_xp = {
					TRACK = benediction
					XP = 1
				}
				add_blood_knight_xp = {
					TRACK = enlightenment
					XP = 1
				}
			}
		}

		# Cleanse traits
		remove_trait = wounded_1
		remove_trait = ill
		remove_trait = lovers_pox
		remove_trait = gout_ridden
		remove_trait = scarred

		remove_trait = beauty_bad_1
		remove_trait = beauty_bad_2
		remove_trait = beauty_bad_3
		remove_trait = beauty_bad

		remove_trait = intellect_bad_1
		remove_trait = intellect_bad_2
		remove_trait = intellect_bad_3
		remove_trait = intellect_bad
		remove_trait = dull

		remove_trait = physique_bad_1
		remove_trait = physique_bad_2
		remove_trait = physique_bad_3
		remove_trait = physique_bad
		remove_trait = weak

		ai_chance = {
			base = 50
			modifier = {
				add = 50
				OR = {
					has_trait = beauty_bad
					has_trait = intellect_bad
					has_trait = physique_bad
				}
			}
		}
	}

	# Condense Lifeforce (1 Minor Lifeforce + 100 Piety -> 1 Major Lifeforce)
	option = {
		name = bm_cast_blood_magic_minor.001.condense
		trigger = {
			has_character_modifier = lifeforce_modifier_minor
			piety >= 100
		}
		show_as_unavailable = {
			has_character_modifier = lifeforce_modifier_minor
		}
		add_piety = -100
		remove_character_modifier = lifeforce_modifier_minor
		add_character_modifier = { modifier = lifeforce_modifier_major }
		if = {
			limit = { has_trait = lifestyle_blood_mage }
			add_xp_bm_dynamic = {
				TRACK = ancient
				XP = 2
			}
		}
		else = {
			add_blood_knight_xp = {
				TRACK = ancient
				XP = 2
			}
		}
		hidden_effect = {
			bm_recount_lifeforce_stacks_effect = yes
		}
		custom_tooltip = bm_cast_blood_magic_minor.001.condense.tt
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				has_character_modifier = lifeforce_modifier_minor
				NOT = { has_character_modifier = lifeforce_modifier_major }
			}
		}
	}

	# Cancel
	option = {
		name = bm_cast_blood_magic_minor.001.cancel
		trigger = {
			is_ai = no
		}
	}
}
