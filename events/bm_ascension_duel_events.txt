namespace = bm_ascension_duel

bm_ascension_duel.0001 = {
	type = character_event
	title = bm_ascension_duel.0001.t
	desc = bm_ascension_duel.0001.desc
	theme = war

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		create_character = {
			template = bm_crimson_reflection_template
			gender = root.gender
			age = root.age
			faith = root.faith
			culture = root.culture
			save_scope_as = bm_clone
		}
		scope:bm_clone = {
			copy_inheritable_appearance_from = root
			set_variable = {
				name = bm_target_prowess
				value = root.prowess
			}
			change_variable = {
				name = bm_target_prowess
				subtract = prowess
			}
			if = {
				limit = { var:bm_target_prowess > 0 }
				add_prowess_skill = var:bm_target_prowess
			}
			force_character_skill_recalculation = yes
		}
	}

	right_portrait = {
		character = scope:bm_clone
		animation = rage
	}

	option = {
		name = bm_ascension_duel.0001.opt
		configure_start_single_combat_effect = {
			SC_INITIATOR = root
			SC_ATTACKER = root
			SC_DEFENDER = scope:bm_clone
			FATALITY = practice
			FIXED = no
			LOCALE = battlefield
			OUTPUT_EVENT = bm_ascension_duel.0002
			INVALIDATION_EVENT = bm_ascension_duel.0003
		}
	}
}

bm_ascension_duel.0002 = {
	type = character_event
	title = bm_ascension_duel.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					this = scope:sc_victor
				}
				desc = bm_ascension_duel.0002.victor_desc
			}
			desc = bm_ascension_duel.0002.loser_desc
		}
	}
	theme = war

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:sc_defender
		animation = martial
	}

	immediate = {
		if = {
			limit = { this = scope:sc_victor }
			bm_elevate_tier_effect = yes
		}
		else = {
			add_character_modifier = {
				modifier = bm_casting_exhaustion_modifier
				years = 1
			}
			if = {
				limit = { NOT = { has_trait = wounded } }
				add_trait = wounded_1
			}
			else_if = {
				limit = { has_trait = wounded_1 }
				remove_trait = wounded_1
				add_trait = wounded_2
			}
			if = {
				limit = { has_trait = lifestyle_bm_acolyte }
				bm_add_blood_mage_xp_effect = {
					TRACK = ancient
					XP = -10
				}
			}
			else_if = {
				limit = { has_trait = lifestyle_blood_mage }
				bm_add_blood_mage_xp_effect = {
					TRACK = ancient
					XP = -10
				}
			}
			send_interface_toast = {
				type = event_toast_effect_bad
				title = bm_ascension_duel_loser_toast
				left_icon = root
				right_icon = scope:sc_defender
				custom_tooltip = bm_ascension_duel_loser_tt
			}
		}
		if = {
			limit = { exists = scope:sc_defender }
			scope:sc_defender = {
				death = {
					death_reason = death_vanished
				}
			}
		}
	}

	option = {
		name = bm_ascension_duel.0002.opt
	}
}
