namespace = bm_crimson_warrior_event

bm_crimson_warrior_event.0001 = {
	type = character_event
	title = bm_crimson_warrior_event.0001.t
	desc = bm_crimson_warrior_event.0001.desc
	theme = martial

	left_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		character = scope:recipient
		animation = rage
	}

	# Option 1: Vanguard (+3 XP)
	option = {
		name = bm_crimson_warrior_event.vanguard
		trigger = {
			scope:recipient = {
				has_trait_xp = {
					trait = lifestyle_crimson_warrior
					track = vanguard
					value < 100
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				scope:recipient = { martial >= 14 }
			}
		}
		scope:recipient = {
			add_crimson_warrior_xp = { TRACK = vanguard XP = 3 }
		}
	}

	# Option 2: Slaughter (+3 XP)
	option = {
		name = bm_crimson_warrior_event.slaughter
		trigger = {
			scope:recipient = {
				has_trait_xp = {
					trait = lifestyle_crimson_warrior
					track = slaughter
					value < 100
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				scope:recipient = { prowess >= 14 }
			}
		}
		scope:recipient = {
			add_crimson_warrior_xp = { TRACK = slaughter XP = 3 }
		}
	}

	# Option 3: Resilience (+3 XP)
	option = {
		name = bm_crimson_warrior_event.resilience
		trigger = {
			scope:recipient = {
				has_trait_xp = {
					trait = lifestyle_crimson_warrior
					track = resilience
					value < 100
				}
			}
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 50
				scope:recipient = { health <= fine_health }
			}
		}
		scope:recipient = {
			add_crimson_warrior_xp = { TRACK = resilience XP = 3 }
		}
	}

	# Fallback if all maxed
	option = {
		name = bm_crimson_warrior_event.all_maxed
		trigger = {
			scope:recipient = {
				has_trait_xp = { trait = lifestyle_crimson_warrior track = vanguard value >= 100 }
				has_trait_xp = { trait = lifestyle_crimson_warrior track = slaughter value >= 100 }
				has_trait_xp = { trait = lifestyle_crimson_warrior track = resilience value >= 100 }
			}
		}
		scope:recipient = {
			add_prowess = 1
		}
	}
}
