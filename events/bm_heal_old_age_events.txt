		show_as_unavailable = {

		}

			limit = {
				scope:recipient = {
					has_trait_xp = {

						track = faltering_heart
						value <= 50
					}
				}
			}

					remove_trait = faltering_heart
				}

				bm_heal_major_opinion_effect = yes
				hidden_effect = { bm_recount_lifeforce_stacks_effect = yes }
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_heal_old_age_toast_treatment_title
				left_icon = root
				right_icon = scope:recipient

					add_trait_xp = {
						trait = faltering_heart
						track = faltering_heart
						value = -50
					}
				}
				bm_heal_old_age_major_cost_effect = yes
				bm_heal_major_opinion_effect = yes
				hidden_effect = { bm_recount_lifeforce_stacks_effect = yes }
			}

			base = 80

				factor = 2
				scope:recipient = root
			}
		}
