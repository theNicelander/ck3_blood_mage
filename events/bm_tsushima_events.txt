# Blood Mage of Tsushima Duel Events

namespace = bm_tsushima

bm_tsushima.0001 = {
	type = character_event
	title = bm_tsushima.0001.title
	desc = bm_tsushima.0001.desc
	theme = learning

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:bm_tsushima_rival
		animation = personality_rational
	}

	immediate = {
		hidden_effect = {
			create_character = {
				age = { 35 45 }
				gender_female_chance = 50
				location = root.location
				culture = root.culture
				faith = faith:ketsudo_faith
				trait = lifestyle_blood_mage
				trait = education_learning_4
				learning = 22
				prowess = 18
				save_scope_as = bm_tsushima_rival
			}
		}
	}

	option = {
		name = bm_tsushima.0001.a

		duel = {
			skill = learning
			target = scope:bm_tsushima_rival

			# SUCCESS
			50 = {
				desc = bm_tsushima_blood_mage_duel_decision.success_tt

				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -45
				}
				modifier = {
					add = bm_tsushima_duel_prowess_bonus_value
				}

				send_interface_toast = {
					title = bm_tsushima_blood_mage_duel_decision.success_tt
					left_icon = root
					bm_become_blood_mage_effect = yes
				}
			}

			# FAILURE - MENTAL COLLAPSE
			25 = {
				desc = bm_tsushima_blood_mage_duel_decision.mental_failure_tt
				send_interface_toast = {
					title = bm_tsushima_blood_mage_duel_decision.mental_failure_tt
					left_icon = root
					add_stress = 30

					if = {
						limit = { has_trait = lunatic_1 }
						add_trait = dull
					}
					if = {
						limit = { has_trait = depressed_1 }
						add_trait = lunatic_1
					}
					if = {
						limit = { has_trait = possessed_1 }
						add_trait = depressed_1
					}
					add_trait = possessed_1
				}
			}

			# FAILURE - PHYSICAL TRAUMA
			25 = {
				desc = bm_tsushima_blood_mage_duel_decision.physical_failure_tt
				send_interface_toast = {
					title = bm_tsushima_blood_mage_duel_decision.physical_failure_tt
					left_icon = root
					add_stress = 20
					if = {
						limit = { has_trait = blind }
						add_piety_level = -1
					}
					if = {
						limit = { has_trait = one_eyed }
						add_trait = blind
					}
					if = {
						limit = { has_trait = weak }
						add_trait = one_eyed
					}
					add_trait = weak
				}
			}

			# FAILURE - INJURY
			50 = {
				desc = bm_tsushima_blood_mage_duel_decision.injury_failure_tt
				send_interface_toast = {
					title = bm_tsushima_blood_mage_duel_decision.injury_failure_tt
					left_icon = root
					add_stress = 10

					if = {
						limit = { has_trait = wounded_2 }
						add_trait = maimed
					}

					if = {
						limit = { NOT = { has_trait = wounded_1 } }
						add_trait = wounded_1
					}
					else = {
						change_trait_rank = {
							trait = wounded
							rank = 1
							max = 2
						}
					}
				}
			}
		}

		ai_chance = {
			base = 100
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = { exists = scope:bm_tsushima_rival }
				scope:bm_tsushima_rival = {
					death = { death_reason = death_vanished }
				}
			}
		}
	}
}
