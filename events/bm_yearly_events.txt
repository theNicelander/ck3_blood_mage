# Yearly event - Add xp to Blood Mages
blood_mage_yearly_events.001 = {
    hidden = yes
    trigger = { is_blood_mage_any_tier_trigger = yes }
    immediate = {
        # Base XP gain for all Blood Mages (silent, no toast)
        bm_add_blood_mage_xp_effect = {
            TRACK = ancient
            XP = 1
        }

        # Ancient attunement check
        if = {
            limit = { has_character_modifier = ancient_attuned }
            send_interface_toast = {
                title = bm_yearly_attunement_ancient_toast
                left_icon = root
                if = {
                    limit = { bm_has_superior_stave_trigger = { STAVE = hulinhjalmur } }
                    bm_add_blood_mage_xp_effect = {
				TRACK = ancient
				XP = 2
			}
                }
                else = {
                    bm_add_blood_mage_xp_effect = {
				TRACK = ancient
				XP = 1
			}
                }
            }

			}

        }

                title = bm_yearly_attunement_fortune_toast

blood_mage_yearly_events.002 = {
    hidden = yes
    trigger = {
        is_blood_mage_any_tier_trigger = yes
        exists = house
        OR = {
            AND = {
                NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = radiance VALUE = 100 } }
                house = {
                    OR = {
                        has_house_modifier = dynasty_blood_radiance_modifier_1
                        has_house_modifier = dynasty_blood_radiance_modifier_2
                        has_house_modifier = dynasty_blood_radiance_modifier_3
                    }
                }
            }
            AND = {
                NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = fury VALUE = 100 } }
                house = {
                    OR = {
                        has_house_modifier = dynasty_blood_fury_modifier_1
                        has_house_modifier = dynasty_blood_fury_modifier_2
                        has_house_modifier = dynasty_blood_fury_modifier_3
                    }
                }
            }
            AND = {
                NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = fortune VALUE = 100 } }
                house = {
                    OR = {
                        has_house_modifier = dynasty_blood_fortune_modifier_1
                        has_house_modifier = dynasty_blood_fortune_modifier_2
                        has_house_modifier = dynasty_blood_fortune_modifier_3
                    }
                }
            }
            AND = {
                NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = cunning VALUE = 100 } }
                house = {
                    OR = {
                        has_house_modifier = dynasty_blood_cunning_modifier_1
                        has_house_modifier = dynasty_blood_cunning_modifier_2
                        has_house_modifier = dynasty_blood_cunning_modifier_3
                    }
                }
            }
            AND = {
                NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = wisdom VALUE = 100 } }
                house = {
                    OR = {
                        has_house_modifier = dynasty_blood_wisdom_modifier_1

                }

            AND = {

                    OR = {
                        has_house_modifier = dynasty_blood_ancient_modifier_1
                        has_house_modifier = dynasty_blood_ancient_modifier_2
                        has_house_modifier = dynasty_blood_ancient_modifier_3
                    }
                }
            }
        }
    }
    immediate = {
        send_interface_toast = {
            title = blood_mage_yearly_events.002.success
            left_icon = root
            # 1. Radiance
            if = {
                limit = { NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = radiance VALUE = 100 } } }
                if = {
                    limit = { house = { has_house_modifier = dynasty_blood_radiance_modifier_3 } }
                    bm_add_blood_mage_xp_effect = { TRACK = radiance XP = 3 }
                }
                else_if = {
                    limit = { house = { has_house_modifier = dynasty_blood_radiance_modifier_2 } }
                    bm_add_blood_mage_xp_effect = { TRACK = radiance XP = 2 }
                }
                else_if = {
                    limit = { house = { has_house_modifier = dynasty_blood_radiance_modifier_1 } }
                    bm_add_blood_mage_xp_effect = { TRACK = radiance XP = 1 }
                }
            }
            # 2. Fury
            if = {
                limit = { NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = fury VALUE = 100 } } }
                if = {
                    limit = { house = { has_house_modifier = dynasty_blood_fury_modifier_3 } }
                    bm_add_blood_mage_xp_effect = { TRACK = fury XP = 3 }
                }
                else_if = {
                    limit = { house = { has_house_modifier = dynasty_blood_fury_modifier_2 } }

                }

                    bm_add_blood_mage_xp_effect = { TRACK = fury XP = 1 }
                }
            }
            # 3. Fortune
            if = {
                limit = { NOT = { bm_has_track_xp_any_tier_trigger = { TRACK = fortune VALUE = 100 } } }
                if = {
                    limit = { house = { has_house_modifier = dynasty_blood_fortune_modifier_3 } }
                    bm_add_blood_mage_xp_effect = { TRACK = fortune XP = 3 }
                }
                else_if = {
                    limit = { house = { has_house_modifier = dynasty_blood_fortune_modifier_2 } }
                    bm_add_blood_mage_xp_effect = { TRACK = fortune XP = 2 }
                }
                else_if = {
                    limit = { house = { has_house_modifier = dynasty_blood_fortune_modifier_1 } }
                    bm_add_blood_mage_xp_effect = { TRACK = fortune XP = 1 }

blood_mage_yearly_events.003 = {
    hidden = yes
    trigger = {
        OR = {
            has_character_modifier = minor_blood_rune_modifier
            has_character_modifier = major_blood_rune_modifier
            has_character_modifier = superior_blood_rune_modifier
        }
    }
    immediate = {
        if = {
            limit = { has_character_modifier = minor_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                80 = {  } # chance of nothing happening
                20 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
        if = {
            limit = { has_character_modifier = major_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                65 = {  } # chance of nothing happening
                20 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
                10 = { # major lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                    }
                }
                5 = { # both minor and major lifeforce
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
        if = {
            limit = { has_character_modifier = superior_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                60 = { } # chance of nothing happening
                10 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
                20 = { # major lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                    }
                }
                10 = { # both minor and major lifeforce
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
    }
}
