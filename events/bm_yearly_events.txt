namespace = blood_mage_yearly_events

# Yearly event - Add xp to Blood Mages
blood_mage_yearly_events.001 = {
    hidden = yes
    trigger = { has_trait = lifestyle_blood_mage }
    immediate = {
        # Base XP gain for all Blood Mages
        add_trait_xp = {
            trait = lifestyle_blood_mage
            track = ancient
            value = 1
        }

        # Ancient attunement check
        if = {
            limit = { has_character_modifier = ancient_attuned }
            random = {
                chance = 50
                send_interface_toast = {
                    title = bm_yearly_attunement_ancient_toast
                    left_icon = root
                    add_trait_xp = {
                        trait = lifestyle_blood_mage
                        track = ancient
                        value = 1
                    }
                }
            }
        }

        # Enlightenment attunement check
        if = {
            limit = { has_character_modifier = enlightenment_attuned }
            random = {
                chance = 50
                send_interface_toast = {
                    title = bm_yearly_attunement_enlightenment_toast
                    left_icon = root
                    add_trait_xp = {
                        trait = lifestyle_blood_mage
                        track = enlightenment
                        value = 1
                    }
                }
            }
        }

        # Bloodline attunement check
        if = {
            limit = { has_character_modifier = bloodline_attuned }
            random = {
                chance = 50
                send_interface_toast = {
                    title = bm_yearly_attunement_bloodline_toast
                    left_icon = root
                    add_trait_xp = {
                        trait = lifestyle_blood_mage
                        track = bloodline
                        value = 1
                    }
                }
            }
        }

        # Benediction attunement check
        if = {
            limit = { has_character_modifier = benediction_attuned }
            random = {
                chance = 50
                send_interface_toast = {
                    title = bm_yearly_attunement_benediction_toast
                    left_icon = root
                    add_trait_xp = {
                        trait = lifestyle_blood_mage
                        track = benediction
                        value = 1
                    }
                }
            }
        }

        # Hematurgy attunement check
        if = {
            limit = { has_character_modifier = hematurgy_attuned }

                chance = 50
                send_interface_toast = {
                    title = bm_yearly_attunement_hematurgy_toast
                    left_icon = root
                    add_trait_xp = {
                        trait = lifestyle_blood_mage

blood_mage_yearly_events.002 = {
    hidden = yes
    trigger = {
        has_trait = lifestyle_blood_mage
        exists = house
    }
    immediate = {
        # Dynasty bloodline, get XP based on number of dynasty modifiers
        random = {
            chance =  {
                value = 0
                if = {
                    limit = {house = { has_house_modifier = dynasty_blood_charisma_modifier }}
                    add = 5
                }
                if = {
                    limit = {house = { has_house_modifier = dynasty_blood_fury_modifier }}
                    add = 5
                }
                if = {
                    limit = {house = { has_house_modifier = dynasty_blood_prosperity_modifier }}
                    add = 5
                }
                if = {
                    limit = {house = { has_house_modifier = dynasty_blood_shadows_modifier }}
                    add = 5
                }
                if = {
                    limit = {house = { has_house_modifier = dynasty_blood_insight_modifier }}
                    add = 5
                }
                if = {
                    limit = {house = { has_house_modifier = dynasty_blood_legacy_modifier }}
                    add = 5
                }
                if = {
                    limit = {house = { has_house_modifier = dynasty_blood_expertise_modifier }}
                    add = 5
                }
            }
            # Give XP
            send_interface_toast = {
                title = blood_mage_yearly_events.002.success
                left_icon = root
                add_trait_xp = {
                    trait = lifestyle_blood_mage
                    track = bloodline
                    value = 1
                }
            }
        }
    }
}

blood_mage_yearly_events.003 = {
    hidden = yes
    trigger = { has_trait = lifestyle_blood_mage }
    immediate = {
        if = {
            limit = { has_character_modifier = minor_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                80 = {  } # chance of nothing happening
                20 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
        if = {
            limit = { has_character_modifier = major_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                65 = {  } # chance of nothing happening
                20 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
                10 = { # major lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                    }
                }
                5 = { # both minor and major lifeforce
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
        if = {
            limit = { has_character_modifier = superior_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                50 = { } # chance of nothing happening
                30 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
                20 = { # major lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                    }
                }
                10 = { # both minor and major lifeforce
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
    }
}
