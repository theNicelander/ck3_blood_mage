# Yearly event - Add xp to Blood Mages
blood_mage_yearly_events.001 = {
    hidden = yes
    trigger = { is_blood_mage_any_tier_trigger = yes }
    immediate = {
        # Base XP gain for all Blood Mages (silent, no toast)
        bm_add_blood_mage_xp_effect = {
            TRACK = ancient
            XP = 1
        }

        # Ancient attunement check
        if = {
            limit = { has_character_modifier = ancient_attuned }
            send_interface_toast = {
                title = bm_yearly_attunement_ancient_toast
                left_icon = root
                if = {
                    limit = { bm_has_superior_stave_trigger = { STAVE = hulinhjalmur } }
                    bm_add_blood_mage_xp_effect = {
				TRACK = ancient
				XP = 2
			}
                }
                else = {
                    bm_add_blood_mage_xp_effect = {
				TRACK = ancient
				XP = 1
			}
                }
            }

blood_mage_yearly_events.003 = {
    hidden = yes
    trigger = {
        OR = {
            has_character_modifier = minor_blood_rune_modifier
            has_character_modifier = major_blood_rune_modifier
            has_character_modifier = superior_blood_rune_modifier
        }
    }
    immediate = {
        if = {
            limit = { has_character_modifier = minor_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                80 = {  } # chance of nothing happening
                20 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
        if = {
            limit = { has_character_modifier = major_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                65 = {  } # chance of nothing happening
                20 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
                10 = { # major lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                    }
                }
                5 = { # both minor and major lifeforce
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
        if = {
            limit = { has_character_modifier = superior_blood_rune_modifier }
            duel = {
                skill = learning
                value = 0
                60 = { } # chance of nothing happening
                10 = { # minor lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
                20 = { # major lifeforce only
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                    }
                }
                10 = { # both minor and major lifeforce
                    send_interface_toast = {
                        title = bm_blood_rune_yearly_event_popup.title
                        left_icon = root
                        add_character_modifier = { modifier = lifeforce_modifier_major }
                        add_character_modifier = { modifier = lifeforce_modifier_minor }
                    }
                }
            }
        }
    }
}
