# 0001: Prisoner Blood Duel Conclusion
bm_prisoner_blood_duel.0001 = {
	type = character_event
	title = bm_prisoner_blood_duel.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					this = scope:sc_victor
				}
				desc = bm_prisoner_blood_duel.0001.victor_desc
			}
			desc = bm_prisoner_blood_duel.0001.loser_desc
		}
	}
	theme = war
	override_icon = {
		reference = "gfx/interface/icons/bm_blood_magic_icon.dds"
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:sc_defender
		animation = marshal
	}

	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { this = scope:sc_victor }
						desc = bm_prisoner_blood_duel.0001.opt_victory
					}
					desc = bm_prisoner_blood_duel.0001.opt_defeat
				}
			}
		}
		if = {
			limit = { this = scope:sc_victor }
			send_interface_toast = {
				type = bm_event_toast_effect_good
				title = bm_prisoner_blood_duel_victor_toast
				left_icon = root
				right_icon = scope:sc_defender
				random_list = {
					0 = {
						modifier = {
							add = bm_prisoner_blood_duel_superior_chance_value
						}
						desc = bm_prisoner_blood_duel.0001.superior_tt
						add_character_modifier = {
							modifier = lifeforce_modifier_superior
						}
					}
					0 = {
						modifier = {
							add = bm_prisoner_blood_duel_major_chance_value
						}
						desc = bm_prisoner_blood_duel.0001.major_tt
						add_character_modifier = {
							modifier = lifeforce_modifier_major
						}
					}
				}
				bm_add_blood_mage_xp_effect = {
					TRACK = fury
					XP = 4
				}
			}
			lifedrain_execution_effect = {
				VICTIM = scope:sc_defender
				EXECUTIONER = root
			}
		}
		else = {
			send_interface_toast = {
				type = bm_event_toast_effect_bad
				title = bm_prisoner_blood_duel_loser_toast
				left_icon = root
				right_icon = scope:sc_defender
				increase_wounds_no_death_effect = { REASON = fight }
				add_stress = 20
				add_character_modifier = {
					modifier = lifeforce_modifier_minor
				}
				bm_add_blood_mage_xp_effect = {
					TRACK = fury
					XP = 1
				}
				scope:sc_defender = {
					release_from_prison = yes
				}
			}
		}
	}
}
