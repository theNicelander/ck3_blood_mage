namespace = bm_compat_lotr_trait_drain

# LotR: Realms in Exile - Middle-earth Blood & Essence Harvest
bm_compat_lotr_trait_drain.0001 = {
	type = character_event
	title = bm_compat_lotr_trait_drain.0001.title
	desc = bm_compat_lotr_trait_drain.0001.desc
	theme = witchcraft
	override_icon = {
		reference = "gfx/interface/icons/bm_blood_magic_icon.dds"
	}

	# 1. Drain Blood of Numenor (tiers 1-10)
	option = {
		name = bm_compat_lotr_trait_drain.0001.numenor
		trigger = {
			bm_compat_lotr_can_drain_numenor_trigger = yes
			trigger_if = {
				limit = { has_trait_rank = { trait = blood_of_numenor rank >= 7 } }
				piety >= 350
			}
			trigger_else_if = {
				limit = { has_trait_rank = { trait = blood_of_numenor rank >= 4 } }
				piety >= 200
			}
			trigger_else = {
				piety >= 100
			}
		}
		show_as_unavailable = {
			bm_compat_lotr_can_drain_numenor_trigger = yes
		}

		if = {
			limit = { has_trait_rank = { trait = blood_of_numenor rank >= 7 } }
			add_piety = -350
		}
		else_if = {
			limit = { has_trait_rank = { trait = blood_of_numenor rank >= 4 } }
			add_piety = -200
		}
		else = {
			add_piety = -100
		}
		bm_trait_drain_commit_cost_and_cooldown_effect = yes
		temporary_trait_drain_attempted = yes

		duel = {
			skill = learning
			target = scope:recipient
			50 = {
				desc = bm_trait_drain.001.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 5
					min = -40
				}
				modifier = { add = trait_drain_modifier }
				modifier = {
					add = -30
					NOT = { has_trait_rank = { trait = blood_of_numenor rank >= 4 } }
				}
				modifier = {
					add = -60
					has_trait_rank = { trait = blood_of_numenor rank >= 4 }
					NOT = { has_trait_rank = { trait = blood_of_numenor rank >= 7 } }
				}
				modifier = {
					add = -100
					has_trait_rank = { trait = blood_of_numenor rank >= 7 }
				}

				send_interface_toast = {
					type = bm_event_toast_effect_good
					title = bm_trait_drain.001.success
					left_icon = root
					right_icon = scope:recipient
					if = {
						limit = { NOT = { has_trait = blood_of_numenor } }
						add_trait = blood_of_numenor_1
					}
					else = {
						change_trait_rank = {
							trait = blood_of_numenor
							rank = 1
							max = 10
						}
					}
					add_xp_bm_dynamic = {
						TRACK = cunning
						XP = 3
					}
					add_xp_bm_dynamic = {
						TRACK = ancient
						XP = 2
					}
				}

				sacrifice_prisoner_effect = {
					VICTIM = scope:recipient
					EXECUTIONER = root
				}
			}
			50 = {
				modifier = { add = recipient_trait_drain_modifier_per_trait }
				bm_trait_drain_duel_fail_effects = yes
			}
		}
		ai_chance = {
			base = 45
			bm_ai_trait_drain_safety_modifier = yes
		}
	}

	# 2. Drain Skinchanger
	option = {
		name = bm_compat_lotr_trait_drain.0001.skinchanger
		trigger = {
			bm_compat_lotr_can_drain_skinchanger_trigger = yes
			piety >= 250
		}
		show_as_unavailable = {
			bm_compat_lotr_can_drain_skinchanger_trigger = yes
		}

		bm_trait_drain_execute_single_effect = {
			TRAIT = skinchanger
			PIETY_COST = -250
			DUEL_PENALTY = -50
			XP_CUNNING = 5
			SUCCESS_DESC = bm_trait_drain.001.success
		}
		ai_chance = {
			base = 35
			modifier = {
				OR = {
					martial >= 12
					prowess >= 12
				}
				add = 15
			}
			bm_ai_trait_drain_safety_modifier = yes
		}
	}

	# 3. Drain Sorcerer (tiers 1-3)
	option = {
		name = bm_compat_lotr_trait_drain.0001.sorcerer
		trigger = {
			bm_compat_lotr_can_drain_sorcerer_trigger = yes
			trigger_if = {
				limit = { has_trait = sorcerer_2 }
				piety >= 400
			}
			trigger_else_if = {
				limit = { has_trait = sorcerer_1 }
				piety >= 250
			}
			trigger_else = {
				piety >= 150
			}
		}
		show_as_unavailable = {
			bm_compat_lotr_can_drain_sorcerer_trigger = yes
		}

		if = {
			limit = { has_trait = sorcerer_2 }
			add_piety = -400
		}
		else_if = {
			limit = { has_trait = sorcerer_1 }
			add_piety = -250
		}
		else = {
			add_piety = -150
		}
		bm_trait_drain_commit_cost_and_cooldown_effect = yes
		temporary_trait_drain_attempted = yes

		duel = {
			skill = learning
			target = scope:recipient
			50 = {
				desc = bm_trait_drain.001.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 5
					min = -40
				}
				modifier = { add = trait_drain_modifier }
				modifier = {
					add = -40
					NOT = { has_trait = sorcerer_2 }
				}
				modifier = {
					add = -70
					has_trait = sorcerer_2
				}

				send_interface_toast = {
					type = bm_event_toast_effect_good
					title = bm_trait_drain.001.success
					left_icon = root
					right_icon = scope:recipient
					if = {
						limit = { NOT = { has_trait = sorcerer } }
						add_trait = sorcerer_1
					}
					else = {
						change_trait_rank = {
							trait = sorcerer
							rank = 1
							max = 3
						}
					}
					add_xp_bm_dynamic = {
						TRACK = cunning
						XP = 4
					}
					add_xp_bm_dynamic = {
						TRACK = wisdom
						XP = 2
					}
				}

				sacrifice_prisoner_effect = {
					VICTIM = scope:recipient
					EXECUTIONER = root
				}
			}
			50 = {
				modifier = { add = recipient_trait_drain_modifier_per_trait }
				bm_trait_drain_duel_fail_effects = yes
			}
		}
		ai_chance = {
			base = 45
			modifier = {
				learning >= 12
				add = 15
			}
			bm_ai_trait_drain_safety_modifier = yes
		}
	}

	# 4. Drain Light of the Two Trees
	option = {
		name = bm_compat_lotr_trait_drain.0001.twotrees
		trigger = {
			bm_compat_lotr_can_drain_twotrees_trigger = yes
			piety >= 750
			piety_level >= 3
		}
		show_as_unavailable = {
			bm_compat_lotr_can_drain_twotrees_trigger = yes
		}

		bm_trait_drain_execute_single_dual_xp_effect = {
			TRAIT = twotreeslight
			PIETY_COST = -750
			DUEL_PENALTY = -100
			XP_CUNNING = 5
			EXTRA_TRACK = ancient
			EXTRA_XP = 5
		}
		ai_chance = {
			base = 50
			bm_ai_trait_drain_safety_modifier = yes
		}
	}

	# Back
	option = {
		name = bm_compat_lotr_trait_drain.0001.back
		if = {
			limit = {
				is_ai = no
				bm_can_drain_vanilla_traits_from_target_trigger = yes
			}
			trigger_event = bm_trait_drain.001
		}
		ai_chance = {
			base = 60
			bm_ai_trait_drain_decline_modifier = yes
		}
	}

	left_portrait = {
		character = root
		animation = rage
	}
	right_portrait = {
		character = scope:recipient
		animation = beg
	}
}
