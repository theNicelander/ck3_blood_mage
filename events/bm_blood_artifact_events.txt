# Step 3: Select Untempered Forged Blood Artifact to Temper with Egill
bm_blood_artifact.0010 = {
	type = character_event
	title = bm_blood_artifact.0010.title
	desc = bm_blood_artifact.0010.desc
	theme = learning
	override_icon = {
		reference = "gfx/interface/icons/bm_blood_magic_icon.dds"
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:bm_egill
		animation = personality_bold
	}

	immediate = {
		hidden_effect = {
			bm_create_egill_effect = yes
			every_character_artifact = {
				limit = {
					has_variable = bm_is_forged_blood_artifact
					NOT = { has_variable = bm_artifact_tempered }
				}
				if = {
					limit = { NOT = { exists = scope:bm_art_opt_1 } }
					save_scope_as = bm_art_opt_1
				}
				else_if = {
					limit = { NOT = { exists = scope:bm_art_opt_2 } }
					save_scope_as = bm_art_opt_2
				}
				else_if = {
					limit = { NOT = { exists = scope:bm_art_opt_3 } }
					save_scope_as = bm_art_opt_3
				}
			}
		}
	}

	option = {
		name = bm_blood_artifact.0010.opt1
		trigger = { exists = scope:bm_art_opt_1 }
		scope:bm_art_opt_1 = { save_scope_as = bm_temper_target }
		trigger_event = { id = bm_blood_artifact.0011 }
	}

	option = {
		name = bm_blood_artifact.0010.opt2
		trigger = { exists = scope:bm_art_opt_2 }
		scope:bm_art_opt_2 = { save_scope_as = bm_temper_target }
		trigger_event = { id = bm_blood_artifact.0011 }
	}

	option = {
		name = bm_blood_artifact.0010.opt3
		trigger = { exists = scope:bm_art_opt_3 }
		scope:bm_art_opt_3 = { save_scope_as = bm_temper_target }
		trigger_event = { id = bm_blood_artifact.0011 }
	}

	option = {
		name = bm_blood_artifact.back
		add_piety_no_experience = 250
		hidden_effect = { bm_vanish_egill_effect = yes }
	}
}

# Step 4: Choose Prowess (SCE) or Learning Duel vs Egill
bm_blood_artifact.0011 = {
	type = character_event
	title = bm_blood_artifact.0011.title
	desc = bm_blood_artifact.0011.desc
	theme = martial
	override_icon = {
		reference = "gfx/interface/icons/bm_blood_magic_icon.dds"
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:bm_egill
		animation = rage
	}

	# Option A: Single Combat Engine (Prowess) Duel
	option = {
		name = bm_blood_artifact.0011.prowess
		trigger = {
			can_start_single_combat_trigger = yes
			scope:bm_egill = { can_start_single_combat_trigger = yes }
		}
		show_as_unavailable = { always = yes }
		blood_magic_used_superior = yes
		add_character_flag = {
			flag = bm_temper_artifact_cooldown
			years = 5
		}
		add_character_flag = bm_temper_sce_pending
		configure_start_single_combat_effect = {
			SC_INITIATOR = root
			SC_ATTACKER = root
			SC_DEFENDER = scope:bm_egill
			FATALITY = practice
			FIXED = no
			LOCALE = wilderness
			OUTPUT_EVENT = bm_blood_artifact.0012
			INVALIDATION_EVENT = bm_blood_artifact.0013
		}
	}

	# Option B: Arcane Contest (Learning) Duel
	option = {
		name = bm_blood_artifact.0011.learning
		blood_magic_used_superior = yes
		add_character_flag = {
			flag = bm_temper_artifact_cooldown
			years = 5
		}
		duel = {
			skill = learning
			target = scope:bm_egill
			50 = {
				desc = bm_blood_artifact.0011.learning_win
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2.5
					min = -40
					max = 40
				}
				bm_temper_duel_win_effect = { TRACK = wisdom }
			}
			50 = {
				desc = bm_blood_artifact.0011.learning_loss
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2.5
					min = -40
					max = 40
				}
				bm_temper_duel_fail_effect = yes
			}
		}
	}

	option = {
		name = bm_blood_artifact.back
		add_piety_no_experience = 250
	}

	# Egill stays alive only while the SCE duel is pending
	after = {
		hidden_effect = {
			if = {
				limit = { NOT = { has_character_flag = bm_temper_sce_pending } }
				bm_vanish_egill_effect = yes
			}
		}
	}
}

# SCE Duel Output Event
bm_blood_artifact.0012 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { scope:sc_victor = root }
			bm_temper_duel_win_effect = { TRACK = fury }
		}
		else = {
			bm_temper_duel_fail_effect = yes
		}
		remove_character_flag = bm_temper_sce_pending
		hidden_effect = { bm_vanish_egill_effect = yes }
	}
}
