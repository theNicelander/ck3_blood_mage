# Single Combat Output Event for Seek Power Battlefield Duel
seek_power.006 = {
    type = character_event
    title = seek_power.006.title
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    exists = scope:sc_victor
                    scope:sc_victor = root
                    exists = scope:duel_opponent
                    scope:duel_opponent = { is_alive = no }
                }
                desc = seek_power.006.desc_slain
            }
            triggered_desc = {
                trigger = {
                    exists = scope:sc_victor
                    scope:sc_victor = root
                }
                desc = seek_power.006.desc_victory
            }
            desc = seek_power.006.desc_defeat
        }
    }
    theme = battle

    immediate = {
        if = {
            limit = { has_variable = bm_seek_power_opponent }
            var:bm_seek_power_opponent = {
                save_scope_as = duel_opponent
            }
            remove_variable = bm_seek_power_opponent
        }
        else_if = {
            limit = { exists = scope:sc_defender }
            scope:sc_defender = {
                save_scope_as = duel_opponent
            }
        }
    }

    # Option 1 (Victor, opponent survived FATALITY = possible): Harvest their vital lifeforce (Major + 25% Superior + 5 Slaughter XP)
    option = {
        name = seek_power.006.a
        trigger = {
            exists = scope:sc_victor
            scope:sc_victor = root
            exists = scope:duel_opponent
            scope:duel_opponent = { is_alive = yes }
        }
        ai_chance = { base = 80 }
        add_blood_knight_xp = { TRACK = slaughter XP = 5 }
        add_character_modifier = { modifier = lifeforce_modifier_major }
        random = {
            chance = 25
            add_character_modifier = { modifier = lifeforce_modifier_superior }
        }
        hidden_effect = {
            bm_recount_lifeforce_stacks_effect = yes
        }
        send_interface_toast = {
            type = event_toast_effect_good
            title = bm_duel_kill_blood_knight_toast
            left_icon = root
            right_icon = scope:duel_opponent
            custom_tooltip = bm_duel_kill_blood_knight_tt
        }
    }

    # Option 2 (Victor, opponent survived): Spare & recruit the defeated warrior to court
    option = {
        name = seek_power.006.spare
        trigger = {
            exists = scope:sc_victor
            scope:sc_victor = root
            exists = scope:duel_opponent
            scope:duel_opponent = { is_alive = yes }
        }
        ai_chance = { base = 20 }
        add_courtier = scope:duel_opponent
        add_blood_knight_xp = { TRACK = vanguard XP = 2 }
        add_character_modifier = { modifier = lifeforce_modifier_minor }
        hidden_effect = {
            bm_recount_lifeforce_stacks_effect = yes
        }
    }

    # Option 3 (Victor, opponent already died in combat and triggered bm_on_character_death_duel)
    option = {
        name = seek_power.006.slain
        trigger = {
            exists = scope:sc_victor
            scope:sc_victor = root
            OR = {
                NOT = { exists = scope:duel_opponent }
                scope:duel_opponent = { is_alive = no }
            }
        }
        ai_chance = { base = 100 }
        add_prestige = 50
    }

    # Option 4 (Defeat): Retreat wounded
    option = {
        name = seek_power.006.defeat
        trigger = {
            OR = {
                NOT = { exists = scope:sc_victor }
                scope:sc_victor != root
            }
        }
        ai_chance = { base = 100 }
        add_stress = 15
        add_blood_knight_xp = { TRACK = vanguard XP = 1 }
    }

    left_portrait = {
        character = root
        animation = marshal
    }
    right_portrait = {
        character = scope:duel_opponent
        animation = pain
    }
}
