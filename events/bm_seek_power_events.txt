seek_power.001 = {
    hidden = yes

    immediate = {
        debug_log = "BloodHunt: dispatcher starting"
        debug_log_scopes = yes
        set_variable = {
            name = bm_seek_power_chose_traveler
            value = no
        }
        random = {
            chance = bm_seek_power_traveler_chance_value
            set_variable = {
                name = bm_seek_power_chose_traveler
                value = yes
            }
        }
        if = {
            limit = {
                has_variable = bm_seek_power_chose_traveler
                var:bm_seek_power_chose_traveler = yes
            }
            remove_variable = bm_seek_power_chose_traveler
            debug_log = "BloodHunt: traveler encounter selected"
            trigger_event = { id = seek_power.003 }
        }
        else = {
            remove_variable = bm_seek_power_chose_traveler
            debug_log = "BloodHunt: rolling non-traveler wilderness encounters"
            duel = {
                skill = prowess
                value = 0
                35 = {
                    desc = seek_power.duel.animal
                    trigger_event = { id = seek_power.002 }
                }
                20 = {
                    desc = seek_power.duel.beast_lair
                    compare_modifier = {
                        value = scope:duel_value
                        multiplier = 0.5
                    }
                    trigger_event = { id = seek_power.008 }
                }

                    desc = seek_power.duel.homestead
                    compare_modifier = {
                        value = scope:duel_value

                    }

    type = character_event
    title = seek_power.006.title
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    exists = scope:sc_victor
                    scope:sc_victor = root
                    exists = scope:duel_opponent
                    scope:duel_opponent = { is_alive = no }
                }
                desc = seek_power.006.desc_slain
            }
            triggered_desc = {
                trigger = {
                    exists = scope:sc_victor
                    scope:sc_victor = root
                }
                desc = seek_power.006.desc_victory
            }
            desc = seek_power.006.desc_defeat
        }
    }
    theme = battle

    left_portrait = {
        character = root
        animation = marshal
    }
    right_portrait = {
        trigger = {
            exists = scope:duel_opponent
        }
        character = scope:duel_opponent
        animation = pain
    }

    immediate = {
        if = {
            limit = { has_variable = bm_seek_power_opponent }
            var:bm_seek_power_opponent = {
                save_scope_as = duel_opponent
                save_scope_as = escort_knight
            }
            remove_variable = bm_seek_power_opponent
        }
        else_if = {
            limit = { exists = scope:sc_defender }
            scope:sc_defender = {
                save_scope_as = duel_opponent
                save_scope_as = escort_knight
            }
        }
    }

    # Option 1 (Victor, opponent survived FATALITY = possible): Harvest their vital lifeforce (Major + 25% Superior + 5 Slaughter XP)
    option = {
        name = seek_power.006.a
        trigger = {
            exists = scope:sc_victor
            scope:sc_victor = root
            exists = scope:duel_opponent
            scope:duel_opponent = { is_alive = yes }
        }
        ai_chance = { base = 80 }
        send_interface_toast = {
            type = event_toast_effect_good
            title = bm_duel_kill_blood_knight_toast
            left_icon = root
            right_icon = scope:duel_opponent
            if = {
                limit = { has_trait = lifestyle_blood_knight }

            }
            if = {
                limit = { has_trait = lifestyle_blood_mage }
                add_blood_mage_xp = { TRACK = hematurgy XP = 5 }
            }
            add_character_modifier = { modifier = lifeforce_modifier_major }
            random = {
                chance = 25
                add_character_modifier = { modifier = lifeforce_modifier_superior }
            }
            hidden_effect = {
                bm_recount_lifeforce_stacks_effect = yes
            }
        }
    }

    # Option 2 (Victor, opponent survived): Spare & recruit the defeated warrior to court (Prowess duel)
    option = {
        name = seek_power.006.spare
        trigger = {
            exists = scope:sc_victor
            scope:sc_victor = root
            exists = scope:duel_opponent
            scope:duel_opponent = { is_alive = yes }
        }
        ai_chance = { base = 20 }
        duel = {
            skill = prowess
            target = scope:duel_opponent
            50 = {
                desc = seek_power.duel.join_knight
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = 2.0
                    min = -45
                    max = 45
                }
                send_interface_toast = {
                    title = seek_power.toast.spare_success
                    left_icon = root
                    right_icon = scope:duel_opponent
                    add_courtier = scope:duel_opponent
                    if = {
                        limit = { has_trait = lifestyle_blood_knight }
                        add_blood_knight_xp = { TRACK = vanguard XP = 2 }
                    }

                        limit = { has_trait = lifestyle_blood_mage }
                        add_blood_mage_xp = { TRACK = enlightenment XP = 2 }
                    }
                    add_character_modifier = { modifier = lifeforce_modifier_minor }

                    }

                    value = scope:duel_value

                    max = 45

                    left_icon = root
                    right_icon = scope:duel_opponent
                }
            }
        }
    }

    # Option 3 (Victor, opponent already died in combat and triggered bm_on_character_death_duel)
    option = {

            scope:sc_victor = root
