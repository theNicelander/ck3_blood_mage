	title = bm_blood_power_event.001.title

	option = {
		name = bm_empower_dynasty

			OR = {

					trait = lifestyle_blood_power
					track = dynasty
					value < 100
				}
			}
		}
		show_as_unavailable = {
			always = yes
		}
		add_piety = -150
		blood_magic_used_major = yes
		send_interface_toast = {
			title = bm_blood_power_event.001.toast
			left_icon = this
			add_shared_bm_bk_xp = {
				TRACK = benediction
				XP = 3
			}
			add_blood_power_xp = { TRACK = dynasty }
		}
		custom_tooltip = bm_empower_dynasty_desc
