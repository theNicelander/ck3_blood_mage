# Transmute Lifeforce Events

namespace = bm_transmute_lifeforce

# Root Event: Category Selection
bm_transmute_lifeforce.0001 = {
	type = character_event
	title = bm_transmute_lifeforce.0001.t
	desc = bm_transmute_lifeforce.0001.desc
	theme = witchcraft
	override_icon = {
		reference = "gfx/interface/icons/trait_level_tracks/ancient.dds"
	}

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# 1. Major Transmutations
	option = {
		name = bm_transmute_lifeforce.0001.major
		trigger = {
			is_blood_mage_any_tier_trigger = yes
		}
		show_as_unavailable = {
			always = yes
		}
		trigger_event = { id = bm_transmute_lifeforce.0002 }
		ai_chance = {
			base = 50
			modifier = {
				factor = 0
				NOT = { has_character_modifier = lifeforce_modifier_major }
			}
		}
	}

	# 2. Reverse (Reclaim Stored Lifeforce)
	option = {
		name = bm_transmute_lifeforce.0001.reverse
		trigger = {
			is_blood_mage_any_tier_trigger = yes
		}
		show_as_unavailable = {
			always = yes
		}
		trigger_event = { id = bm_transmute_lifeforce.0004 }
		ai_chance = {
			base = 50
			modifier = {
				factor = 0
				NOT = { has_character_modifier = lifeforce_modifier_radiant }
				NOT = { has_character_modifier = lifeforce_modifier_furious }
				NOT = { has_character_modifier = lifeforce_modifier_fortunate }
				NOT = { has_character_modifier = lifeforce_modifier_cunning }
				NOT = { has_character_modifier = lifeforce_modifier_wise }
				NOT = { has_character_modifier = lifeforce_modifier_ancient }
			}
		}
	}

	# 4. Back
	option = {
		name = bm_transmute_lifeforce.cancel
		trigger = {
			always = yes
		}
		ai_chance = {
			base = 0
		}
	}
}
