# Event 1: A Drop on the Lips (Lover / Soulmate / Spouse)
bm_interpersonal_events.0001 = {
	type = character_event
	title = bm_interpersonal_events.0001.t
	desc = bm_interpersonal_events.0001.desc
	theme = love
	left_portrait = {
		character = root
		animation = flirtation
	}
	right_portrait = {
		character = scope:partner
		animation = love
	}

	trigger = {
		OR = {
			has_trait = lifestyle_blood_mage
			has_trait = lifestyle_blood_knight
		}
		OR = {
			any_relation = { type = soulmate is_alive = yes }
			any_relation = { type = lover is_alive = yes }
			any_spouse = { is_alive = yes }
			any_consort = { is_alive = yes }
		}
	}

	immediate = {
		add_character_flag = { flag = bm_interpersonal_event_cooldown years = 3 }
		if = {
			limit = { any_relation = { type = soulmate is_alive = yes } }
			random_relation = { type = soulmate limit = { is_alive = yes } save_scope_as = partner }
		}
		else_if = {
			limit = { any_relation = { type = lover is_alive = yes } }
			random_relation = { type = lover limit = { is_alive = yes } save_scope_as = partner }
		}
		else_if = {
			limit = { any_spouse = { is_alive = yes } }
			random_spouse = { limit = { is_alive = yes } save_scope_as = partner }
		}
		else = {
			random_consort = { limit = { is_alive = yes } save_scope_as = partner }
		}
	}

	# Option A: Share drop of Lifeforce
	option = {
		name = bm_interpersonal_events.0001.a
		trigger = {
			has_character_modifier = lifeforce_modifier_minor
		}
		show_as_unavailable = {
			always = yes
		}
		send_interface_toast = {
			title = bm_interpersonal_events.0001.a.toast
			left_icon = root
			right_icon = scope:partner
			blood_magic_used_minor = yes
			scope:partner = {
				add_character_modifier = {
					modifier = bm_blessed_blood_modifier
					years = 10
				}
				add_opinion = {
					target = root
					modifier = magic_opinion_positive
					opinion = 30
				}
				add_stress = -30
			}
			add_stress = -30
			if = {
				limit = { has_trait = lifestyle_blood_mage }
				add_blood_mage_xp = {
					TRACK = benediction
					XP = 1
				}
			}
			else_if = {
				limit = { has_trait = lifestyle_blood_knight }
				add_blood_knight_xp = {
					TRACK = benediction
					XP = 1
				}
			}
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
	}

	# Option B: Drink from offered vein
	option = {
		name = bm_interpersonal_events.0001.b
		send_interface_toast = {
			title = bm_interpersonal_events.0001.b.toast
			left_icon = root
			right_icon = scope:partner
			scope:partner = {
				add_character_modifier = {
					modifier = lifeforce_modifier_negative_minor
					years = 1
				}
			}
			if = {
				limit = { has_relation_soulmate = scope:partner }
				add_character_modifier = { modifier = lifeforce_modifier_major }
			}
			else = {
				add_character_modifier = { modifier = lifeforce_modifier_minor }
			}
			if = {
				limit = { has_trait = lifestyle_blood_mage }
				add_blood_mage_xp = {
					TRACK = hematurgy
					XP = 1
				}
			}
			else_if = {
				limit = { has_trait = lifestyle_blood_knight }
				add_blood_knight_xp = {
					TRACK = slaughter
					XP = 1
				}
			}
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
	}

	# Option C: Initiate into Crimson Mysteries
	option = {
		name = bm_interpersonal_events.0001.c
		trigger = {
			scope:partner = {
				NOT = { has_trait = lifestyle_blood_mage }
				NOT = { has_trait = lifestyle_blood_knight }
			}
			has_character_modifier = lifeforce_modifier_major
		}
		show_as_unavailable = {
			scope:partner = {
				NOT = { has_trait = lifestyle_blood_mage }
				NOT = { has_trait = lifestyle_blood_knight }
			}
		}
		send_interface_toast = {
			title = bm_interpersonal_events.0001.c.toast
			left_icon = root
			right_icon = scope:partner
			blood_magic_used_major = yes
			if = {
				limit = { root = { has_trait = lifestyle_blood_mage } }
				scope:partner = { bm_become_blood_mage_effect = yes }
			}
			else = {
				scope:partner = { bm_become_blood_knight_effect = yes }
			}
			set_relation_soulmate = scope:partner
			if = {
				limit = { has_trait = lifestyle_blood_mage }
				add_blood_mage_xp = {
					TRACK = bloodline
					XP = 1
				}
			}
			else_if = {
				limit = { has_trait = lifestyle_blood_knight }
				add_blood_knight_xp = {
					TRACK = benediction
					XP = 1
				}
			}
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
	}
}

# Event 2: The Fever of Awakening (Child / Ward Age 6–15)

	title = bm_interpersonal_events.0002.t
	desc = bm_interpersonal_events.0002.desc

	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:child
		animation = sick
	}

		OR = {
			has_trait = lifestyle_blood_mage
			has_trait = lifestyle_blood_knight
		}

			any_child = {
				is_alive = yes
				age >= 6
				age <= 15
			}

				type = ward

				age <= 15
			}
		}
	}

	immediate = {
		add_character_flag = { flag = bm_interpersonal_event_cooldown years = 3 }
		if = {
			limit = {
				any_child = {
					is_alive = yes

					age <= 15
				}
			}
			random_child = {
				limit = {
					is_alive = yes
					age >= 6
					age <= 15
				}
				save_scope_as = child
			}
		}
		else = {
			random_relation = {
				type = ward
				limit = {
					is_alive = yes
					age >= 6
					age <= 15
				}
				save_scope_as = child

					opinion = 20
				}
			}

				limit = { has_trait = lifestyle_blood_mage }
				add_blood_mage_xp = {
					TRACK = bloodline
					XP = 1
				}
			}
			else_if = {
				limit = { has_trait = lifestyle_blood_knight }
				add_blood_knight_xp = {
					TRACK = enlightenment
					XP = 1
				}
			}
			hidden_effect = {
				bm_recount_lifeforce_stacks_effect = yes
			}
		}
	}

		}
		show_as_unavailable = {
