namespace = bm_blood_rune

bm_blood_rune.001 = {
    type = character_event
    title = bm_blood_rune.001.title
    desc = bm_blood_rune.001.desc
    theme = witchcraft

    # Minor Blood Rune
    option = {
        name = bm_blood_rune.001.minor
        trigger = {
            NOT = {
                has_character_modifier = minor_blood_rune_modifier
                has_character_modifier = major_blood_rune_modifier
                has_character_modifier = superior_blood_rune_modifier
            }
        }
        add_character_modifier = {
            modifier = minor_blood_rune_modifier
        }
        ai_chance = {  base = 100 }
    }

    # Major Blood Rune
    option = {
        name = bm_blood_rune.001.major
        trigger = {
            has_character_modifier = minor_blood_rune_modifier
        }
        add_character_modifier = {
            modifier = major_blood_rune_modifier
        }
        remove_character_modifier = minor_blood_rune_modifier
        ai_chance = {  base = 100 }
    }

    # Superior Blood Rune
    option = {
        name = bm_blood_rune.001.superior
        trigger = {
            has_character_modifier = major_blood_rune_modifier
        }
        add_character_modifier = {
            modifier = superior_blood_rune_modifier
        }
        remove_character_modifier = major_blood_rune_modifier
        ai_chance = {  base = 100 }
    }

    # Portraits
    left_portrait = {
        character = root
        animation = pain
    }
}
