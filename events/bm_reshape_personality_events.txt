namespace = bm_reshape_personality

bm_reshape_personality.0001 = {
	type = character_event
	title = bm_reshape_personality.0001.title
	desc = bm_reshape_personality.0001.desc
	theme = witchcraft

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:recipient
		animation = personality_fearful
	}

	immediate = {
		scope:recipient = {
			every_character_trait = {
				limit = { has_trait_category = personality }
				add_to_list = recipient_personality_traits
			}
		}
		if = {
			limit = { list_size:recipient_personality_traits > 0 }
			ordered_in_list = {
				list = recipient_personality_traits
				save_scope_as = recipient_ptrait_1
				remove_from_list = recipient_personality_traits
			}
		}
		if = {
			limit = { list_size:recipient_personality_traits > 0 }
			ordered_in_list = {
				list = recipient_personality_traits
				save_scope_as = recipient_ptrait_2
				remove_from_list = recipient_personality_traits
			}
		}
		if = {
			limit = { list_size:recipient_personality_traits > 0 }
			ordered_in_list = {
				list = recipient_personality_traits
				save_scope_as = recipient_ptrait_3
				remove_from_list = recipient_personality_traits
			}
		}
		if = {
			limit = { list_size:recipient_personality_traits > 0 }
			ordered_in_list = {
				list = recipient_personality_traits
				save_scope_as = recipient_ptrait_4
				remove_from_list = recipient_personality_traits
			}
		}
	}

	option = {
		name = bm_reshape_personality.0001.option_1
		trigger = { exists = scope:recipient_ptrait_1 }
		scope:recipient_ptrait_1 = { save_scope_as = trait_to_remove }
		trigger_event = bm_reshape_personality.0002
		ai_chance = { base = 10 }
	}

	option = {
		name = bm_reshape_personality.0001.option_2
		trigger = { exists = scope:recipient_ptrait_2 }
		scope:recipient_ptrait_2 = { save_scope_as = trait_to_remove }
		trigger_event = bm_reshape_personality.0002
		ai_chance = { base = 10 }
	}

	option = {
		name = bm_reshape_personality.0001.option_3
		trigger = { exists = scope:recipient_ptrait_3 }
		scope:recipient_ptrait_3 = { save_scope_as = trait_to_remove }
		trigger_event = bm_reshape_personality.0002
		ai_chance = { base = 10 }
	}

	option = {
		name = bm_reshape_personality.0001.option_4
		trigger = { exists = scope:recipient_ptrait_4 }
		scope:recipient_ptrait_4 = { save_scope_as = trait_to_remove }
		trigger_event = bm_reshape_personality.0002
		ai_chance = { base = 10 }
	}

	option = {
		name = bm_reshape_personality.0001.culture
		trigger = {
			custom_tooltip = {
				text = bm_reshape_personality.0001.culture.already_culture_tt
				NOT = { scope:recipient.culture = root.culture }
			}
		}
		show_as_unavailable = {
			always = yes
		}
		remove_character_modifier = lifeforce_modifier_major
		hidden_effect = { bm_recount_lifeforce_stacks_effect = yes }
		duel = {
			skill = learning
			target = scope:recipient
			50 = {
				desc = bm_reshape_personality.duel.culture_success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2.25
					min = -45
					max = 45
				}
				scope:recipient = {
					set_culture_same_as = root
				}
				send_interface_toast = {
					title = bm_reshape_personality.toast.culture_title
					desc = bm_reshape_personality.toast.culture_desc
					left_icon = root
					right_icon = scope:recipient
					add_shared_bm_bk_xp = {
						TRACK = enlightenment
						XP = 4
					}
				}
			}
			50 = {
				desc = bm_reshape_personality.duel.culture_failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2.25
					min = -45
					max = 45
				}
				remove_interaction_cooldown_against = {
					interaction = bm_reshape_personality_interaction
					target = scope:recipient
				}
				scope:recipient = {
					set_variable = {
						name = bm_influence_personality_failed_actor
						value = root
						years = 1
					}
				}
				send_interface_toast = {
					title = bm_reshape_personality.toast.culture_failure.title
					desc = bm_reshape_personality.toast.culture_failure.desc
					left_icon = root
					right_icon = scope:recipient
					add_shared_bm_bk_xp = {
						TRACK = enlightenment
						XP = 1
					}
				}
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				scope:recipient.culture = root.culture
			}
		}
	}

	option = {
		name = bm_reshape_personality.0001.cancel
		add_piety = 250
		remove_interaction_cooldown_against = {
			interaction = bm_reshape_personality_interaction
			target = scope:recipient
		}
		ai_chance = { base = 0 }
	}
}
