namespace = bm_buy_captives

# Hidden setup event: spawns 1-3 random captives and opens the broker showcase
bm_buy_captives.0001 = {
	hidden = yes

	immediate = {
		bm_spawn_broker_captive_effect = { SAVE_AS = captive_1 }
		random = {
			chance = 65
			bm_spawn_broker_captive_effect = { SAVE_AS = captive_2 }
			random = {
				chance = 50
				bm_spawn_broker_captive_effect = { SAVE_AS = captive_3 }
			}
		}
		trigger_event = bm_buy_captives.0002
	}
}

# Broker showcase event: buy any individual captive, buy all remaining, or leave
bm_buy_captives.0002 = {
	type = character_event
	title = bm_buy_captives.0002.t
	desc = bm_buy_captives.0002.desc
	theme = intrigue

	left_portrait = {
		character = scope:captive_1
		triggered_animation = {
			trigger = { scope:captive_1 = { is_imprisoned = no } }
			animation = prison_dungeon
		}
		animation = fear
	}
	center_portrait = {
		character = scope:captive_2
		triggered_animation = {
			trigger = { scope:captive_2 = { is_imprisoned = no } }
			animation = prison_dungeon
		}
		animation = fear
	}
	right_portrait = {
		character = scope:captive_3
		triggered_animation = {
			trigger = { scope:captive_3 = { is_imprisoned = no } }
			animation = prison_dungeon
		}
		animation = fear
	}

	# Buy Captive 1
	option = {
		name = bm_buy_captives.0002.a
		trigger = {
			exists = scope:captive_1
			scope:captive_1 = {
				is_alive = yes
				is_imprisoned = no
			}
			short_term_gold >= scope:captive_1.bm_captive_price_value
		}
		show_as_unavailable = {
			exists = scope:captive_1
			scope:captive_1 = {
				is_alive = yes
				is_imprisoned = no
			}
		}
		bm_purchase_broker_captive_effect = { TARGET = scope:captive_1 }
		if = {
			limit = {
				OR = {
					AND = {
						exists = scope:captive_2
						scope:captive_2 = { is_alive = yes is_imprisoned = no }
					}
					AND = {
						exists = scope:captive_3
						scope:captive_3 = { is_alive = yes is_imprisoned = no }
					}
				}
			}
			trigger_event = bm_buy_captives.0002
		}
		else = {
			bm_finalize_captive_broker_cooldown_effect = yes
		}
	}

	# Buy Captive 2
	option = {
		name = bm_buy_captives.0002.b
		trigger = {
			exists = scope:captive_2
			scope:captive_2 = {
				is_alive = yes
				is_imprisoned = no
			}
			short_term_gold >= scope:captive_2.bm_captive_price_value
		}
		show_as_unavailable = {
			exists = scope:captive_2
			scope:captive_2 = {
				is_alive = yes
				is_imprisoned = no
			}
		}
		bm_purchase_broker_captive_effect = { TARGET = scope:captive_2 }
		if = {
			limit = {
				OR = {
					AND = {
						exists = scope:captive_1
						scope:captive_1 = { is_alive = yes is_imprisoned = no }
					}
					AND = {
						exists = scope:captive_3
						scope:captive_3 = { is_alive = yes is_imprisoned = no }
					}
				}
			}
			trigger_event = bm_buy_captives.0002
		}
		else = {
			bm_finalize_captive_broker_cooldown_effect = yes
		}
	}

	# Buy Captive 3
	option = {
		name = bm_buy_captives.0002.c
		trigger = {
			exists = scope:captive_3
			scope:captive_3 = {
				is_alive = yes
				is_imprisoned = no
			}
			short_term_gold >= scope:captive_3.bm_captive_price_value
		}
		show_as_unavailable = {
			exists = scope:captive_3
			scope:captive_3 = {
				is_alive = yes
				is_imprisoned = no
			}
		}
		bm_purchase_broker_captive_effect = { TARGET = scope:captive_3 }
		if = {
			limit = {
				OR = {
					AND = {
						exists = scope:captive_1
						scope:captive_1 = { is_alive = yes is_imprisoned = no }
					}
					AND = {
						exists = scope:captive_2
						scope:captive_2 = { is_alive = yes is_imprisoned = no }
					}
				}
			}
			trigger_event = bm_buy_captives.0002
		}
		else = {
			bm_finalize_captive_broker_cooldown_effect = yes
		}
	}

	# Buy All Remaining (shown when >= 2 unbought captives remain)
	option = {
		name = bm_buy_captives.0002.d
		trigger = {
			calc_true_if = {
				amount >= 2
				AND = {
					exists = scope:captive_1
					scope:captive_1 = { is_alive = yes is_imprisoned = no }
				}
				AND = {
					exists = scope:captive_2
					scope:captive_2 = { is_alive = yes is_imprisoned = no }
				}
				AND = {
					exists = scope:captive_3
					scope:captive_3 = { is_alive = yes is_imprisoned = no }
				}
			}
			short_term_gold >= bm_captive_all_remaining_price_value
		}
		show_as_unavailable = {
			calc_true_if = {
				amount >= 2
				AND = {
					exists = scope:captive_1
					scope:captive_1 = { is_alive = yes is_imprisoned = no }
				}
				AND = {
					exists = scope:captive_2
					scope:captive_2 = { is_alive = yes is_imprisoned = no }
				}
				AND = {
					exists = scope:captive_3
					scope:captive_3 = { is_alive = yes is_imprisoned = no }
				}
			}
		}
		if = {
			limit = {
				exists = scope:captive_1
				scope:captive_1 = { is_alive = yes is_imprisoned = no }
			}
			bm_purchase_broker_captive_effect = { TARGET = scope:captive_1 }
		}
		if = {
			limit = {
				exists = scope:captive_2
				scope:captive_2 = { is_alive = yes is_imprisoned = no }
			}
			bm_purchase_broker_captive_effect = { TARGET = scope:captive_2 }
		}
		if = {
			limit = {
				exists = scope:captive_3
				scope:captive_3 = { is_alive = yes is_imprisoned = no }
			}
			bm_purchase_broker_captive_effect = { TARGET = scope:captive_3 }
		}
		bm_finalize_captive_broker_cooldown_effect = yes
	}

	# Back / Leave
	option = {
		name = bm_buy_captives.0002.e
		bm_cleanup_unbought_captives_effect = yes
		bm_finalize_captive_broker_cooldown_effect = yes
	}
}
