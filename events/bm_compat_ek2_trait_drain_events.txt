# Elder Kings 2 - Tamrielic Blood & Arcane Harvest
bm_compat_ek2_trait_drain.0001 = {
	type = character_event
	title = bm_compat_ek2_trait_drain.0001.title
	desc = bm_compat_ek2_trait_drain.0001.desc
	theme = witchcraft
	override_icon = {
		reference = "gfx/interface/icons/bm_blood_magic_icon.dds"
	}

	# 1. Drain Arcana (tiers 1-3, purges arcana_bad first)
	option = {
		name = bm_compat_ek2_trait_drain.0001.arcana
		trigger = {
			bm_compat_ek2_can_drain_arcana_trigger = yes
			trigger_if = {
				limit = { has_trait = arcana_good_2 }
				piety >= 250
			}
			trigger_else_if = {
				limit = { has_trait = arcana_good_1 }
				piety >= 175
			}
			trigger_else = {
				piety >= 100
			}
		}
		show_as_unavailable = {
			bm_compat_ek2_can_drain_arcana_trigger = yes
		}

		if = {
			limit = { has_trait = arcana_good_2 }
			add_piety = -250
		}
		else_if = {
			limit = { has_trait = arcana_good_1 }
			add_piety = -175
		}
		else = {
			add_piety = -100
		}
		bm_trait_drain_commit_cost_and_cooldown_effect = yes
		temporary_trait_drain_attempted = yes

		duel = {
			skill = learning
			target = scope:recipient
			50 = {
				desc = bm_trait_drain.001.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 5
					min = -40
				}
				modifier = { add = trait_drain_modifier }

				send_interface_toast = {
					type = bm_event_toast_effect_good
					title = bm_trait_drain.001.success
					left_icon = root
					right_icon = scope:recipient
					if = {
						limit = { has_trait = arcana_bad_1 }
						remove_trait = arcana_bad_1
					}
					else_if = {
						limit = { has_trait = arcana_bad }
						change_trait_rank = {
							trait = arcana_bad
							rank = -1
							max = 3
						}
					}
					else_if = {
						limit = { NOT = { has_trait = arcana_good } }
						add_trait = arcana_good_1
					}
					else = {
						change_trait_rank = {
							trait = arcana_good
							rank = 1
							max = 3
						}
					}
					add_xp_bm_dynamic = {
						TRACK = cunning
						XP = 3
					}
				}

				sacrifice_prisoner_effect = {
					VICTIM = scope:recipient
					EXECUTIONER = root
				}
			}
			50 = {
				scope:recipient = { temporary_negative_recently_trait_drained = yes }
				modifier = { add = recipient_trait_drain_modifier_per_trait }
				send_interface_toast = {
					title = bm_trait_drain.001.fail
					drain_trait_failed_effects = yes
					left_icon = root
					right_icon = scope:recipient
				}
			}
		}
		ai_chance = { base = 100 }
	}

	option = {

		trigger = {

		}

		temporary_trait_drain_attempted = yes

		duel = {
			skill = learning

				desc = bm_trait_drain.001.success

					min = -40
				}
				modifier = { add = trait_drain_modifier }
				modifier = {
					add = -40
					NOT = { has_trait_rank = { trait = lifespan rank >= 3 } }
				}
				modifier = {
					add = -70
					has_trait_rank = { trait = lifespan rank >= 3 }
				}

				send_interface_toast = {
					type = bm_event_toast_effect_good
					title = bm_trait_drain.001.success
					left_icon = root
					right_icon = scope:recipient
					if = {
						limit = { NOT = { has_trait = lifespan } }
						add_trait = lifespan_1
					}
					else = {
						change_trait_rank = {
							trait = lifespan
							rank = 1
							max = 4
						}
					}
					add_xp_bm_dynamic = {
						TRACK = cunning
						XP = 4
					}
					add_xp_bm_dynamic = {
						TRACK = ancient
						XP = 3
					}
				}

				sacrifice_prisoner_effect = {
					VICTIM = scope:recipient
					EXECUTIONER = root
				}
			}
			50 = {
				scope:recipient = { temporary_negative_recently_trait_drained = yes }
				modifier = { add = recipient_trait_drain_modifier_per_trait }
				send_interface_toast = {
					title = bm_trait_drain.001.fail
					drain_trait_failed_effects = yes
					left_icon = root
					right_icon = scope:recipient
				}
			}
		}
		ai_chance = { base = 100 }
	}
