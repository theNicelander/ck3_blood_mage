namespace = bm_geyser_duel

# Duel of Learning against the 111-year-old Wise One by the Geyser
bm_geyser_duel.0001 = {
	type = character_event
	title = bm_geyser_duel.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_character_flag = bm_geyser_retried
				}
				desc = bm_geyser_duel.0001.desc_retry
			}
			desc = bm_geyser_duel.0001.desc
		}
	}
	theme = learning
	override_icon = {
		reference = "gfx/interface/icons/bm_blood_magic_icon.dds"
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:bm_egill
		animation = personality_bold
	}

	immediate = {
		hidden_effect = {
			bm_create_egill_effect = yes
		}
	}

	option = {
		name = {
			trigger = {
				has_character_flag = bm_geyser_retried
			}
			text = bm_geyser_duel.0001.a_retry
		}
		name = {
			trigger = {
				NOT = { has_character_flag = bm_geyser_retried }
			}
			text = bm_geyser_duel.0001.a
		}

		trigger = {
			NOT = { has_trait = lifestyle_bm_archmage }
			OR = {
				has_character_flag = bm_geyser_paid_piety
				piety_level >= 2
			}
		}
		show_as_unavailable = {
			always = yes
		}

		if = {
			limit = { NOT = { has_character_flag = bm_geyser_paid_piety } }
			add_piety_level = -1
			add_character_flag = bm_geyser_paid_piety
		}

		duel = {
			skills = { learning prowess }
			value = 25
			target = scope:bm_egill

			# SUCCESS
			50 = {
				desc = bm_geyser_duel.success_tt

				compare_modifier = {
					value = scope:duel_value
					multiplier = 2.0
					min = -40
					max = 40
				}

				send_interface_toast = {
					title = bm_geyser_duel.success_tt
					left_icon = root
					bm_become_blood_acolyte_effect = yes
				}
			}

			# FAILURE
			50 = {
				desc = bm_geyser_duel.failure_tt

				compare_modifier = {
					value = scope:duel_value
					multiplier = -2.0
					min = -40
					max = 40
				}

				random_list = {
					# INJURY
					50 = {
						desc = bm_geyser_duel.injury_failure_tt
						send_interface_toast = {
							title = bm_geyser_duel.injury_failure_tt
							left_icon = root
							add_stress = 10

							if = {
								limit = { NOT = { has_trait = wounded } }
								add_trait = wounded_1
							}
							else_if = {
								limit = { has_trait = wounded_1 }
								change_trait_rank = {
									trait = wounded
									rank = 1
									max = 2
								}
							}
						}
					}

					# PHYSICAL TRAUMA
					25 = {
						desc = bm_geyser_duel.physical_failure_tt
						send_interface_toast = {
							title = bm_geyser_duel.physical_failure_tt
							left_icon = root
							add_stress = 20
							if = {
								limit = { has_trait = blind }
								add_piety_level = -1
							}
							if = {
								limit = { has_trait = one_eyed }
								add_trait = blind
							}
							if = {
								limit = { has_trait = weak }
								add_trait = one_eyed
							}
							add_trait = weak
						}
					}

					# MENTAL COLLAPSE
					10 = {
						desc = bm_geyser_duel.mental_failure_tt
						send_interface_toast = {
							title = bm_geyser_duel.mental_failure_tt
							left_icon = root
							add_stress = 30

							if = {
								limit = { has_trait = lunatic_1 }
								add_trait = dull
							}
							if = {
								limit = { has_trait = depressed_1 }
								add_trait = lunatic_1
							}
							if = {
								limit = { has_trait = possessed_1 }
								add_trait = depressed_1
							}
							add_trait = possessed_1
						}
					}
				}

				if = {
					limit = { is_alive = yes }
					hidden_effect = {
						add_character_flag = bm_geyser_retrying
						add_character_flag = bm_geyser_retried
						trigger_event = { id = bm_geyser_duel.0001 }
					}
				}
			}
		}

		ai_chance = {
			base = 50
			modifier = {
				learning >= 20
				add = 25
			}
			modifier = {
				prowess >= 25
				add = 25
			}
			modifier = {
				learning < 10
				prowess < 10
				factor = 0.2
			}
		}
	}

	option = {
		name = bm_geyser_duel.0001.b
		custom_tooltip = bm_geyser_duel.0001.b.tt

		ai_chance = {
			base = 100
			modifier = {
				NOT = { bm_geyser_duel_has_bad_traits_trigger = yes }
				factor = 0
			}
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = {
					has_character_flag = bm_geyser_retrying
				}
				remove_character_flag = bm_geyser_retrying
			}
			else = {
				remove_character_flag = bm_geyser_retried
				remove_character_flag = bm_geyser_paid_piety
				bm_vanish_egill_effect = yes
			}
		}
	}
}
