namespace = bm_geyser_duel

# Duel of Learning against the 111-year-old Wise One by the Geyser
bm_geyser_duel.0001 = {
	type = character_event
	title = bm_geyser_duel.0001.t
	desc = bm_geyser_duel.0001.desc
	theme = learning

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:bm_geyser_wise_one
		animation = personality_rational
	}

	immediate = {
		hidden_effect = {
			create_character = {
				age = 111
				gender_female_chance = 50
				location = root.location
				culture = root.culture
				faith = faith:blodtru_faith
				trait = lifestyle_blood_mage
				trait = education_learning_4
				trait = whole_of_body
				learning = 24
				prowess = 15
				save_scope_as = bm_geyser_wise_one
			}
			if = {
				limit = {
					root.location.county ?= title:c_tsushima
				}
				scope:bm_geyser_wise_one = {
					set_character_faith = faith:ketsudo_faith
				}
			}
		}
	}

	option = {
		name = bm_geyser_duel.0001.a

		duel = {
			skill = learning
			target = scope:bm_geyser_wise_one

			# SUCCESS
			50 = {
				desc = bm_geyser_duel.success_tt

				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -45
				}
				modifier = {
					bm_is_blood_cult_faith_trigger = yes
					add = 15
				}

				send_interface_toast = {
					title = bm_geyser_duel.success_tt
					left_icon = root
					bm_become_blood_mage_effect = yes
				}
			}

			# FAILURE - MENTAL COLLAPSE
			25 = {
				desc = bm_geyser_duel.mental_failure_tt
				send_interface_toast = {
					title = bm_geyser_duel.mental_failure_tt
					left_icon = root
					add_stress = 30

					if = {
						limit = { has_trait = lunatic_1 }
						add_trait = dull
					}
					if = {
						limit = { has_trait = depressed_1 }
						add_trait = lunatic_1
					}
					if = {
						limit = { has_trait = possessed_1 }
						add_trait = depressed_1
					}
					add_trait = possessed_1
				}
			}

			# FAILURE - PHYSICAL TRAUMA
			25 = {
				desc = bm_geyser_duel.physical_failure_tt
				send_interface_toast = {
					title = bm_geyser_duel.physical_failure_tt
					left_icon = root
					add_stress = 20
					if = {
						limit = { has_trait = blind }
						add_piety_level = -1
					}
					if = {
						limit = { has_trait = one_eyed }
						add_trait = blind
					}
					if = {
						limit = { has_trait = weak }
						add_trait = one_eyed
					}
					add_trait = weak
				}
			}

			# FAILURE - INJURY
			50 = {
				desc = bm_geyser_duel.injury_failure_tt
				send_interface_toast = {
					title = bm_geyser_duel.injury_failure_tt
					left_icon = root
					add_stress = 10

					if = {
						limit = { has_trait = wounded_2 }
						add_trait = maimed
					}

					if = {
						limit = { NOT = { has_trait = wounded_1 } }
						add_trait = wounded_1
					}
					else = {
						change_trait_rank = {
							trait = wounded
							rank = 1
							max = 2
						}
					}
				}
			}
		}

		ai_chance = {
			base = 100
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = { exists = scope:bm_geyser_wise_one }
				scope:bm_geyser_wise_one = {
					death = { death_reason = death_vanished }
				}
			}
		}
	}
}
