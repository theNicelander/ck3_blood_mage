namespace = bm_blood_cultist

# Conversion to the Cult of Quintessence
bm_blood_cultist.0001 = {
	type = character_event
	title = bm_blood_cultist.0001.t
	desc = bm_blood_cultist.0001.desc
	theme = learning

	left_portrait = {
		character = root
		animation = personality_zealous
	}

	# Option A: Open proclamation
	option = {
		name = bm_blood_cultist.0001.a
		set_character_faith = faith:quintessence_faith

		# Convert capital county if owned
		if = {
			limit = {
				exists = capital_county
				capital_county.holder = root
			}
			capital_county = {
				set_county_faith = faith:quintessence_faith
			}
		}

		add_character_modifier = {
			modifier = lifeforce_modifier_major
		}
		add_piety = 150

		ai_chance = {
			base = 60
			modifier = {
				add = 20
				has_trait = ambitious
			}
		}
	}

	# Option B: Veiled / Hermetic path
	option = {
		name = bm_blood_cultist.0001.b
		set_character_faith = faith:quintessence_faith

		add_character_modifier = {
			modifier = intrigue_lifeforce_modifier
			years = 10
		}
		add_lifestyle_xp = {
			lifestyle = intrigue_lifestyle
			value = 250
		}

		ai_chance = {
			base = 40
			modifier = {
				add = 30
				has_trait = cynical
			}
		}
	}

	# Option C: Reconsider / Step back
	option = {
		name = bm_blood_cultist.0001.c
		add_piety = 500

		ai_chance = {
			base = 0
		}
	}
}
