	# Intellect
	option = {
		name = bm_bestow_traits.0001.intellect
		trigger = {
			custom_tooltip = {
				text = bm_bestow_traits.0001.intellect.cannot_advance_tt
				bm_bestow_traits_can_bestow_intellect_trigger = yes
			}
		}
		show_as_unavailable = {
			scope:actor = { has_trait = intellect_good }
		}
		blood_magic_used_major = yes
		if = {
			limit = {
				OR = {
					AND = {
						scope:actor = { has_trait = intellect_good_1 }
						scope:recipient = { NOT = { has_trait = intellect_good } }
					}
					AND = {
						scope:actor = { has_trait = intellect_good_2 }
						scope:recipient = { has_trait = intellect_good_1 }
					}
					AND = {
						scope:actor = { has_trait = intellect_good_3 }
						scope:recipient = { has_trait = intellect_good_2 }
					}
				}
			}
			duel = {
				skill = learning
				value = 20
				50 = {
					desc = bm_bestow_traits.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = 1.333
						min = -40
						max = 40
					}
					send_interface_toast = {
						type = event_toast_effect_good
						title = bm_bestow_traits.toast.intellect.title
						desc = bm_bestow_traits.toast.intellect.desc
						left_icon = root
						right_icon = scope:recipient
						scope:recipient = {
							bm_bestow_intellect_rank_up_effect = yes
						}
						bm_bestow_traits_grant_xp_and_opinion_effect = yes
						hidden_effect = { bm_recount_lifeforce_stacks_effect = yes }
					}
				}
				50 = {
					desc = bm_bestow_traits.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = -1.333
						min = -40
						max = 40
					}
					send_interface_toast = {
						title = bm_bestow_traits.toast.failure.title
						desc = bm_bestow_traits.toast.failure.desc
						left_icon = root
						right_icon = scope:recipient
						add_stress = minor_stress_gain
						hidden_effect = { bm_recount_lifeforce_stacks_effect = yes }
					}
				}
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = bm_bestow_traits.toast.intellect.title
				desc = bm_bestow_traits.toast.intellect.desc

				}

			}
		}
		ai_chance = { base = 10 }
